52class BaseHstoreDumper(RecursiveDumper):
53 def dump(self, obj: Hstore) -> Buffer | None:
54 if not obj:
55 return b""
56
57 tokens: list[str] = []
58
59 def add_token(s: str) -> None:
60 tokens.append('"')
61 tokens.append(_re_escape.sub(r"\\\1", s))
62 tokens.append('"')
63
64 for k, v in obj.items():
65 if not isinstance(k, str):
66 raise e.DataError("hstore keys can only be strings")
67 add_token(k)
68
69 tokens.append("=>")
70
71 if v is None:
72 tokens.append("NULL")
73 elif not isinstance(v, str):
74 raise e.DataError("hstore keys can only be strings")
75 else:
76 add_token(v)
77
78 tokens.append(",")
79
80 del tokens[-1]
81 data = "".join(tokens)
82 dumper = self._tx.get_dumper(data, PyFormat.TEXT)
83 return dumper.dump(data)

123class HstoreLoader(RecursiveLoader):
124 def load(self, data: Buffer) -> Hstore:
125 loader = self._tx.get_loader(TEXT_OID, self.format)
126 s: str = loader.load(data)
127
128 rv: Hstore = {}
129 start = 0
130 for m in _re_hstore.finditer(s):
131 if m is None or m.start() != start:
132 raise e.DataError(f"error parsing hstore pair at char {start}")
133 k = _re_unescape.sub(r"\1", m.group(1))
134 if (v := m.group(2)) is not None:
135 v = _re_unescape.sub(r"\1", v)
136
137 rv[k] = v
138 start = m.end()
139
140 if start < len(s):
141 raise e.DataError(f"error parsing hstore: unparsed data after char {start}")
142
143 return rv

188def register_hstore(info: TypeInfo, context: AdaptContext | None = None) -> None:
189 """Register the adapters to load and dump hstore.
190
191 :param info: The object with the information about the hstore type.
192 :param context: The context where to register the adapters. If `!None`,
193 register it globally.
194
195 .. note::
196
197 Registering the adapters doesn't affect objects already created, even
198 if they are children of the registered context. For instance,
199 registering the adapter globally doesn't affect already existing
200 connections.
201 """
202 # A friendly error warning instead of an AttributeError in case fetch()
203 # failed and it wasn't noticed.
204 if not info:
205 raise TypeError("no info passed. Is the 'hstore' extension loaded?")
206
207 # Register arrays and type info
208 info.register(context)
209
210 adapters = context.adapters if context else postgres.adapters
211
212 # Generate and register customized dumpers
213 adapters.register_dumper(dict, _make_hstore_dumper(info.oid))
214 adapters.register_dumper(dict, _make_hstore_binary_dumper(info.oid))
215
216 # Register the loaders on the oid
217 adapters.register_loader(info.oid, HstoreLoader)
218 adapters.register_loader(info.oid, HstoreBinaryLoader)
