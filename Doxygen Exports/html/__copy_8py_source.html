178class LibpqWriter(Writer):
179 """
180 An `Writer` to write copy data to a Postgres database.
181 """

190 def write(self, data: Buffer) -> None:
191 if len(data) <= MAX_BUFFER_SIZE:
192 # Most used path: we don't need to split the buffer in smaller
193 # bits, so don't make a copy.
194 self.connection.wait(copy_to(self._pgconn, data, flush=PREFER_FLUSH))
195 else:
196 # Copy a buffer too large in chunks to avoid causing a memory
197 # error in the libpq, which may cause an infinite loop (#255).
198 for i in range(0, len(data), MAX_BUFFER_SIZE):
199 self.connection.wait(
200 copy_to(
201 self._pgconn, data[i : i + MAX_BUFFER_SIZE], flush=PREFER_FLUSH
202 )
203 )
204
205 def finish(self, exc: BaseException | None = None) -> None:
206 bmsg: bytes | None
207 if exc:
208 msg = f"error from Python: {type(exc).__qualname__} - {exc}"
209 bmsg = msg.encode(self._pgconn._encoding, "replace")
210 else:
211 bmsg = None
212
213 try:
214 res = self.connection.wait(copy_end(self._pgconn, bmsg))
215 # The QueryCanceled is expected if we sent an exception message to
216 # pgconn.put_copy_end(). The Python exception that generated that
217 # cancelling is more important, so don't clobber it.
218 except e.QueryCanceled:
219 if not bmsg:
220 raise
221 else:
222 self.cursor._results = [res]

244 """Push data to the server when available from the copy queue.
245
246 Terminate reading when the queue receives a false-y value, or in case
247 of error.

249 The function is designed to be run in a separate task.
250 """

254 except BaseException as ex:

260 # warning: reference loop, broken by _write_end
261 self._worker = spawn(self.worker)

264 if self._worker_error:
