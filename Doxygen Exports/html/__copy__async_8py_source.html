32class AsyncCopy(BaseCopy["AsyncConnection[Any]"]):
33 """Manage an asynchronous :sql:`COPY` operation.
34
35 :param cursor: the cursor where the operation is performed.
36 :param binary: if `!True`, write binary format.
37 :param writer: the object to write to destination. If not specified, write
38 to the `!cursor` connection.
39
40 Choosing `!binary` is not necessary if the cursor has executed a
41 :sql:`COPY` operation, because the operation result describes the format
42 too. The parameter is useful when a `!Copy` object is created manually and
43 no operation is performed on the cursor, such as when using ``writer=``\\
44 `~psycopg.copy.FileWriter`.
45 """

128 async def finish(self, exc: BaseException | None) -> None:
129 """Terminate the copy operation and free the resources allocated.
130
131 You shouldn't need to call this function yourself: it is usually called
132 by exit. It is available if, despite what is documented, you end up
133 using the `Copy` object outside a block.
134 """
135 if self._direction == COPY_IN:
136 if not exc:
137 if data := self.formatter.end():
138 await self._write(data)
139 await self.writer.finish(exc)
140 self._finished = True
141 else:
142 if not exc:
143 return
144
145 if self._pgconn.transaction_status != ACTIVE:
146 # The server has already finished to send copy data. The connection
147 # is already in a good state.
148 return

151 # (which might or might not have been already transferred entirely to
152 # the client, so we won't necessary see the exception associated with
153 # canceling).
154 await self.connection._try_cancel()

177class AsyncLibpqWriter(AsyncWriter):
178 """
179 An `AsyncWriter` to write copy data to a Postgres database.
180 """
181
182 __module__ = "psycopg.copy"
183
184 def __init__(self, cursor: AsyncCursor[Any]):
185 self.cursor = cursor
186 self.connection = cursor.connection
187 self._pgconn = self.connection.pgconn
188
189 async def write(self, data: Buffer) -> None:
190 if len(data) <= MAX_BUFFER_SIZE:
191 # Most used path: we don't need to split the buffer in smaller
192 # bits, so don't make a copy.
193 await self.connection.wait(copy_to(self._pgconn, data, flush=PREFER_FLUSH))
194 else:
195 # Copy a buffer too large in chunks to avoid causing a memory
196 # error in the libpq, which may cause an infinite loop (#255).
197 for i in range(0, len(data), MAX_BUFFER_SIZE):
198 await self.connection.wait(
199 copy_to(
200 self._pgconn, data[i : i + MAX_BUFFER_SIZE], flush=PREFER_FLUSH
201 )
202 )
203
204 async def finish(self, exc: BaseException | None = None) -> None:
205 bmsg: bytes | None
206 if exc:
207 msg = f"error from Python: {type(exc).__qualname__} - {exc}"
208 bmsg = msg.encode(self._pgconn._encoding, "replace")
209 else:
210 bmsg = None
211
212 try:
213 res = await self.connection.wait(copy_end(self._pgconn, bmsg))
214 # The QueryCanceled is expected if we sent an exception message to
215 # pgconn.put_copy_end(). The Python exception that generated that
216 # cancelling is more important, so don't clobber it.
217 except e.QueryCanceled:
218 if not bmsg:
219 raise
220 else:
221 self.cursor._results = [res]

224class AsyncQueuedLibpqWriter(AsyncLibpqWriter):
225 """
226 `AsyncWriter` using a buffer to queue data to write.
227
228 `write()` returns immediately, so that the main thread can be CPU-bound
229 formatting messages, while a worker thread can be IO-bound waiting to write
230 on the connection.
231 """
232
233 __module__ = "psycopg.copy"
234
235 def __init__(self, cursor: AsyncCursor[Any]):
236 super().__init__(cursor)
237
238 self._queue: AQueue[Buffer] = AQueue(maxsize=QUEUE_SIZE)
239 self._worker: AWorker | None = None
240 self._worker_error: BaseException | None = None
241
242 async def worker(self) -> None:
243 """Push data to the server when available from the copy queue.
244
245 Terminate reading when the queue receives a false-y value, or in case
246 of error.
247
248 The function is designed to be run in a separate task.
249 """
250 try:
251 while data := (await self._queue.get()):
252 await self.connection.wait(
253 copy_to(self._pgconn, data, flush=PREFER_FLUSH)
254 )
255 except BaseException as ex:
256 # Propagate the error to the main thread.
257 self._worker_error = ex
258
259 async def write(self, data: Buffer) -> None:
260 if not self._worker:
261 # warning: reference loop, broken by _write_end
262 self._worker = aspawn(self.worker)
263
264 # If the worker thread raies an exception, re-raise it to the caller.
265 if self._worker_error:
266 raise self._worker_error
267
268 if len(data) <= MAX_BUFFER_SIZE:
269 # Most used path: we don't need to split the buffer in smaller
270 # bits, so don't make a copy.
271 await self._queue.put(data)
272 else:
273 # Copy a buffer too large in chunks to avoid causing a memory
274 # error in the libpq, which may cause an infinite loop (#255).
275 for i in range(0, len(data), MAX_BUFFER_SIZE):
276 await self._queue.put(data[i : i + MAX_BUFFER_SIZE])
277
278 async def finish(self, exc: BaseException | None = None) -> None:
279 await self._queue.put(b"")
280
281 if self._worker:
282 await agather(self._worker)
283 self._worker = None # break reference loops if any
284
285 # Check if the worker thread raised any exception before terminating.
286 if self._worker_error:
287 raise self._worker_error
288
289 await super().finish(exc)
