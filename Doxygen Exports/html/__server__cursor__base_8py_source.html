102 def _declare_gen(
103 self, query: Query, params: Params | None = None, binary: bool | None = None
104 ) -> PQGen[None]:
105 """Generator implementing `ServerCursor.execute()`."""

109 # If the cursor is being reused, the previous one must be closed.

131 def _describe_gen(self) -> PQGen[None]:
132 self._pgconn.send_describe_portal(self._name.encode(self._encoding))
133 results = yield from execute(self._pgconn)
134 self._check_results(results)
135 self._results = results

139 def _close_gen(self) -> PQGen[None]:
140 ts = self._conn.pgconn.transaction_status
141
142 # if the connection is not in a sane state, don't even try
143 if ts != IDLE and ts != INTRANS:
144 return
145
146 # If we are IDLE, a WITHOUT HOLD cursor will surely have gone already.
147 if not self._withhold and ts == IDLE:
148 return
149
150 # if we didn't declare the cursor ourselves we still have to close it
151 # but we must make sure it exists.
152 if not self._described:
153 query = sql.SQL(
154 "SELECT 1 FROM pg_catalog.pg_cursors WHERE name = {}"
155 ).format(sql.Literal(self._name))
156 res = yield from self._conn._exec_command(query)
157 # pipeline mode otherwise, unsupported here.
158 assert res is not None
159 if res.ntuples == 0:
160 return
161
162 query = sql.SQL("CLOSE {}").format(sql.Identifier(self._name))
163 yield from self._conn._exec_command(query)

165 def _fetch_gen(self, num: int | None) -> PQGen[list[Row]]:
166 if self.closed:
167 raise e.InterfaceError("the cursor is closed")
168 # If we are stealing the cursor, make sure we know its shape
169 if not self._described:
170 yield from self._start_query()
171 yield from self._describe_gen()
172
173 query = sql.SQL("FETCH FORWARD {} FROM {}").format(
174 sql.SQL("ALL") if num is None else sql.Literal(num),
175 sql.Identifier(self._name),
176 )
177 res = yield from self._conn._exec_command(query, result_format=self._format)
178 # pipeline mode otherwise, unsupported here.
179 assert res is not None
180
181 self.pgresult = res
182 self._tx.set_pgresult(res, set_loaders=False)
183 return self._tx.load_rows(0, res.ntuples, self._make_row)

185 def _scroll_gen(self, value: int, mode: str) -> PQGen[None]:
186 if mode not in ("relative", "absolute"):
187 raise ValueError(f"bad mode: {mode}. It should be 'relative' or 'absolute'")
188 query = sql.SQL("MOVE{} {} FROM {}").format(
189 sql.SQL(" ABSOLUTE" if mode == "absolute" else ""),
190 sql.Literal(value),
191 sql.Identifier(self._name),
192 )
193 yield from self._conn._exec_command(query)

195 def _make_declare_statement(self, query: Query) -> Query:
196 parts = [sql.SQL("DECLARE"), sql.Identifier(self._name)]
197 if self._scrollable is not None:
198 parts.append(sql.SQL("SCROLL" if self._scrollable else "NO SCROLL"))
199 parts.append(sql.SQL("CURSOR"))
200 if self._withhold:
201 parts.append(sql.SQL("WITH HOLD"))
202 parts.append(sql.SQL("FOR "))
203 declare = sql.SQL(" ").join(parts)
204
205 if isinstance(query, Template):
206 # t"{declare:q}{query:q}" but compatible with Python < 3.14
207 return Template(
208 Interpolation(declare, "declare", None, "q"),
209 Interpolation(query, "query", None, "q"),
210 )
211
212 if isinstance(query, bytes):
213 query = query.decode(self._encoding)
214 if not isinstance(query, sql.Composable):
215 query = sql.SQL(query)
216
217 return declare + query
