53 def _find_list_element(self, L: list[Any], format: PyFormat) -> Any:
54 """
55 Find the first non-null element of an eventually nested list
56 """
57 items = list(self._flatiter(L, set()))
58 if not (types := {type(item): item for item in items}):
59 return None
60
61 if len(types) == 1:
62 t, v = types.popitem()
63 else:
64 # More than one type in the list. It might be still good, as long
65 # as they dump with the same oid (e.g. IPv4Network, IPv6Network).
66 dumpers = [self._tx.get_dumper(item, format) for item in types.values()]
67 if len({d.oid for d in dumpers}) == 1:
68 t, v = types.popitem()
69 else:
70 raise e.DataError(
71 "cannot dump lists of mixed types;"
72 f" got: {', '.join(sorted(t.__name__ for t in types))}"
73 )
74
75 # Checking for precise type. If the type is a subclass (e.g. Int4)
76 # we assume the user knows what type they are passing.
77 if t is not int:
78 return v
79
80 # If we got an int, let's see what is the biggest one in order to
81 # choose the smallest OID and allow Postgres to do the right cast.
82 imax: int = max(items)
83 imin: int = min(items)
84 if imin >= 0:
85 return imax
86 else:
87 return max(imax, -imin - 1)

89 def _flatiter(self, L: list[Any], seen: set[int]) -> Any:
90 if id(L) in seen:
91 raise e.DataError("cannot dump a recursive list")
92
93 seen.add(id(L))
94
95 for item in L:
96 if type(item) is list:
97 yield from self._flatiter(item, seen)
98 elif item is not None:
99 yield item
100
101 return None

116class ListDumper(BaseListDumper):
117 delimiter = b","
118
119 def get_key(self, obj: list[Any], format: PyFormat) -> DumperKey:
120 if self.oid:
121 return self.cls
122
123 if (item := self._find_list_element(obj, format)) is None:
124 return self.cls
125
126 sd = self._tx.get_dumper(item, format)
127 return (self.cls, sd.get_key(item, format))
128
129 def upgrade(self, obj: list[Any], format: PyFormat) -> BaseListDumper:
130 # If we have an oid we don't need to upgrade
131 if self.oid:
132 return self
133
134 if (item := self._find_list_element(obj, format)) is None:
135 # Empty lists can only be dumped as text if the type is unknown.
136 return self
137
138 sd = self._tx.get_dumper(item, PyFormat.from_pq(self.format))
139 dumper = type(self)(self.cls, self._tx)
140 dumper.sub_dumper = sd
141
142 # We consider an array of unknowns as unknown, so we can dump empty
143 # lists or lists containing only None elements.
144 if sd.oid != INVALID_OID:
145 info = self._get_base_type_info(sd.oid)
146 dumper.oid = info.array_oid or TEXT_ARRAY_OID
147 dumper.delimiter = info.delimiter.encode()
148 else:
149 dumper.oid = INVALID_OID
150
151 return dumper
152
153 # Double quotes and backslashes embedded in element values will be
154 # backslash-escaped.
155 _re_esc = re.compile(rb'(["\\])')
156
157 def dump(self, obj: list[Any]) -> Buffer | None:
158 tokens: list[Buffer] = []
159 needs_quotes = _get_needs_quotes_regexp(self.delimiter).search
160
161 def dump_list(obj: list[Any]) -> None:
162 if not obj:
163 tokens.append(b"{}")
164 return
165
166 tokens.append(b"{")
167 for item in obj:
168 if isinstance(item, list):
169 dump_list(item)
170 elif item is not None:
171 if (ad := self._dump_item(item)) is None:
172 tokens.append(b"NULL")
173 else:
174 if needs_quotes(ad):
175 if not isinstance(ad, bytes):
176 ad = bytes(ad)
177 ad = b'"' + self._re_esc.sub(rb"\\\1", ad) + b'"'
178 tokens.append(ad)
179 else:
180 tokens.append(b"NULL")
181
182 tokens.append(self.delimiter)
183
184 tokens[-1] = b"}"
185
186 dump_list(obj)
187
188 return b"".join(tokens)
189
190 def _dump_item(self, item: Any) -> Buffer | None:
191 if self.sub_dumper:
192 return self.sub_dumper.dump(item)
193 else:
194 return self._tx.get_dumper(item, PY_TEXT).dump(item)

197@cache
198def _get_needs_quotes_regexp(delimiter: bytes) -> re.Pattern[bytes]:
199 """Return a regexp to recognise when a value needs quotes
200
201 from https://www.postgresql.org/docs/current/arrays.html#ARRAYS-IO
202
203 The array output routine will put double quotes around element values if
204 they are empty strings, contain curly braces, delimiter characters,
205 double quotes, backslashes, or white space, or match the word NULL.
206 """
207 return re.compile(
208 rb"""(?xi)
209 ^$ # the empty string
210 | ["{}%s\\\s] # or a char to escape
211 | ^null$ # or the word NULL
212 """
213 % delimiter
214 )

246 def dump(self, obj: list[Any]) -> Buffer | None:
247 # Postgres won't take unknown for element oid: fall back on text
248 sub_oid = self.sub_dumper and self.sub_dumper.oid or TEXT_OID

250 if not obj:
251 return _pack_head(0, 0, sub_oid)

253 data: list[Buffer] = [b"", b""] # placeholders to avoid a resize
254 dims: list[int] = []
255 hasnull = 0

257 def calc_dims(L: list[Any]) -> None:
258 if isinstance(L, self.cls):
259 if not L:
260 raise e.DataError("lists cannot contain empty lists")
261 dims.append(len(L))
262 calc_dims(L[0])

264 calc_dims(obj)

266 def dump_list(L: list[Any], dim: int) -> None:
267 nonlocal hasnull
268 if len(L) != dims[dim]:
269 raise e.DataError("nested lists have inconsistent lengths")
270
271 if dim == len(dims) - 1:
272 for item in L:
273 if item is not None:
274 # If we get here, the sub_dumper must have been set
275 item = self.sub_dumper.dump(item) # type: ignore[union-attr]
276 if item is not None:
277 data.append(pack_len(len(item)))
278 data.append(item)
279 else:
280 hasnull = 1
281 data.append(ARRAY_NULL)
282 else:
283 for item in L:
284 if not isinstance(item, self.cls):
285 raise e.DataError("nested lists have inconsistent depths")
286 dump_list(item, dim + 1) # type: ignore
287
288 dump_list(obj, 0)
289
290 data[0] = _pack_head(len(dims), hasnull, sub_oid)
291 data[1] = b"".join(_pack_dim(dim, 1) for dim in dims)
292 return b"".join(data)

378def _load_text(
379 data: Buffer,
380 loader: Loader,
381 delimiter: bytes = b",",
382 __re_unescape: re.Pattern[bytes] = re.compile(rb"\\(.)"),
383) -> list[Any]:
384 rv = None
385 stack: list[Any] = []
386 a: list[Any] = []
387 rv = a
388 load = loader.load
389
390 # Remove the dimensions information prefix (``[...]=``)
391 if data and data[0] == b"["[0]:
392 if isinstance(data, memoryview):
393 data = bytes(data)
394
395 if (idx := data.find(b"=")) == -1:
396 raise e.DataError("malformed array: no '=' after dimension information")
397 data = data[idx + 1 :]
398
399 re_parse = _get_array_parse_regexp(delimiter)
400 for m in re_parse.finditer(data):
401 if (t := m.group(1)) == b"{":
402 if stack:
403 stack[-1].append(a)
404 stack.append(a)
405 a = []
406
407 elif t == b"}":
408 if not stack:
409 raise e.DataError("malformed array: unexpected '}'")
410 rv = stack.pop()
411
412 else:
413 if not stack:
414 wat = t[:10].decode("utf8", "replace") + "..." if len(t) > 10 else ""
415 raise e.DataError(f"malformed array: unexpected '{wat}'")
416 if t == b"NULL":
417 v = None
418 else:
419 if t.startswith(b'"'):
420 t = __re_unescape.sub(rb"\1", t[1:-1])
421 v = load(t)
422
423 stack[-1].append(v)
424
425 assert rv is not None
426 return rv

438 | [^"{}%s\\]+ # or an unquoted non-empty string
439 ) ,?
440 """
441 % delimiter

445def _load_binary(data: Buffer, tx: Transformer) -> list[Any]:
446 ndims, hasnull, oid = _unpack_head(data)
447 load = tx.get_loader(oid, PQ_BINARY).load
448
449 if not ndims:
450 return []
451
452 p = 12 + 8 * ndims
453 dims = [_unpack_dim(data, i)[0] for i in range(12, p, 8)]
454 nelems = prod(dims)
455
456 out: list[Any] = [None] * nelems
457 for i in range(nelems):
458 size = unpack_len(data, p)[0]
459 p += 4
460 if size == -1:
461 continue
462 out[i] = load(data[p : p + size])
463 p += size
464
465 # fon ndims > 1 we have to aggregate the array into sub-arrays
466 for dim in dims[-1:0:-1]:
467 out = [out[i : i + dim] for i in range(0, len(out), dim)]
468
469 return out
