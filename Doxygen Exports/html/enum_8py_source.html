38class EnumInfo(TypeInfo):
39 """Manage information about an enum type."""
40
41 def __init__(
42 self,
43 name: str,
44 oid: int,
45 array_oid: int,
46 labels: Sequence[str],
47 ):

53 @classmethod
54 def _get_info_query(cls, conn: BaseConnection[Any]) -> QueryNoTemplate:
55 return sql.SQL(
56 """\
57SELECT name, oid, array_oid, array_agg(label) AS labels
58FROM (
59 SELECT
60 t.typname AS name, t.oid AS oid, t.typarray AS array_oid,
61 e.enumlabel AS label
62 FROM pg_type t
63 LEFT JOIN pg_enum e
64 ON e.enumtypid = t.oid
65 WHERE t.oid = {regtype}
66 ORDER BY e.enumsortorder
67) x
68GROUP BY name, oid, array_oid
69"""
70 ).format(regtype=cls._to_regtype(conn))

124def register_enum(
125 info: EnumInfo,
126 context: AdaptContext | None = None,
127 enum: type[E] | None = None,
128 *,
129 mapping: EnumMapping[E] = None,
130) -> None:
131 """Register the adapters to load and dump a enum type.
132
133 :param info: The object with the information about the enum to register.
134 :param context: The context where to register the adapters. If `!None`,
135 register it globally.
136 :param enum: Python enum type matching to the PostgreSQL one. If `!None`,
137 a new enum will be generated and exposed as `EnumInfo.enum`.
138 :param mapping: Override the mapping between `!enum` members and `!info`
139 labels.
140 """
141
142 if not info:
143 raise TypeError("no info passed. Is the requested enum available?")
144
145 if enum is None:
146 enum = cast(type[E], _make_enum(info.name, tuple(info.labels)))
147
148 info.enum = enum
149 adapters = context.adapters if context else postgres.adapters
150 info.register(context)
151
152 load_map = _make_load_map(info, enum, mapping, context)
153
154 loader = _make_loader(info.name, info.enum, load_map)
155 adapters.register_loader(info.oid, loader)
156
157 loader = _make_binary_loader(info.name, info.enum, load_map)
158 adapters.register_loader(info.oid, loader)
159
160 dump_map = _make_dump_map(info, enum, mapping, context)
161
162 dumper = _make_dumper(info.enum, info.oid, dump_map)
163 adapters.register_dumper(info.enum, dumper)
164
165 dumper = _make_binary_dumper(info.enum, info.oid, dump_map)
166 adapters.register_dumper(info.enum, dumper)

210def _make_load_map(
211 info: EnumInfo, enum: type[E], mapping: EnumMapping[E], context: AdaptContext | None
212) -> _HEnumLoadMap[E]:
213 enc = conn_encoding(context.connection if context else None)
214 rv = []
215 for label in info.labels:
216 try:
217 member = enum[label]
218 except KeyError:
219 # tolerate a missing enum, assuming it won't be used. If it is we
220 # will get a DataError on fetch.
221 pass
222 else:
223 rv.append((label.encode(enc), member))
224
225 if mapping:
226 if isinstance(mapping, Mapping):
227 mapping = list(mapping.items())
228
229 for member, label in mapping:
230 rv.append((label.encode(enc), member))
231
232 return tuple(rv)

235def _make_dump_map(
236 info: EnumInfo, enum: type[E], mapping: EnumMapping[E], context: AdaptContext | None
237) -> _HEnumDumpMap[E]:
238 enc = conn_encoding(context.connection if context else None)
239 rv = []
240 for member in enum:
241 rv.append((member, member.name.encode(enc)))
242
243 if mapping:
244 if isinstance(mapping, Mapping):
245 mapping = list(mapping.items())
246
247 for member, label in mapping:
248 rv.append((member, label.encode(enc)))
249
250 return tuple(rv)
