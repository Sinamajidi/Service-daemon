19def split_attempts(params: ConnMapping) -> list[ConnDict]:
20 """
21 Split connection parameters with a sequence of hosts into separate attempts.
22 """
23
24 def split_val(key: str) -> list[str]:
25 val = get_param(params, key)
26 return val.split(",") if val else []
27
28 hosts = split_val("host")
29 hostaddrs = split_val("hostaddr")
30 ports = split_val("port")
31
32 if hosts and hostaddrs and len(hosts) != len(hostaddrs):
33 raise e.OperationalError(
34 f"could not match {len(hosts)} host names"
35 f" with {len(hostaddrs)} hostaddr values"
36 )
37
38 nhosts = max(len(hosts), len(hostaddrs))
39
40 if 1 < len(ports) != nhosts:
41 raise e.OperationalError(
42 f"could not match {len(ports)} port numbers to {len(hosts)} hosts"
43 )
44
45 # A single attempt to make. Don't mangle the conninfo string.
46 if nhosts <= 1:
47 return [{**params}]
48
49 if len(ports) == 1:
50 ports *= nhosts
51
52 # Now all lists are either empty or have the same length
53 rv = []
54 for i in range(nhosts):
55 attempt = {**params}
56 if hosts:
57 attempt["host"] = hosts[i]
58 if hostaddrs:
59 attempt["hostaddr"] = hostaddrs[i]
60 if ports:
61 attempt["port"] = ports[i]
62 rv.append(attempt)
63
64 return rv

67def get_param(params: ConnMapping, name: str) -> str | None:
68 """
69 Return a value from a connection string.

71 The value may be also specified in a PG* env var.
72 """

98def get_param_def(keyword: str, _cache: dict[str, ParamDef] = {}) -> ParamDef | None:
99 """
100 Return the ParamDef of a connection string parameter.
101 """
102 if not _cache:
103 defs = pq.Conninfo.get_defaults()
104 for d in defs:
105 cd = ParamDef(
106 keyword=d.keyword.decode(),
107 envvar=d.envvar.decode() if d.envvar else "",
108 compiled=d.compiled.decode() if d.compiled is not None else None,
109 )
110 _cache[cd.keyword] = cd
111
112 return _cache.get(keyword)
