37class RowMaker(Protocol[Row]):
38 """
39 Callable protocol taking a sequence of value and returning an object.
40
41 The sequence of value is what is returned from a database query, already
42 adapted to the right Python types. The return value is the object that your
43 program would like to receive: by default (`tuple_row()`) it is a simple
44 tuple, but it may be any type of object.
45
46 Typically, `!RowMaker` functions are returned by `RowFactory`.
47 """
48
49 def __call__(self, __values: Sequence[Any]) -> Row: ...

52class RowFactory(Protocol[Row]):
53 """
54 Callable protocol taking a `~psycopg.Cursor` and returning a `RowMaker`.
55
56 A `!RowFactory` is typically called when a `!Cursor` receives a result.
57 This way it can inspect the cursor state (for instance the
58 `~psycopg.Cursor.description` attribute) and help a `!RowMaker` to create
59 a complete object.
60
61 For instance the `dict_row()` `!RowFactory` uses the names of the column to
62 define the dictionary key and returns a `!RowMaker` function which would
63 use the values to create a dictionary for each record.
64 """
65
66 def __call__(self, __cursor: Cursor[Any]) -> RowMaker[Row]: ...

129def namedtuple_row(cursor: BaseCursor[Any, Any]) -> RowMaker[NamedTuple]:
130 """Row factory to represent rows as `~collections.namedtuple`.
131
132 The field names are taken from the column names of the returned columns,
133 with some mangling to deal with invalid names.
134 """
135 if (res := cursor.pgresult) and (nfields := _get_nfields(res)) is not None:
136 nt = _make_nt(cursor._encoding, *(res.fname(i) for i in range(nfields)))
137 return nt._make
138 else:
139 return no_result

142@functools.lru_cache(512)
143def _make_nt(enc: str, *names: bytes) -> type[NamedTuple]:
144 snames = tuple(_as_python_identifier(n.decode(enc)) for n in names)
145 return namedtuple("Row", snames) # type: ignore[return-value]

247def _get_nfields(res: PGresult) -> int | None:
248 """
249 Return the number of columns in a result, if it returns tuples else None
250
251 Take into account the special case of results with zero columns.
252 """
253 nfields = res.nfields
254
255 if (
256 res.status == TUPLES_OK
257 or res.status == SINGLE_TUPLE
258 or res.status == TUPLES_CHUNK
259 # "describe" in named cursors
260 or (res.status == COMMAND_OK and nfields)
261 ):
262 return nfields
263 else:
264 return None
