275 def _maybe_prepare_gen(
276 self,
277 pgq: PostgresQuery,
278 *,
279 prepare: bool | None = None,
280 binary: bool | None = None,
281 ) -> PQGen[None]:
282 # Check if the query is prepared or needs preparing
283 prep, name = self._get_prepared(pgq, prepare)
284 if prep is Prepare.NO:
285 # The query must be executed without preparing
286 self._execute_send(pgq, binary=binary)
287 else:
288 # If the query is not already prepared, prepare it.
289 if prep is Prepare.SHOULD:
290 self._send_prepare(name, pgq)
291 if not self._conn._pipeline:
292 (result,) = yield from execute(self._pgconn)
293 if result.status == FATAL_ERROR:
294 raise e.error_from_result(result, encoding=self._encoding)
295 # Then execute it.
296 self._send_query_prepared(name, pgq, binary=binary)
297
298 # Update the prepare state of the query.
299 # If an operation requires to flush our prepared statements cache,
300 # it will be added to the maintenance commands to execute later.
301 key = self._conn._prepared.maybe_add_to_cache(pgq, prep, name)
302
303 if self._conn._pipeline:
304 queued = None
305 if key is not None:
306 queued = (key, prep, name)
307 self._conn._pipeline.result_queue.append((self, queued))
308 return
309
310 # run the query
311 results = yield from execute(self._pgconn)
312
313 if key is not None:
314 self._conn._prepared.validate(key, prep, name, results)
315
316 self._check_results(results)
317 self._set_results(results)

324 def _stream_send_gen(
325 self,
326 query: Query,
327 params: Params | None = None,
328 *,
329 binary: bool | None = None,
330 size: int,
331 ) -> PQGen[None]:
332 """Generator to send the query for `Cursor.stream()`."""
333 yield from self._start_query(query)
334 pgq = self._convert_query(query, params)
335 self._execute_send(pgq, binary=binary, force_extended=True)
336 if size < 1:
337 raise ValueError("size must be >= 1")
338 elif size == 1:
339 self._pgconn.set_single_row_mode()
340 else:
341 capabilities.has_stream_chunked(check=True)
342 self._pgconn.set_chunked_rows_mode(size)
343 self._last_query = query
344 yield from send(self._pgconn)

346 def _stream_fetchone_gen(self, first: bool) -> PQGen[PGresult | None]:
347 res: PGresult | None = yield from fetch(self._pgconn)
348 if res is None:
349 return None
350
351 status = res.status
352 if status == SINGLE_TUPLE or status == TUPLES_CHUNK:
353 self.pgresult = res
354 self._tx.set_pgresult(res, set_loaders=first)
355 if first:
356 self._make_row = self._make_row_maker()
357 return res
358
359 elif status == TUPLES_OK or status == COMMAND_OK:
360 # End of single row results
361 while res:
362 res = yield from fetch(self._pgconn)
363 if status != TUPLES_OK:
364 raise e.ProgrammingError(
365 "the operation in stream() didn't produce a result"
366 )
367 return None
368
369 else:
370 # Errors, unexpected values
371 return self._raise_for_result(res)

373 def _start_query(self, query: Query | None = None) -> PQGen[None]:
374 """Generator to start the processing of a query.
375
376 It is implemented as generator because it may send additional queries,
377 such as `begin`.
378 """

380 raise e.InterfaceError("the cursor is closed")
381
382 self._reset()
383 if not self._last_query or (self._last_query is not query):
384 self._last_query = None

388 def _start_copy_gen(
389 self, statement: Query, params: Params | None = None
390 ) -> PQGen[None]:
391 """Generator implementing sending a command for `Cursor.copy()."""
392
393 # The connection gets in an unrecoverable state if we attempt COPY in
394 # pipeline mode. Forbid it explicitly.
395 if self._conn._pipeline:
396 raise e.NotSupportedError("COPY cannot be used in pipeline mode")
397
398 yield from self._start_query()
399
400 # Merge the params client-side
401 if params or isinstance(statement, Template):
402 pgq = PostgresClientQuery(self._tx)
403 pgq.convert(statement, params)
404 statement = pgq.query
405
406 query = self._convert_query(statement)
407
408 self._execute_send(query, binary=False)
409 if len(results := (yield from execute(self._pgconn))) != 1:
410 raise e.ProgrammingError("COPY cannot be mixed with other operations")
411
412 self._check_copy_result(results[0])
413 self._set_results(results)

415 def _execute_send(
416 self,
417 query: PostgresQuery,
418 *,
419 force_extended: bool = False,
420 binary: bool | None = None,
421 ) -> None:
422 """
423 Implement part of execute() before waiting common to sync and async.
424
425 This is not a generator, but a normal non-blocking function.
426 """
427 if binary is None:
428 fmt = self.format
429 else:
430 fmt = BINARY if binary else TEXT
431
432 self._query = query
433
434 if self._conn._pipeline:
435 # In pipeline mode always use PQsendQueryParams - see #314
436 # Multiple statements in the same query are not allowed anyway.
437 self._conn._pipeline.command_queue.append(
438 partial(
439 self._pgconn.send_query_params,
440 query.query,
441 query.params,
442 param_formats=query.formats,
443 param_types=query.types,
444 result_format=fmt,
445 )
446 )
447 elif force_extended or query.params or fmt == BINARY:
448 self._pgconn.send_query_params(
449 query.query,
450 query.params,
451 param_formats=query.formats,
452 param_types=query.types,
453 result_format=fmt,
454 )
455 else:
456 # If we can, let's use simple query protocol,
457 # as it can execute more than one statement in a single query.
458 self._pgconn.send_query(query.query)

467 def _check_results(self, results: list[PGresult]) -> None:
468 """
469 Verify that the results of a query are valid.
470
471 Verify that the query returned at least one result and that they all
472 represent a valid result from the database.
473 """
474 if not results:
475 raise e.InternalError("got no result from the query")
476
477 for res in results:
478 if (
479 (status := res.status) != TUPLES_OK
480 and status != COMMAND_OK
481 and status != EMPTY_QUERY
482 and status != COPY_BOTH
483 ):
484 self._raise_for_result(res)

486 def _raise_for_result(self, result: PGresult) -> NoReturn:
487 """
488 Raise an appropriate error message for an unexpected database result
489 """
490 if (status := result.status) == FATAL_ERROR:
491 raise e.error_from_result(result, encoding=self._encoding)
492 elif status == PIPELINE_ABORTED:
493 raise e.PipelineAborted("pipeline aborted")
494 elif status == COPY_IN or status == COPY_OUT or status == COPY_BOTH:
495 raise e.ProgrammingError(
496 "COPY cannot be used with this method; use copy() instead"
497 )
498 else:
499 raise e.InternalError(
500 "unexpected result status from query:" f" {pq.ExecStatus(status).name}"
501 )

503 def _select_current_result(self, i: int, format: pq.Format | None = None) -> None:
504 """
505 Select one of the results in the cursor as the active one.
506 """
507 self._iresult = i
508 res = self.pgresult = self._results[i]
509
510 # Note: the only reason to override format is to correctly set
511 # binary loaders on server-side cursors, because send_describe_portal
512 # only returns a text result.
513 self._tx.set_pgresult(res, format=format)
514
515 self._pos = 0
516
517 if res.status == TUPLES_OK:
518 self._rowcount = self.pgresult.ntuples
519
520 # COPY_OUT has never info about nrows. We need such result for the
521 # columns in order to return a `description`, but not overwrite the
522 # cursor rowcount (which was set by the Copy object).
523 elif res.status != COPY_OUT:
524 nrows = self.pgresult.command_tuples
525 self._rowcount = nrows if nrows is not None else -1
526
527 self._make_row = self._make_row_maker()
528
529 def _set_results(self, results: list[PGresult]) -> None:
530 if self._execmany_returning is None:
531 # Received from execute()
532 self._results[:] = results
533 self._select_current_result(0)
534 # Received from executemany()
535 elif self._execmany_returning:
536 first_batch = not self._results
537 self._results.extend(results)
538 if first_batch:
539 self._select_current_result(0)
540 else:
541 # In non-returning case, set rowcount to the cumulated number of
542 # rows of executed queries.
543 for res in results:
544 self._rowcount += res.command_tuples or 0

601 def _check_result_for_fetch(self) -> PGresult:
602 if self.closed:
603 raise e.InterfaceError("the cursor is closed")
604
605 if not (res := self.pgresult):
606 raise e.ProgrammingError("no result available")
607
608 if (status := res.status) == TUPLES_OK:
609 return res
610 elif status == FATAL_ERROR:
611 raise e.error_from_result(res, encoding=self._encoding)
612 elif status == PIPELINE_ABORTED:
613 raise e.PipelineAborted("pipeline aborted")
614 else:
615 if res.command_status:
616 detail = f" (command status: {res.command_status.decode()})"
617 else:
618 try:
619 status_name = pq.ExecStatus(status).name
620 except ValueError:
621 status_name = f"{status} - unknown"
622 detail = f" (result status: {status_name})"
623 raise e.ProgrammingError(
624 f"the last operation didn't produce records{detail}"
625 )

627 def _check_copy_result(self, result: PGresult) -> None:
628 """
629 Check that the value returned in a copy() operation is a legit COPY.
630 """
631 status = result.status
632 if status == COPY_IN or status == COPY_OUT:
633 return
634 elif status == FATAL_ERROR:
635 raise e.error_from_result(result, encoding=self._encoding)
636 else:
637 raise e.ProgrammingError(
638 "copy() should be used only with COPY ... TO STDOUT or COPY ..."
639 f" FROM STDIN statements, got {pq.ExecStatus(status).name}"

642 def _scroll(self, value: int, mode: str) -> None:
643 res = self._check_result_for_fetch()
644 if mode == "relative":
645 newpos = self._pos + value
646 elif mode == "absolute":
647 newpos = value
648 else:
649 raise ValueError(f"bad mode: {mode}. It should be 'relative' or 'absolute'")
650 if not 0 <= newpos < res.ntuples:
651 raise IndexError("position out of bound")
652 self._pos = newpos
