103 def resolve(self, params: dict[str, Any]) -> dict[str, Any]:
104 """Update the parameters host and port after SRV lookup."""
105 if not (attempts := self._get_attempts(params)):
106 return params
107
108 hps = []
109 for hp in attempts:
110 if hp.totry:
111 hps.extend(self._resolve_srv(hp))
112 else:
113 hps.append(hp)
114
115 return self._return_params(params, hps)

131 def _get_attempts(self, params: dict[str, Any]) -> list[HostPort]:
132 """
133 Return the list of host, and for each host if SRV lookup must be tried.
134
135 Return an empty list if no lookup is requested.
136 """
137 # If hostaddr is defined don't do any resolution.
138 if params.get("hostaddr", os.environ.get("PGHOSTADDR", "")):
139 return []
140
141 host_arg: str = params.get("host", os.environ.get("PGHOST", ""))
142 hosts_in = host_arg.split(",")
143 port_arg: str = str(params.get("port", os.environ.get("PGPORT", "")))
144 if len(ports_in := port_arg.split(",")) == 1:
145 # If only one port is specified, it applies to all the hosts.
146 ports_in *= len(hosts_in)
147 if len(ports_in) != len(hosts_in):
148 # ProgrammingError would have been more appropriate, but this is
149 # what the raise if the libpq fails connect in the same case.
150 raise e.OperationalError(
151 f"cannot match {len(hosts_in)} hosts with {len(ports_in)} port numbers"
152 )
153
154 out = []
155 srv_found = False
156 for host, port in zip(hosts_in, ports_in):
157 if (m := self.re_srv_rr.match(host)) or port.lower() == "srv":
158 srv_found = True
159 target = m.group("target") if m else None
160 hp = HostPort(host=host, port=port, totry=True, target=target)
161 else:
162 hp = HostPort(host=host, port=port)
163 out.append(hp)
164
165 return out if srv_found else []

181 def _get_solved_entries(
182 self, hp: HostPort, entries: Sequence[SRV]
183 ) -> list[HostPort]:
184 if not entries:
185 # No SRV entry found. Delegate the libpq a QNAME=target lookup
186 if hp.target and hp.port.lower() != "srv":
187 return [HostPort(host=hp.target, port=hp.port)]
188 else:
189 return []
190
191 # If there is precisely one SRV RR, and its Target is "." (the root
192 # domain), abort.
193 if len(entries) == 1 and str(entries[0].target) == ".":
194 return []
195
196 return [
197 HostPort(host=str(entry.target).rstrip("."), port=str(entry.port))
198 for entry in self.sort_rfc2782(entries)
199 ]

201 def _return_params(
202 self, params: dict[str, Any], hps: list[HostPort]
203 ) -> dict[str, Any]:
204 if not hps:
205 # Nothing found, we ended up with an empty list
206 raise e.OperationalError("no host found after SRV RR lookup")
207
208 out = params.copy()
209 out["host"] = ",".join(hp.host for hp in hps)
210 out["port"] = ",".join(str(hp.port) for hp in hps)
211 return out

213 def sort_rfc2782(self, ans: Sequence[SRV]) -> list[SRV]:
214 """
215 Implement the priority/weight ordering defined in RFC 2782.
216 """
217 # Divide the entries by priority:
218 priorities: DefaultDict[int, list[SRV]] = defaultdict(list)
219 out: list[SRV] = []
220 for entry in ans:
221 priorities[entry.priority].append(entry)
222
223 for pri, entries in sorted(priorities.items()):
224 if len(entries) == 1:
225 out.append(entries[0])
226 continue
227
228 entries.sort(key=lambda ent: ent.weight)
229 total_weight = sum(ent.weight for ent in entries)
230 while entries:
231 r = randint(0, total_weight)
232 csum = 0
233 for i, ent in enumerate(entries):
234 csum += ent.weight
235 if csum >= r:
236 break
237 out.append(ent)
238 total_weight -= ent.weight
239 del entries[i]
240
241 return out
