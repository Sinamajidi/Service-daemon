465 def get_booking_with_details(self, booking_id: Union[str, UUID]) -> Optional[Dict[str, Any]]:
466 """
467 Get booking with all related information (tenant, unit, service type).
468 Demonstrates complex JOIN processing in database.
469 """
470 query = """
471 SELECT
472 b.*,
473 t.id as tenant_id, u.full_name as tenant_name, u.email as tenant_email,
474 un.unit_number, un.building_id,
475 st.name as service_type_name, st.code as service_type_code,
476 a.provider_id, a.provider_staff_id, a.status as assignment_status
477 FROM bookings b
478 LEFT JOIN tenants t ON b.tenant_id = t.id
479 LEFT JOIN users u ON t.user_id = u.id
480 LEFT JOIN units un ON b.unit_id = un.id
481 LEFT JOIN service_types st ON b.service_type_id = st.id
482 LEFT JOIN assignments a ON b.id = a.booking_id
483 WHERE b.id = %s
484 """
485 return self.dal.execute_custom_query(query, (str(booking_id),), fetch='one')
