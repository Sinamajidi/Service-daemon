56def wait_selector(gen: PQGen[RV], fileno: int, interval: float = 0.0) -> RV:
57 """
58 Wait for a generator using the best strategy available.
59
60 :param gen: a generator performing database operations and yielding
61 `Ready` values when it would block.
62 :param fileno: the file descriptor to wait on.
63 :param interval: interval (in seconds) to check for other interrupt, e.g.
64 to allow Ctrl-C.
65 :return: whatever `!gen` returns on completion.
66
67 Consume `!gen`, scheduling `fileno` for completion when it is reported to
68 block. Once ready again send the ready state back to `!gen`.
69 """
70 if interval is None:
71 raise ValueError("indefinite wait not supported anymore")
72 try:
73 s = next(gen)
74 with DefaultSelector() as sel:
75 sel.register(fileno, s)
76 while True:
77 if not (rlist := sel.select(timeout=interval)):
78 # Check if it was a timeout or we were disconnected
79 _check_fd_closed(fileno)
80 gen.send(READY_NONE)
81 continue
82
83 ready = rlist[0][1]
84 s = gen.send(ready)
85 sel.register(fileno, s)
86
87 except StopIteration as ex:
88 rv: RV = ex.value
89 return rv

126async def wait_async(gen: PQGen[RV], fileno: int, interval: float = 0.0) -> RV:
127 """
128 Coroutine waiting for a generator to complete.
129
130 :param gen: a generator performing database operations and yielding
131 `Ready` values when it would block.
132 :param fileno: the file descriptor to wait on.
133 :param interval: interval (in seconds) to check for other interrupt, e.g.
134 to allow Ctrl-C.
135 :return: whatever `!gen` returns on completion.
136
137 Behave like in `wait()`, but exposing an `asyncio` interface.
138 """
139 if interval is None:
140 raise ValueError("indefinite wait not supported anymore")
141
142 # Use an event to block and restart after the fd state changes.
143 # Not sure this is the best implementation but it's a start.
144 ev = Event()
145 loop = get_event_loop()
146 ready: int
147 s: Wait
148
149 def wakeup(state: Ready) -> None:
150 nonlocal ready
151 ready |= state
152 ev.set()
153
154 try:
155 s = next(gen)
156 while True:
157 reader = s & WAIT_R
158 writer = s & WAIT_W
159 if not (reader or writer):
160 raise e.InternalError(f"bad poll status: {s}")
161 ev.clear()
162 ready = 0
163 if reader:
164 loop.add_reader(fileno, wakeup, READY_R)
165 if writer:
166 loop.add_writer(fileno, wakeup, READY_W)
167 try:
168 try:
169 await wait_for(ev.wait(), interval)
170 except TimeoutError:
171 pass
172 finally:
173 if reader:
174 loop.remove_reader(fileno)
175 if writer:
176 loop.remove_writer(fileno)
177 s = gen.send(ready)
178
179 except OSError as ex:
180 # Assume the connection was closed
181 raise e.OperationalError("connection socket closed") from ex
182 except StopIteration as ex:
183 rv: RV = ex.value
184 return rv

252 """
253 Wait for a generator using select where supported.
254
255 BUG: on Linux, can't select on FD >= 1024. On Windows it's fine.
256 """
257 if interval is None:
258 raise ValueError("indefinite wait not supported anymore")
259 try:
260 s = next(gen)
261
262 empty = ()
263 fnlist = (fileno,)
264 while True:
265 rl, wl, xl = select.select(
266 fnlist if s & WAIT_R else empty,
267 fnlist if s & WAIT_W else empty,
268 fnlist,
269 interval,
270 )
271 if xl:
272 _check_fd_closed(fileno)
273 # Unlikely: the exception should have been raised above
274 raise e.OperationalError("connection socket closed")
275 ready = 0
276 if rl:
277 ready = READY_R
278 if wl:
279 ready |= READY_W
280
281 s = gen.send(ready)
282
283 except OSError as ex:
284 # This happens on macOS but not on Linux (the xl list is set)
285 raise e.OperationalError("connection socket closed") from ex

301def wait_epoll(gen: PQGen[RV], fileno: int, interval: float = 0.0) -> RV:
302 """
303 Wait for a generator using epoll where supported.
304
305 Parameters are like for `wait()`. If it is detected that the best selector
306 strategy is `epoll` then this function will be used instead of `wait`.
307
308 See also: https://linux.die.net/man/2/epoll_ctl
309
310 BUG: if the connection FD is closed, `epoll.poll()` hangs. Same for
311 EpollSelector. For this reason, wait_poll() is currently preferable.
312 To reproduce the bug:
313
314 export PSYCOPG_WAIT_FUNC=wait_epoll
315 pytest tests/test_concurrency.py::test_concurrent_close
316 """
317 if interval is None:
318 raise ValueError("indefinite wait not supported anymore")
319 try:
320 s = next(gen)
321
322 if interval < 0:
323 interval = 0.0
324
325 with select.epoll() as epoll:
326 evmask = _epoll_evmasks[s]
327 epoll.register(fileno, evmask)
328 while True:
329 if not (fileevs := epoll.poll(interval)):
330 _check_fd_closed(fileno)
331 gen.send(READY_NONE)
332 continue
333 ev = fileevs[0][1]
334 ready = 0
335 if ev & select.EPOLLIN:
336 ready = READY_R
337 if ev & select.EPOLLOUT:
338 ready |= READY_W
339 s = gen.send(ready)
340 evmask = _epoll_evmasks[s]
341 epoll.modify(fileno, evmask)
342
343 except StopIteration as ex:
344 rv: RV = ex.value
345 return rv

359def wait_poll(gen: PQGen[RV], fileno: int, interval: float = 0.0) -> RV:
360 """
361 Wait for a generator using poll where supported.
362
363 Parameters are like for `wait()`.
364 """
365 if interval is None:
366 raise ValueError("indefinite wait not supported anymore")
367 try:
368 s = next(gen)
369
370 if interval < 0:
371 interval = 0
372 else:
373 interval = int(interval * 1000.0)
374
375 poll = select.poll()
376 evmask = _poll_evmasks[s]
377 poll.register(fileno, evmask)
378 while True:
379 if not (fileevs := poll.poll(interval)):
380 gen.send(READY_NONE)
381 continue
382
383 ev = fileevs[0][1]
384
385 ready = 0
386 if ev & select.POLLIN:
387 ready = READY_R
388 if ev & select.POLLOUT:
389 ready |= READY_W
390
391 if not ready and ev & POLL_BAD:
392 _check_fd_closed(fileno)
393 # Unlikely: the exception should have been raised above
394 raise e.OperationalError("connection socket closed")
395
396 s = gen.send(ready)
397 evmask = _poll_evmasks[s]
398 poll.modify(fileno, evmask)
399
400 except StopIteration as ex:
401 rv: RV = ex.value
402 return rv

405def _is_select_patched() -> bool:
406 """
407 Detect if some greenlet library has patched the select library.
408
409 If this is the case, avoid to use the wait_c function as it doesn't behave
410 in a collaborative way.
411
412 Currently supported: gevent.
413 """
414 # If not imported, don't import it.
415 if m := sys.modules.get("gevent.monkey"):
416 try:
417 if m.is_module_patched("select"):
418 return True
419 except Exception as ex:
420 logger.warning("failed to detect gevent monkey-patching: %s", ex)
421
422 return False
423
424
425if _psycopg:
426 wait_c = _psycopg.wait_c
427
428
429# Choose the best wait strategy for the platform.
430#
431# the selectors objects have a generic interface but come with some overhead,
432# so we also offer more finely tuned implementations.
433
434wait: WaitFunc
435
436# Allow the user to choose a specific function for testing
437if "PSYCOPG_WAIT_FUNC" in os.environ:
438 fname = os.environ["PSYCOPG_WAIT_FUNC"]
439 if not fname.startswith("wait_") or fname not in globals():
440 raise ImportError(
441 "PSYCOPG_WAIT_FUNC should be the name of an available wait function;"
442 f" got {fname!r}"
443 )
444 wait = globals()[fname]
445
446# On Windows, for the moment, avoid using wait_c, because it was reported to
447# use excessive CPU (see #645).
448# TODO: investigate why.
449elif _psycopg and sys.platform != "win32" and not _is_select_patched():
450 wait = wait_c
451
452elif selectors.DefaultSelector is getattr(selectors, "SelectSelector", None):
453 # On Windows, SelectSelector should be the default.
454 wait = wait_select
455
456elif hasattr(selectors, "PollSelector"):
457 # On linux, EpollSelector is the default. However, it hangs if the fd is
458 # closed while polling.
459 wait = wait_poll
460
461else:
462 wait = wait_selector
