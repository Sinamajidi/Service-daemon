36# Size of data to accumulate before sending it down the network. We fill a
37# buffer this size field by field, and when it passes the threshold size
38# we ship it, so it may end up being bigger than this.
39BUFFER_SIZE = 32 * 1024
40
41# Maximum data size we want to queue to send to the libpq copy. Sending a
42# buffer too big to be handled can cause an infinite loop in the libpq
43# (#255) so we want to split it in more digestable chunks.
44MAX_BUFFER_SIZE = 4 * BUFFER_SIZE
45# Note: making this buffer too large, e.g.
46# MAX_BUFFER_SIZE = 1024 * 1024
47# makes operations *way* slower! Probably triggering some quadraticity
48# in the libpq memory management and data sending.
49
50# Max size of the write queue of buffers. More than that copy will block
51# Each buffer should be around BUFFER_SIZE size.
52QUEUE_SIZE = 1024
53
54# On certain systems, memmove seems particularly slow and flushing often is
55# more performing than accumulating a larger buffer. See #746 for details.
56PREFER_FLUSH = sys.platform == "darwin"

59class BaseCopy(Generic[ConnectionType]):
60 """
61 Base implementation for the copy user interface.
62
63 Two subclasses expose real methods with the sync/async differences.
64
65 The difference between the text and binary format is managed by two
66 different `Formatter` subclasses.
67
68 Writing (the I/O part) is implemented in the subclasses by a `Writer` or
69 `AsyncWriter` instance. Normally writing implies sending copy data to a
70 database, but a different writer might be chosen, e.g. to stream data into
71 a file for later use.
72 """
73
74 formatter: Formatter

76 def __init__(
77 self, cursor: BaseCursor[ConnectionType, Any], *, binary: bool | None = None
78 ):
79 self.cursor = cursor
80 self.connection = cursor.connection
81 self._pgconn = self.connection.pgconn
82
83 if result := cursor.pgresult:
84 self._direction = result.status
85 if self._direction != COPY_IN and self._direction != COPY_OUT:
86 raise e.ProgrammingError(
87 "the cursor should have performed a COPY operation;"
88 f" its status is {pq.ExecStatus(self._direction).name} instead"
89 )
90 else:
91 self._direction = COPY_IN
92
93 if binary is None:
94 binary = bool(result and result.binary_tuples)
95
96 tx: Transformer = getattr(cursor, "_tx", None) or adapt.Transformer(cursor)
97 if binary:
98 self.formatter = BinaryFormatter(tx)
99 else:
100 self.formatter = TextFormatter(tx, encoding=self._pgconn._encoding)
101
102 self._finished = False

113 def set_types(self, types: Sequence[int | str]) -> None:
114 """
115 Set the types expected in a COPY operation.
116
117 The types must be specified as a sequence of oid or PostgreSQL type
118 names (e.g. ``int4``, ``timestamptz[]``).
119
120 This operation overcomes the lack of metadata returned by PostgreSQL
121 when a COPY operation begins:
122
123 - On :sql:`COPY TO`, `!set_types()` allows to specify what types the
124 operation returns. If `!set_types()` is not used, the data will be
125 returned as unparsed strings or bytes instead of Python objects.
126
127 - On :sql:`COPY FROM`, `!set_types()` allows to choose what type the
128 database expects. This is especially useful in binary copy, because
129 PostgreSQL will apply no cast rule.
130
131 """
132 registry = self.cursor.adapters.types
133 oids = [t if isinstance(t, int) else registry.get_oid(t) for t in types]
134
135 if self._direction == COPY_IN:
136 self.formatter.transformer.set_dumper_types(oids, self.formatter.format)
137 else:
138 self.formatter.transformer.set_loader_types(oids, self.formatter.format)

140 # High level copy protocol generators (state change of the Copy object)
141
142 def _read_gen(self) -> PQGen[Buffer]:
143 if self._finished:
144 return memoryview(b"")
145
146 res = yield from copy_from(self._pgconn)
147 if isinstance(res, memoryview):
148 return res
149
150 # res is the final PGresult
151 self._finished = True
152
153 # This result is a COMMAND_OK which has info about the number of rows
154 # returned, but not about the columns, which is instead an information
155 # that was received on the COPY_OUT result at the beginning of COPY.
156 # So, don't replace the results in the cursor, just update the rowcount.
157 nrows = res.command_tuples
158 self.cursor._rowcount = nrows if nrows is not None else -1
159 return memoryview(b"")
160
161 def _read_row_gen(self) -> PQGen[tuple[Any, ...] | None]:
162 if not (data := (yield from self._read_gen())):
163 return None
164
165 if (row := self.formatter.parse_row(data)) is None:
166 # Get the final result to finish the copy operation
167 yield from self._read_gen()
168 self._finished = True
169 return None
170
171 return row
172
173 def _end_copy_out_gen(self) -> PQGen[None]:
174 try:
175 while (yield from self._read_gen()):
176 pass
177 except e.QueryCanceled:
178 pass

206class TextFormatter(Formatter):
207 format = TEXT
208
209 def __init__(self, transformer: Transformer, encoding: str = "utf-8"):
210 super().__init__(transformer)
211 self._encoding = encoding
212
213 def parse_row(self, data: Buffer) -> tuple[Any, ...] | None:
214 rv: tuple[Any, ...] | None = None
215 if data:
216 rv = parse_row_text(data, self.transformer)
217
218 return rv
219
220 def write(self, buffer: Buffer | str) -> Buffer:
221 data = self._ensure_bytes(buffer)
222 self._signature_sent = True
223 return data
224
225 def write_row(self, row: Sequence[Any]) -> Buffer:
226 # Note down that we are writing in row mode: it means we will have
227 # to take care of the end-of-copy marker too
228 self._row_mode = True
229
230 format_row_text(row, self.transformer, self._write_buffer)
231 if len(self._write_buffer) > BUFFER_SIZE:
232 buffer, self._write_buffer = self._write_buffer, bytearray()
233 return buffer
234 else:
235 return b""
236
237 def end(self) -> Buffer:
238 buffer, self._write_buffer = self._write_buffer, bytearray()
239 return buffer
240
241 def _ensure_bytes(self, data: Buffer | str) -> Buffer:
242 if isinstance(data, str):
243 return data.encode(self._encoding)
244 else:
245 # Assume, for simplicity, that the user is not passing stupid
246 # things to the write function. If that's the case, things
247 # will fail downstream.
248 return data

251class BinaryFormatter(Formatter):
252 format = BINARY
253
254 def __init__(self, transformer: Transformer):
255 super().__init__(transformer)
256 self._signature_sent = False
257
258 def parse_row(self, data: Buffer) -> tuple[Any, ...] | None:
259 rv: tuple[Any, ...] | None = None
260
261 if not self._signature_sent:
262 if data[: len(_binary_signature)] != _binary_signature:
263 raise e.DataError(
264 "binary copy doesn't start with the expected signature"
265 )
266 self._signature_sent = True
267 data = data[len(_binary_signature) :]
268
269 if data != _binary_trailer:
270 rv = parse_row_binary(data, self.transformer)
271
272 return rv
273
274 def write(self, buffer: Buffer | str) -> Buffer:
275 data = self._ensure_bytes(buffer)
276 self._signature_sent = True
277 return data
278
279 def write_row(self, row: Sequence[Any]) -> Buffer:
280 # Note down that we are writing in row mode: it means we will have
281 # to take care of the end-of-copy marker too
282 self._row_mode = True
283
284 if not self._signature_sent:
285 self._write_buffer += _binary_signature
286 self._signature_sent = True
287
288 format_row_binary(row, self.transformer, self._write_buffer)
289 if len(self._write_buffer) > BUFFER_SIZE:
290 buffer, self._write_buffer = self._write_buffer, bytearray()
291 return buffer
292 else:
293 return b""
294
295 def end(self) -> Buffer:
296 # If we have sent no data we need to send the signature
297 # and the trailer
298 if not self._signature_sent:
299 self._write_buffer += _binary_signature
300 self._write_buffer += _binary_trailer
301
302 elif self._row_mode:
303 # if we have sent data already, we have sent the signature
304 # too (either with the first row, or we assume that in
305 # block mode the signature is included).
306 # Write the trailer only if we are sending rows (with the
307 # assumption that who is copying binary data is sending the
308 # whole format).
309 self._write_buffer += _binary_trailer
310
311 buffer, self._write_buffer = self._write_buffer, bytearray()
312 return buffer
313
314 def _ensure_bytes(self, data: Buffer | str) -> Buffer:
315 if isinstance(data, str):
316 raise TypeError("cannot copy str data in binary mode: use bytes instead")
317 else:
318 # Assume, for simplicity, that the user is not passing stupid
319 # things to the write function. If that's the case, things
320 # will fail downstream.
321 return data

324def _format_row_text(row: Sequence[Any], tx: Transformer, out: bytearray) -> None:
325 """Convert a row of objects to the data to send for copy."""
326 if not row:
327 out += b"\n"
328 return
329
330 adapted = tx.dump_sequence(row, [PY_TEXT] * len(row))
331 for b in adapted:
332 out += _dump_re.sub(_dump_sub, b) if b is not None else rb"\N"
333 out += b"\t"
334
335 out[-1:] = b"\n"

338def _format_row_binary(row: Sequence[Any], tx: Transformer, out: bytearray) -> None:
339 """Convert a row of objects to the data to send for binary copy."""
340 adapted = tx.dump_sequence(row, [PY_BINARY] * len(row))
341 out += _pack_int2(len(row))
342 for b in adapted:
343 if b is not None:
344 out += _pack_int4(len(b))
345 out += b
346 else:
347 out += _binary_null

350def _parse_row_text(data: Buffer, tx: Transformer) -> tuple[Any, ...]:
351 if not isinstance(data, bytes):
352 data = bytes(data)
353 fields = data.split(b"\t")
354 fields[-1] = fields[-1][:-1] # drop \n
355 row = [None if f == b"\\N" else _load_re.sub(_load_sub, f) for f in fields]
356 return tx.load_sequence(row)
357
358
359def _parse_row_binary(data: Buffer, tx: Transformer) -> tuple[Any, ...]:
360 row: list[Buffer | None] = []
361 nfields = _unpack_int2(data, 0)[0]
362 pos = 2
363 for i in range(nfields):
364 length = _unpack_int4(data, pos)[0]
365 pos += 4
366 if length >= 0:
367 row.append(data[pos : pos + length])
368 pos += length
369 else:
370 row.append(None)
371
372 return tx.load_sequence(row)
