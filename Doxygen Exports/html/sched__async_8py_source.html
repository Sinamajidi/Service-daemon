1"""
2A minimal scheduler to schedule tasks to run in the future (async version).
3
4Inspired to the standard library `sched.scheduler`, but designed for
5multi-thread usage from the ground up, not as an afterthought. Tasks can be
6scheduled in front of the one currently running and `Scheduler.run()` can be
7left running without any tasks scheduled.
8
9Tasks are called "Task", not "Event", here, because we actually make use of
10`[threading/asyncio].Event` and the two would be confusing.
11"""

46 async def enter(self, delay: float, action: Callable[[], Any] | None) -> Task:
47 """Enter a new task in the queue delayed in the future.
48
49 Schedule a `!None` to stop the execution.
50 """
51 time = monotonic() + delay

54 async def enterabs(self, time: float, action: Callable[[], Any] | None) -> Task:
55 """Enter a new task in the queue at an absolute time.
56
57 Schedule a `!None` to stop the execution.
58 """
59 task = Task(time, action)
60 async with self._lock:
61 heappush(self._queue, task)
62 first = self._queue[0] is task
63
64 if first:
65 self._event.set()
66
67 return task

69 async def run(self) -> None:
70 """Execute the events scheduled."""
71 q = self._queue
72 while True:
73 async with self._lock:
74 now = monotonic()
75 if task := (q[0] if q else None):
76 if task.time <= now:
77 heappop(q)
78 else:
79 delay = task.time - now
80 task = None
81 else:
82 delay = self.EMPTY_QUEUE_TIMEOUT
83 self._event.clear()
84
85 if task:
86 if not task.action:
87 break
88 try:
89 await task.action()
90 except CLIENT_EXCEPTIONS as e:
91 logger.warning(
92 "scheduled task run %s failed: %s: %s",
93 task.action,
94 e.__class__.__name__,
95 e,
96 )
97 else:
98 # Block for the expected timeout or until a new task scheduled
99 await self._event.wait_timeout(delay)
