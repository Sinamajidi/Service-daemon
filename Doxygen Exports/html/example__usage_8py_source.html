203def example_complex_queries(created_ids):
204 """! @brief Demonstrate complex queries that leverage database processing."""
205 logger.info("="*60)
206 logger.info("STEP 5: COMPLEX QUERIES (Database-side processing)")
207 logger.info("="*60)
208
209 booking_access = get_booking_access()
210 tenant_access = get_tenant_access()
211 unit_access = get_unit_access()
212
213 # Get booking with all related details (JOIN query processed in database)
214 booking_details = booking_access.get_booking_with_details(created_ids['booking_id'])
215 logger.info(f"Booking with details: {booking_details}")
216
217 # Get tenant with user information (JOIN query)
218 tenant_info = tenant_access.get_tenant_with_user_info(created_ids['tenant_id'])
219 logger.info(f"Tenant with user info: {tenant_info}")
220
221 # Get vacant units (subquery processed in database)
222 vacant_ids = unit_access.get_vacant_unit_ids()
223 logger.info(f"Vacant units: {vacant_ids}")
224
225 # Custom aggregation query
226 dal = get_dal()
227 total_bookings = dal.execute_aggregation(
228 table='bookings',
229 agg_function='COUNT',
230 column='*',
231 where_clause="status != 'cancelled'",
232 params=()
233 )
234 logger.info(f"Total non-cancelled bookings: {total_bookings}")
235
236 # Average price of bookings
237 avg_price = dal.execute_aggregation(
238 table='bookings',
239 agg_function='AVG',
240 column='price_cents',
241 where_clause="price_cents IS NOT NULL",
242 params=()
243 )
244 logger.info(f"Average booking price: ${avg_price/100:.2f}" if avg_price else "No prices set")
245

290def example_lazy_instantiation():
291 """! @brief Demonstrate the lazy instantiation pattern.
292
293 Objects are only created when explicitly requested.
294 """
295 logger.info("="*60)
296 logger.info("STEP 7: LAZY INSTANTIATION PATTERN")
297 logger.info("="*60)
298
299 user_access = get_user_access()
300
301 # Scenario 1: GUI requests list of user IDs (no objects created)
302 logger.info("Scenario 1: Get IDs only (no objects)")
303 user_ids = user_access.get_all_user_ids()
304 logger.info(f"Retrieved {len(user_ids)} user IDs - NO objects instantiated")
305
306 # Scenario 2: GUI requests complete data for one user (still no object)

310 logger.info(f"Retrieved user data - returned as DICTIONARY, not object")

318 logger.info(f"Object: {user_obj}")
