138class InetBinaryLoader(_LazyIpaddressLoader):
139 format = Format.BINARY
140
141 def load(self, data: Buffer) -> Address | Interface:
142 if isinstance(data, memoryview):
143 data = bytes(data)
144
145 prefix = data[1]
146 packed = data[4:]
147 if data[0] == PGSQL_AF_INET:
148 if prefix == IPV4_PREFIXLEN:
149 return IPv4Address(packed)
