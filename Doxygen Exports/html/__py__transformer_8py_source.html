171 def dump_sequence(
172 self, params: Sequence[Any], formats: Sequence[PyFormat]
173 ) -> Sequence[Buffer | None]:
174 nparams = len(params)
175 out: list[Buffer | None] = [None] * nparams
176
177 # If we have dumpers, it means set_dumper_types had been called, in
178 # which case self.types and self.formats are set to sequences of the
179 # right size.
180 if self._row_dumpers:
181 if len(self._row_dumpers) != nparams:
182 raise e.DataError(
183 f"expected {len(self._row_dumpers)} values in row, got {nparams}"
184 )
185 for i in range(nparams):
186 if (param := params[i]) is not None:
187 out[i] = self._row_dumpers[i].dump(param)
188 return out
189
190 types = [self._get_none_oid()] * nparams
191 pqformats = [TEXT] * nparams
192
193 for i in range(nparams):
194 if (param := params[i]) is None:
195 continue
196 dumper = self.get_dumper(param, formats[i])
197 out[i] = dumper.dump(param)
198 types[i] = dumper.oid
199 pqformats[i] = dumper.format
200
201 self.types = tuple(types)
202 self.formats = pqformats
203
204 return out

206 def as_literal(self, obj: Any) -> bytes:
207 dumper = self.get_dumper(obj, PY_TEXT)
208 rv = dumper.quote(obj)
209 # If the result is quoted, and the oid not unknown or text,
210 # add an explicit type cast.
211 # Check the last char because the first one might be 'E'.
212 oid = dumper.oid
213 if oid and rv and rv[-1] == b"'"[0] and oid != TEXT_OID:
214 try:
215 type_sql = self._oid_types[oid]
216 except KeyError:
217 if ti := self.adapters.types.get(oid):
218 if oid < 8192:
219 # builtin: prefer "timestamptz" to "timestamp with time zone"
220 type_sql = ti.name.encode(self.encoding)
221 else:
222 type_sql = ti.regtype.encode(self.encoding)
223 if oid == ti.array_oid:
224 type_sql += b"[]"
225 else:
226 type_sql = b""
227 self._oid_types[oid] = type_sql
228
229 if type_sql:
230 rv = b"%s::%s" % (rv, type_sql)
231
232 if not isinstance(rv, bytes):
233 rv = bytes(rv)
234 return rv

236 def get_dumper(self, obj: Any, format: PyFormat) -> abc.Dumper:
237 """
238 Return a Dumper instance to dump `!obj`.
239 """
240 # Normally, the type of the object dictates how to dump it
241 key = type(obj)
242
243 # Reuse an existing Dumper class for objects of the same type
244 cache = self._dumpers[format]
245 try:
246 dumper = cache[key]
247 except KeyError:
248 # If it's the first time we see this type, look for a dumper
249 # configured for it.
250 try:
251 dcls = self.adapters.get_dumper(key, format)
252 except e.ProgrammingError as ex:
253 raise ex from None
254 else:
255 cache[key] = dumper = dcls(key, self)
256
257 # Check if the dumper requires an upgrade to handle this specific value
258 if (key1 := dumper.get_key(obj, format)) is key:
259 return dumper
260
261 # If it does, ask the dumper to create its own upgraded version
262 try:
263 return cache[key1]
264 except KeyError:
265 dumper = cache[key1] = dumper.upgrade(obj, format)
266 return dumper

300 def load_rows(self, row0: int, row1: int, make_row: RowMaker[Row]) -> list[Row]:
301 if not (res := self._pgresult):
302 raise e.InterfaceError("result not set")
303
304 if not (0 <= row0 <= self._ntuples and 0 <= row1 <= self._ntuples):
305 raise e.InterfaceError(
306 f"rows must be included between 0 and {self._ntuples}"
307 )
308
309 records = []
310 for row in range(row0, row1):

312 for col in range(self._nfields):
313 if (val := res.get_value(row, col)) is not None:
314 record[col] = self._row_loaders[col](val)
315 records.append(make_row(record))
