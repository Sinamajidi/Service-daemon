30 def _hash_password(self, password: str) -> str:
31 """! @brief Hash a plaintext password for storage."""
32 salt = os.urandom(16)
33 iterations = 120_000
34 digest = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, iterations)
35 return "pbkdf2_sha256${}${}${}".format(
36 iterations,
37 base64.b64encode(salt).decode("utf-8"),
38 base64.b64encode(digest).decode("utf-8"),
39 )

41 def _maybe_hash_password(self, value: Optional[str]) -> Optional[str]:
42 """! @brief Hash plaintext passwords while leaving existing hashes intact."""
43 if not value:
44 return None
45 if value.startswith("pbkdf2_sha256$"):
46 return value
47 return self._hash_password(value)

195 def insert_record(self, table: str, data: Dict[str, Any]) -> str:
196 """
197 Insert a new record.
198
199 Args:
200 table: Table name
201 data: Dictionary of column: value pairs
202
203 Returns:
204 UUID of inserted record as string
205 """
206 if table == "users" and "password_hash" in data:
207 data["password_hash"] = self._maybe_hash_password(data.get("password_hash"))
208
209 columns = ', '.join(data.keys())
210 placeholders = ', '.join(['%s'] * len(data))
211 values = tuple(data.values())
212
213 query = f"""
214 INSERT INTO {table} ({columns})
215 VALUES ({placeholders})
216 RETURNING id
217 """
218
219 result = self.db.execute_query(query, values, fetch='one', dict_cursor=True)
220 return str(result['id']) if result else None
