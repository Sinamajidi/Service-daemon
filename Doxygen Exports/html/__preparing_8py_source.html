57 def get(
58 self, query: PostgresQuery, prepare: bool | None = None
59 ) -> tuple[Prepare, bytes]:
60 """
61 Check if a query is prepared, tell back whether to prepare it.
62 """
63 if prepare is False or self.prepare_threshold is None:
64 # The user doesn't want this query to be prepared
65 return Prepare.NO, b""
66
67 if name := self._names.get(key := self.key(query)):
68 # The query was already prepared in this session
69 return Prepare.YES, name
70
71 count = self._counts.get(key, 0)
72 if count >= self.prepare_threshold or prepare:
73 # The query has been executed enough times and needs to be prepared
74 name = f"_pg3_{self._prepared_idx}".encode()
75 self._prepared_idx += 1
76 return Prepare.SHOULD, name
77 else:
78 # The query is not to be prepared yet
79 return Prepare.NO, b""

81 def _should_discard(self, prep: Prepare, results: Sequence[PGresult]) -> bool:
82 """Check if we need to discard our entire state: it should happen on
83 rollback or on dropping objects, because the same object may get
84 recreated and postgres would fail internal lookups.
85 """
86 if self._names or prep == Prepare.SHOULD:
87 for result in results:
88 if result.status != COMMAND_OK:
89 continue
90 cmdstat = result.command_status
91 if cmdstat and (cmdstat.startswith(b"DROP ") or cmdstat == b"ROLLBACK"):
92 return self.clear()
93 return False

95 @staticmethod
96 def _check_results(results: Sequence[PGresult]) -> bool:
97 """Return False if 'results' are invalid for prepared statement cache."""
98 if len(results) != 1:
99 # We cannot prepare a multiple statement
100 return False
101
102 if COMMAND_OK != results[0].status != TUPLES_OK:
103 # We don't prepare failed queries or other weird results
104 return False
105
106 return True

108 def _rotate(self) -> None:
109 """Evict an old value from the cache.
110
111 If it was prepared, deallocate it. Do it only once: if the cache was
112 resized, deallocate gradually.
113 """
114 if len(self._counts) > self.prepared_max:
115 self._counts.popitem(last=False)
116
117 if len(self._names) > self.prepared_max:
118 name = self._names.popitem(last=False)[1]
119 self._to_flush.append(name)

121 def maybe_add_to_cache(
122 self, query: PostgresQuery, prep: Prepare, name: bytes
123 ) -> Key | None:
124 """Handle 'query' for possible addition to the cache.
125
126 If a new entry has been added, return its key. Return None otherwise
127 (meaning the query is already in cache or cache is not enabled).
128 """
129 # don't do anything if prepared statements are disabled
130 if self.prepare_threshold is None:
131 return None
132
133 if (key := self.key(query)) in self._counts:
134 if prep is Prepare.SHOULD:
135 del self._counts[key]
136 self._names[key] = name
137 else:
138 self._counts[key] += 1
139 self._counts.move_to_end(key)
140 return None
141
142 elif key in self._names:
143 self._names.move_to_end(key)
144 return None
145
146 else:
147 if prep is Prepare.SHOULD:
148 self._names[key] = name
149 else:
150 self._counts[key] = 1
151 return key

153 def validate(
154 self, key: Key, prep: Prepare, name: bytes, results: Sequence[PGresult]
155 ) -> None:
156 """Validate cached entry with 'key' by checking query 'results'.
157
158 Possibly record a command to perform maintenance on database side.
159 """
160 if self._should_discard(prep, results):
161 return
162
163 if not self._check_results(results):
164 self._names.pop(key, None)
165 self._counts.pop(key, None)
166 else:
167 self._rotate()

169 def clear(self) -> bool:
170 """Clear the cache of the maintenance commands.
171
172 Clear the internal state and prepare a command to clear the state of
173 the server.
174 """
175 self._counts.clear()
176 if self._names:
177 self._names.clear()
178 self._to_flush.clear()
179 self._to_flush.append(None)
180 return True
181 else:
182 return False

184 def maintain_gen(self, conn: BaseConnection[Any]) -> PQGen[None]:
185 """
186 Generator to send the commands to perform periodic maintenance
