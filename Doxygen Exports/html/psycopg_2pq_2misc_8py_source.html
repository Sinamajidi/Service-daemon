53def find_libpq_full_path() -> str | None:
54 if sys.platform == "win32":
55 if (libname := ctypes.util.find_library("libpq.dll")) is None:
56 return None
57 libname = str(Path(libname).resolve())
58 elif sys.platform == "darwin":
59 libname = ctypes.util.find_library("libpq.dylib")
60 # (hopefully) temporary hack: libpq not in a standard place
61 # https://github.com/orgs/Homebrew/discussions/3595
62 # If pg_config is available and agrees, let's use its indications.
63 if not libname:
64 try:
65 import subprocess as sp
66
67 libdir = sp.check_output(["pg_config", "--libdir"]).strip().decode()
68 if not os.path.exists(libname := os.path.join(libdir, "libpq.dylib")):
69 libname = None
70 except Exception as ex:
71 logger.debug("couldn't use pg_config to find libpq: %s", ex)
72
73 else:
74 libname = ctypes.util.find_library("pq")
75
76 return libname

142def connection_summary(pgconn: abc.PGconn) -> str:
143 """
144 Return summary information on a connection.
145
146 Useful for __repr__
147 """
148 parts = []
149 if pgconn.status == OK:
150 # Put together the [STATUS]
151 status: str = TransactionStatus(pgconn.transaction_status).name
152 if pgconn.pipeline_status:
153 status += f", pipeline={PipelineStatus(pgconn.pipeline_status).name}"
154
155 # Put together the (CONNECTION)
156 if not pgconn.host.startswith(b"/"):
157 parts.append(("host", pgconn.host.decode()))
158 if (port := pgconn.port.decode() or get_compiled_port()) != "5432":
159 parts.append(("port", port))
160 if pgconn.user != pgconn.db:
161 parts.append(("user", pgconn.user.decode()))
162 parts.append(("database", pgconn.db.decode()))
163
164 else:
165 try:
166 status = ConnStatus(pgconn.status).name
167 except ValueError:
168 # It might happen if a new status on connection appears
169 # before upgrading the ConnStatus enum.
170 status = f"status={pgconn.status} (unkndown)"
171
172 if sparts := " ".join("%s=%s" % part for part in parts):
173 sparts = f" ({sparts})"
174 return f"[{status}]{sparts}"

177def version_pretty(version: int) -> str:
178 """
179 Return a pretty representation of a PostgreSQL version
180
181 For instance: 140002 -> 14.2, 90610 -> 9.6.10
182 """
183 version, patch = divmod(version, 100)
184 major, minor = divmod(version, 100)
185 if major >= 10 and minor == 0:
186 return f"{major}.{patch}"
187 else:
188 return f"{major}.{minor}.{patch}"
