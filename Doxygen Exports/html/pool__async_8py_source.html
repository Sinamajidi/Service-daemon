171 async def wait(self, timeout: float = 30.0) -> None:
172 """
173 Wait for the pool to be full (with `min_size` connections) after creation.
174
175 Close the pool, and raise `PoolTimeout`, if not ready within *timeout*
176 sec.
177
178 Calling this method is not mandatory: you can try and use the pool
179 immediately after its creation. The first client will be served as soon
180 as a connection is ready. You can use this method if you prefer your
181 program to terminate in case the environment is not configured
182 properly, rather than trying to stay up the hardest it can.
183 """
184 self._check_open_getconn()
185
186 async with self._lock:
187 assert not self._pool_full_event
188 if len(self._pool) >= self._min_size:
189 return
190 self._pool_full_event = AEvent()
191
192 logger.info("waiting for pool %r initialization", self.name)
193 if not await self._pool_full_event.wait_timeout(timeout):
194 await self.close() # stop all the tasks
195 raise PoolTimeout(f"pool initialization incomplete after {timeout} sec")
196
197 async with self._lock:
198 assert self._pool_full_event
199 self._pool_full_event = None
200
201 logger.info("pool %r is ready to use", self.name)

204 async def connection(self, timeout: float | None = None) -> AsyncIterator[ACT]:
205 """Context manager to obtain a connection from the pool.
206
207 Return the connection immediately if available, otherwise wait up to
208 *timeout* or `self.timeout` seconds and throw `PoolTimeout` if a
209 connection is not available in time.
210
211 Upon context exit, return the connection to the pool. Apply the normal
212 :ref:`connection context behaviour <with-connection>` (commit/rollback
213 the transaction in case of success/error). If the connection is no more
214 in working state, replace it with a new one.
215 """
216 conn = await self.getconn(timeout=timeout)
217 try:
218 t0 = monotonic()
219 async with conn:
220 yield conn
221 finally:
222 await self.putconn(conn)
223 t1 = monotonic()
224 self._stats[self._USAGE_MS] += int(1000.0 * (t1 - t0))

226 async def getconn(self, timeout: float | None = None) -> ACT:
227 """Obtain a connection from the pool.
228
229 You should preferably use `connection()`. Use this function only if
230 it is not possible to use the connection as context manager.
231
232 After using this function you *must* call a corresponding `putconn()`:
233 failing to do so will deplete the pool. A depleted pool is a sad pool:
234 you don't want a depleted pool.
235 """
236 if timeout is None:
237 timeout = self.timeout
238 deadline = monotonic() + timeout
239
240 logger.info("connection requested from %r", self.name)
241 self._stats[self._REQUESTS_NUM] += 1
242
243 self._check_open_getconn()
244
245 try:
246 return await self._getconn_with_check_loop(deadline)
247
248 # Re-raise the timeout exception presenting the user the global
249 # timeout, not the per-attempt one.
250 except PoolTimeout:
251 raise PoolTimeout(

255 async def _getconn_with_check_loop(self, deadline: float) -> ACT:
256 attempt: AttemptWithBackoff | None = None
257
258 while True:
259 conn = await self._getconn_unchecked(deadline - monotonic())
260 try:
261 await self._check_connection(conn)
262 except CLIENT_EXCEPTIONS:
263 await self._putconn(conn, from_getconn=True)
264 else:
265 logger.info("connection given by %r", self.name)
266 return conn
267
268 # Delay further checks to avoid a busy loop, using the same
269 # backoff policy used in reconnection attempts.
270 now = monotonic()
271 if not attempt:
272 attempt = AttemptWithBackoff(timeout=deadline - now)
273 else:
274 attempt.update_delay(now)
275
276 if attempt.time_to_give_up(now):
277 raise PoolTimeout()
278 else:
279 await asleep(attempt.delay)

281 async def _getconn_unchecked(self, timeout: float) -> ACT:
282 # Critical section: decide here if there's a connection ready
283 # or if the client needs to wait.
284 async with self._lock:
285 if not (conn := (await self._get_ready_connection(timeout))):
286 # No connection available: put the client in the waiting queue
287 t0 = monotonic()
288 pos: WaitingClient[ACT] = WaitingClient()
289 self._waiting.append(pos)
290 self._stats[self._REQUESTS_QUEUED] += 1
291
292 # If there is space for the pool to grow, let's do it
293 self._maybe_grow_pool()
294
295 # If we are in the waiting queue, wait to be assigned a connection
296 # (outside the critical section, so only the waiting client is locked)
297 if not conn:
298 try:
299 conn = await pos.wait(timeout=timeout)
300 except CLIENT_EXCEPTIONS:
301 self._stats[self._REQUESTS_ERRORS] += 1
302 raise
303 finally:
304 t1 = monotonic()
305 self._stats[self._REQUESTS_WAIT_MS] += int(1000.0 * (t1 - t0))
306
307 # Tell the connection it belongs to a pool to avoid closing on __exit__
308 # Note that this property shouldn't be set while the connection is in
309 # the pool, to avoid to create a reference loop.
310 conn._pool = self
311 return conn

406 async def open(self, wait: bool = False, timeout: float = 30.0) -> None:
407 """Open the pool by starting connecting and and accepting clients.
408
409 If *wait* is `!False`, return immediately and let the background worker
410 fill the pool if `min_size` > 0. Otherwise wait up to *timeout* seconds
411 for the requested number of connections to be ready (see `wait()` for
412 details).
413
414 It is safe to call `!open()` again on a pool already open (because the
415 method was already called, or because the pool context was entered, or
416 because the pool was initialized with *open* = `!True`) but you cannot
417 currently re-open a closed pool.
418 """
419 # Make sure the lock is created after there is an event loop
420 self._ensure_lock()
421
422 async with self._lock:
423 self._open()
424
425 if wait:
426 await self.wait(timeout=timeout)

482 async def close(self, timeout: float = 5.0) -> None:
483 """Close the pool and make it unavailable to new clients.
484
485 All the waiting and future clients will fail to acquire a connection
486 with a `PoolClosed` exception. Currently used connections will not be
487 closed until returned to the pool.
488
489 Wait *timeout* seconds for threads to terminate their job, if positive.
490 If the timeout expires the pool is closed anyway, although it may raise
491 some warnings on exit.
492 """
493 if self._closed:
494 return
495
496 async with self._lock:
497 self._closed = True
498 logger.debug("pool %r closed", self.name)
499
500 # Take waiting client and pool connections out of the state
501 waiting = list(self._waiting)
502 self._waiting.clear()
503 connections = list(self._pool)
504 self._pool.clear()
505
506 # Take the workers out of the pool. Will stop them outside the lock
507 workers = await self._signal_stop_worker()
508
509 # Now that the flag _closed is set, getconn will fail immediately,
510 # putconn will just close the returned connection.
511
512 # Wait for the worker tasks to terminate
513 await agather(*workers, timeout=timeout)
514
515 # Close the connections that were still in the pool
516 for conn in connections:
517 await self._close_connection(conn)
518
519 # Signal to eventual clients in the queue that business is closed.
520 for pos in waiting:
521 await pos.fail(PoolClosed(f"the pool {self.name!r} is closed"))

523 async def _signal_stop_worker(self) -> list[AWorker]:
524 # Stop the scheduler
525 await self._sched.enter(0, None)
526
527 # Stop the worker tasks
528 workers, self._workers = self._workers[:], []
529 for _ in workers:
530 self.run_task(StopWorker(self))
531
532 if self._sched_runner: # likely
533 workers.append(self._sched_runner)
534 self._sched_runner = None
535
536 return workers

568 async def check(self) -> None:
569 """Verify the state of the connections currently in the pool.
570
571 Test each connection: if it works return it to the pool, otherwise
572 dispose of it and create a new one.
573 """
574 async with self._lock:
575 conns = list(self._pool)
576 self._pool.clear()
577
578 # Give a chance to the pool to grow if it has no connection.
579 # In case there are enough connection, or the pool is already
580 # growing, this is a no-op.
581 self._maybe_grow_pool()
582
583 while conns:
584 conn = conns.pop()
585
586 # Check for expired connections
587 if conn._expire_at <= monotonic():
588 logger.info("discarding expired connection %s", conn)
589 await self._close_connection(conn)
590 self.run_task(AddConnection(self))
591 continue
592
593 # Check for broken connections
594 try:
595 await self.check_connection(conn)
596 except CLIENT_EXCEPTIONS:
597 self._stats[self._CONNECTIONS_LOST] += 1
598 logger.warning("discarding broken connection: %s", conn)
599 self.run_task(AddConnection(self))
600 else:
601 await self._add_to_pool(conn)

672 async def _connect(self, timeout: float | None = None) -> ACT:
673 """Return a new connection configured for the pool."""
674 self._stats[self._CONNECTIONS_NUM] += 1
675 conninfo = await self._resolve_conninfo()
676 kwargs = await self._resolve_kwargs()
677 if timeout:
678 kwargs = kwargs.copy()
679 kwargs["connect_timeout"] = max(round(timeout), 1)
680 t0 = monotonic()
681 try:
682 conn = await self.connection_class.connect(conninfo, **kwargs)
683 except CLIENT_EXCEPTIONS:
684 self._stats[self._CONNECTIONS_ERRORS] += 1
685 raise
686 else:
687 t1 = monotonic()
688 self._stats[self._CONNECTIONS_MS] += int(1000.0 * (t1 - t0))
689
690 conn._pool = self
691
692 if self._configure:
693 await self._configure(conn)
694 if (status := conn.pgconn.transaction_status) != TransactionStatus.IDLE:
695 sname = TransactionStatus(status).name
696 raise e.ProgrammingError(
697 f"connection left in status {sname} by configure function"
698 f" {self._configure}: discarded"
699 )
700
701 # Set an expiry date, with some randomness to avoid mass reconnection
702 self._set_connection_expiry_date(conn)
703 return conn

722 async def _add_connection(
723 self, attempt: AttemptWithBackoff | None, growing: bool = False
724 ) -> None:
725 """Try to connect and add the connection to the pool.
726
727 If failed, reschedule a new attempt in the future for a few times, then
728 give up, decrease the pool connections number and call
729 `self.reconnect_failed()`.
730
731 """
732 now = monotonic()
733 if not attempt:
734 attempt = AttemptWithBackoff(timeout=self.reconnect_timeout)
735
736 try:
737 conn = await self._connect()
738 except CLIENT_EXCEPTIONS as ex:
739 logger.warning("error connecting in %r: %s", self.name, ex)
740 if attempt.time_to_give_up(now):
741 logger.warning(
742 "reconnection attempt in pool %r failed after %s sec",
743 self.name,
744 self.reconnect_timeout,
745 )
746 async with self._lock:
747 self._nconns -= 1
748 # If we have given up with a growing attempt, allow a new one.
749 if growing and self._growing:
750 self._growing = False
751 await self.reconnect_failed()
752 else:
753 attempt.update_delay(now)
754 await self.schedule_task(
755 AddConnection(self, attempt, growing=growing), attempt.delay
756 )
757 return

759 logger.info("adding new connection to the pool")
760 await self._add_to_pool(conn)
761 if growing:
762 async with self._lock:
763 # Keep on growing if the pool is not full yet, or if there are
764 # clients waiting and the pool can extend.
765 if self._nconns < self._min_size or (
766 self._nconns < self._max_size and self._waiting
767 ):
768 self._nconns += 1
769 logger.info("growing pool %r to %s", self.name, self._nconns)
770 self.run_task(AddConnection(self, growing=True))
771 else:
772 self._growing = False
773
774 async def _return_connection(self, conn: ACT, from_getconn: bool) -> None:
775 """
776 Return a connection to the pool after usage.
777 """
778 await self._reset_connection(conn)
779 if from_getconn:
780 if conn.pgconn.transaction_status == TransactionStatus.UNKNOWN:
781 self._stats[self._CONNECTIONS_LOST] += 1
782 # Connection no more in working state: create a new one.
783 logger.info("not serving connection found broken")
784 self.run_task(AddConnection(self))
785 return
786
787 else:
788 if conn.pgconn.transaction_status == TransactionStatus.UNKNOWN:
789 self._stats[self._RETURNS_BAD] += 1
790 # Connection no more in working state: create a new one.
791 logger.warning("discarding closed connection: %s", conn)
792 self.run_task(AddConnection(self))
793 return
794
795 # Check if the connection is past its best before date
796 if conn._created_at <= self._drained_at or conn._expire_at <= monotonic():
797 logger.info("discarding expired connection")
798 await self._close_connection(conn)
799 self.run_task(AddConnection(self))
800 return
801
802 await self._add_to_pool(conn)

804 async def _add_to_pool(self, conn: ACT) -> None:
805 """
806 Add a connection to the pool.
807
808 The connection can be a fresh one or one already used in the pool.
809
810 If a client is already waiting for a connection pass it on, otherwise
811 put it back into the pool
812 """
813 # Remove the pool reference from the connection before returning it
814 # to the state, to avoid to create a reference loop.
815 # Also disable the warning for open connection in conn.__del__
816 conn._pool = None
817
818 # Early bailout in case the pool is closed. Don't add anything to the
819 # state. There is still a remote chance that the pool will be closed
820 # between here and entering the lock. Therefore we will make another
821 # check later.
822 if self._closed:
823 await self._close_connection(conn)
824 return
825
826 # Critical section: if there is a client waiting give it the connection
827 # otherwise put it back into the pool.
828 async with self._lock:
829
830 # Check if the pool was closed by the time we arrived here. It is
831 # unlikely but it doesn't seem impossible, if the worker was adding
832 # this connection while the main process is closing the pool.
833 # Now that we are in the critical section we know for real.
834 if self._closed:
835 await self._close_connection(conn)
836 return
837
838 while self._waiting:
839 # If there is a client waiting (which is still waiting and
840 # hasn't timed out), give it the connection and notify it.
841
842 if await self._waiting.popleft().set(conn):
843 break
844 else:
845 # No client waiting for a connection: put it back into the pool
846 self._pool.append(conn)
847 # If we have been asked to wait for pool init, notify the
848 # waiter if the pool is full.
849 if self._pool_full_event and len(self._pool) >= self._min_size:
850 self._pool_full_event.set()
851
852 async def _reset_connection(self, conn: ACT) -> None:
853 """
854 Bring a connection to IDLE state or close it.
855 """
856 if (status := conn.pgconn.transaction_status) == TransactionStatus.IDLE:
857 pass
858 elif status == TransactionStatus.UNKNOWN:
859 # Connection closed
860 return
861
862 elif status == TransactionStatus.INTRANS or status == TransactionStatus.INERROR:
863 # Connection returned with an active transaction
864 logger.warning("rolling back returned connection: %s", conn)
865 try:
866 await conn.rollback()
867 except CLIENT_EXCEPTIONS as ex:
868 logger.warning(
869 "rollback failed: %s: %s. Discarding connection %s",
870 ex.__class__.__name__,
871 ex,
872 conn,
873 )
874 await self._close_connection(conn)
875
876 elif status == TransactionStatus.ACTIVE:
877 # Connection returned during an operation. Bad... just close it.
878 logger.warning("closing returned connection: %s", conn)
879 await self._close_connection(conn)
880
881 if self._reset:
882 try:
883 await self._reset(conn)
884 if (status := conn.pgconn.transaction_status) != TransactionStatus.IDLE:
885 sname = TransactionStatus(status).name
886 raise e.ProgrammingError(
887 f"connection left in status {sname} by reset function"
888 f" {self._reset}: discarded"
889 )
890 except CLIENT_EXCEPTIONS as ex:
891 logger.warning("error resetting connection: %s", ex)
892 await self._close_connection(conn)
893
894 async def _close_connection(self, conn: ACT) -> None:
895 conn._pool = None
896 await conn.close()
897
898 async def _shrink_pool(self) -> None:
899 to_close: ACT | None = None
900
901 async with self._lock:
902 # Reset the min number of connections used
903 nconns_min = self._nconns_min
904 self._nconns_min = len(self._pool)
905
906 # If the pool can shrink and connections were unused, drop one
907 if self._nconns > self._min_size and nconns_min > 0 and self._pool:
908 to_close = self._pool.popleft()
909 self._nconns -= 1
910 self._nconns_min -= 1
911
912 if to_close:
913 logger.info(
914 "shrinking pool %r to %s because %s unused connections"
915 " in the last %s sec",
916 self.name,
917 self._nconns,
918 nconns_min,
919 self.max_idle,
920 )
921 await self._close_connection(to_close)

923 def _get_measures(self) -> dict[str, int]:
924 rv = super()._get_measures()
925 rv[self._REQUESTS_WAITING] = len(self._waiting)
926 return rv

929class WaitingClient(Generic[ACT]):
930 """A position in a queue for a client waiting for a connection."""
931
932 __slots__ = ("conn", "error", "_cond")
933
934 def __init__(self) -> None:
935 self.conn: ACT | None = None
936 self.error: BaseException | None = None
937
938 # The WaitingClient behaves in a way similar to an Event, but we need
939 # to notify reliably the flagger that the waiter has "accepted" the
940 # message and it hasn't timed out yet, otherwise the pool may give a
941 # connection to a client that has already timed out getconn(), which
942 # will be lost.
943 self._cond = ACondition()
944
945 async def wait(self, timeout: float) -> ACT:
946 """Wait for a connection to be set and return it.
947
948 Raise an exception if the wait times out or if fail() is called.
949 """
950 async with self._cond:
951 if not (self.conn or self.error):
952 try:
953 if not await self._cond.wait_timeout(timeout):
954 self.error = PoolTimeout(
955 f"couldn't get a connection after {timeout:.2f} sec"
956 )
957 except CLIENT_EXCEPTIONS as ex:
958 self.error = ex
959
960 if self.conn:
961 return self.conn
962 else:
963 assert self.error
964 raise self.error
965
966 async def set(self, conn: ACT) -> bool:
967 """Signal the client waiting that a connection is ready.
968
969 Return True if the client has "accepted" the connection, False
970 otherwise (typically because wait() has timed out).
971 """
972 async with self._cond:
973 if self.conn or self.error:
974 return False
975
976 self.conn = conn
977 self._cond.notify_all()
978 return True
979
980 async def fail(self, error: Exception) -> bool:
981 """Signal the client that, alas, they won't have a connection today.
982
983 Return True if the client has "accepted" the error, False otherwise
984 (typically because wait() has timed out).
985 """
986 async with self._cond:
987 if self.conn or self.error:
988 return False
989
990 self.error = error
991 self._cond.notify_all()
992 return True
