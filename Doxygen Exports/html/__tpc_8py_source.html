53 @classmethod
54 def _parse_string(cls, s: str) -> Xid:
55 if not (m := _re_xid.match(s)):
56 raise ValueError("bad Xid format")
57
58 format_id = int(m.group(1))
59 gtrid = b64decode(m.group(2)).decode()
60 bqual = b64decode(m.group(3)).decode()
61 return cls.from_parts(format_id, gtrid, bqual)

63 @classmethod
64 def from_parts(cls, format_id: int | None, gtrid: str, bqual: str | None) -> Xid:
65 if format_id is not None:
66 if bqual is None:
67 raise TypeError("if format_id is specified, bqual must be too")
68 if not 0 <= format_id < 0x80000000:
69 raise ValueError("format_id must be a non-negative 32-bit integer")
70 if len(bqual) > 64:

80 def _as_tid(self) -> str:
81 """
82 Return the PostgreSQL transaction_id for this XA xid.
83
84 PostgreSQL wants just a string, while the DBAPI supports the XA
85 standard and thus a triple. We use the same conversion algorithm
86 implemented by JDBC in order to allow some form of interoperation.
87
88 see also: the pgjdbc implementation
89 http://cvs.pgfoundry.org/cgi-bin/cvsweb.cgi/jdbc/pgjdbc/org/
90 postgresql/xa/RecoveredXid.java?rev=1.2
91 """
92 if self.format_id is None or self.bqual is None:
93 # Unparsed xid: return the gtrid.
94 return self.gtrid
95
96 # XA xid: mash together the components.
97 egtrid = b64encode(self.gtrid.encode()).decode()
98 ebqual = b64encode(self.bqual.encode()).decode()
99
100 return f"{self.format_id}_{egtrid}_{ebqual}"
