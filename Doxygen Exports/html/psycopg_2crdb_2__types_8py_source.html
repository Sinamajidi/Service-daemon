54def _register_postgres_adapters(context: AdaptContext) -> None:
55 # Same adapters used by PostgreSQL, or a good starting point for customization
56
57 from ..types import array, bool, composite, datetime, numeric, numpy, string, uuid
58
59 array.register_default_adapters(context)

61 datetime.register_default_adapters(context)
62 string.register_default_adapters(context)
63 uuid.register_default_adapters(context)
64
65 # Both numpy Decimal and uint64 dumpers use the numeric oid, but the former
66 # covers the entire numeric domain, whereas the latter only deals with
67 # integers. For this reason, if we specify dumpers by oid, we want to make
68 # sure to get the Decimal dumper. We enforce that by registering the
69 # numeric dumpers last.
70 numpy.register_default_adapters(context)
71 bool.register_default_adapters(context)
