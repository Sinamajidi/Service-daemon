437 def close(self, timeout: float = 5.0) -> None:
438 """Close the pool and make it unavailable to new clients.
439
440 All the waiting and future clients will fail to acquire a connection
441 with a `PoolClosed` exception. Currently used connections will not be
442 closed until returned to the pool.
443
444 Wait *timeout* seconds for threads to terminate their job, if positive.
445 If the timeout expires the pool is closed anyway, although it may raise
446 some warnings on exit.
447 """
448 if self._closed:
449 return
450
451 with self._lock:
452 self._closed = True
453 logger.debug("pool %r closed", self.name)
454
455 # Take waiting client and pool connections out of the state
456 waiting = list(self._waiting)
457 self._waiting.clear()
458 connections = list(self._pool)
459 self._pool.clear()
460
461 # Take the workers out of the pool. Will stop them outside the lock
462 workers = self._signal_stop_worker()

464 # Now that the flag _closed is set, getconn will fail immediately,
465 # putconn will just close the returned connection.
466
467 # Wait for the worker tasks to terminate
468 gather(*workers, timeout=timeout)

470 # Close the connections that were still in the pool
471 for conn in connections:

475 for pos in waiting:

671 If failed, reschedule a new attempt in the future for a few times, then

756 # to the state, to avoid to create a reference loop.
757 # Also disable the warning for open connection in conn.__del__
758 conn._pool = None
759
760 # Early bailout in case the pool is closed. Don't add anything to the
761 # state. There is still a remote chance that the pool will be closed
762 # between here and entering the lock. Therefore we will make another
763 # check later.
764 if self._closed:
765 self._close_connection(conn)
766 return
767
768 # Critical section: if there is a client waiting give it the connection
769 # otherwise put it back into the pool.
770 with self._lock:
771 # Check if the pool was closed by the time we arrived here. It is
772 # unlikely but it doesn't seem impossible, if the worker was adding
773 # this connection while the main process is closing the pool.
774 # Now that we are in the critical section we know for real.
775 if self._closed:
776 self._close_connection(conn)
777 return
778
779 while self._waiting:
780 # If there is a client waiting (which is still waiting and
781 # hasn't timed out), give it the connection and notify it.
782
783 if self._waiting.popleft().set(conn):
784 break
785 else:
786 # No client waiting for a connection: put it back into the pool
787 self._pool.append(conn)
788 # If we have been asked to wait for pool init, notify the
789 # waiter if the pool is full.
790 if self._pool_full_event and len(self._pool) >= self._min_size:
791 self._pool_full_event.set()
792
793 def _reset_connection(self, conn: CT) -> None:
794 """
795 Bring a connection to IDLE state or close it.
796 """
797 if (status := conn.pgconn.transaction_status) == TransactionStatus.IDLE:
798 pass
799 elif status == TransactionStatus.UNKNOWN:
800 # Connection closed
801 return
802 elif status == TransactionStatus.INTRANS or status == TransactionStatus.INERROR:
803 # Connection returned with an active transaction
804 logger.warning("rolling back returned connection: %s", conn)
805 try:
806 conn.rollback()
807 except CLIENT_EXCEPTIONS as ex:
808 logger.warning(
809 "rollback failed: %s: %s. Discarding connection %s",
810 ex.__class__.__name__,
811 ex,
812 conn,
813 )
814 self._close_connection(conn)
815 elif status == TransactionStatus.ACTIVE:
816 # Connection returned during an operation. Bad... just close it.
817 logger.warning("closing returned connection: %s", conn)
818 self._close_connection(conn)
819
820 if self._reset:
821 try:
822 self._reset(conn)
823 if (status := conn.pgconn.transaction_status) != TransactionStatus.IDLE:
824 sname = TransactionStatus(status).name
825 raise e.ProgrammingError(
826 f"connection left in status {sname} by reset function {self._reset}: discarded"
827 )
828 except CLIENT_EXCEPTIONS as ex:
829 logger.warning("error resetting connection: %s", ex)
830 self._close_connection(conn)
831
832 def _close_connection(self, conn: CT) -> None:
833 conn._pool = None
834 conn.close()
835
836 def _shrink_pool(self) -> None:
837 to_close: CT | None = None
838
839 with self._lock:
840 # Reset the min number of connections used
841 nconns_min = self._nconns_min
842 self._nconns_min = len(self._pool)
843
844 # If the pool can shrink and connections were unused, drop one
845 if self._nconns > self._min_size and nconns_min > 0 and self._pool:
846 to_close = self._pool.popleft()
847 self._nconns -= 1
848 self._nconns_min -= 1
849
850 if to_close:
851 logger.info(
852 "shrinking pool %r to %s because %s unused connections in the last %s sec",
853 self.name,
854 self._nconns,
855 nconns_min,
856 self.max_idle,
857 )
858 self._close_connection(to_close)
859
860 def _get_measures(self) -> dict[str, int]:
861 rv = super()._get_measures()
862 rv[self._REQUESTS_WAITING] = len(self._waiting)
863 return rv

866class WaitingClient(Generic[CT]):
867 """A position in a queue for a client waiting for a connection."""
868
869 __slots__ = ("conn", "error", "_cond")
870
871 def __init__(self) -> None:
872 self.conn: CT | None = None
873 self.error: BaseException | None = None
874
875 # The WaitingClient behaves in a way similar to an Event, but we need
876 # to notify reliably the flagger that the waiter has "accepted" the
877 # message and it hasn't timed out yet, otherwise the pool may give a
878 # connection to a client that has already timed out getconn(), which
879 # will be lost.
880 self._cond = Condition()
881
882 def wait(self, timeout: float) -> CT:
883 """Wait for a connection to be set and return it.
884
885 Raise an exception if the wait times out or if fail() is called.
886 """
887 with self._cond:
888 if not (self.conn or self.error):
889 try:
890 if not self._cond.wait(timeout):
891 self.error = PoolTimeout(
892 f"couldn't get a connection after {timeout:.2f} sec"
893 )
894 except CLIENT_EXCEPTIONS as ex:
895 self.error = ex
896
897 if self.conn:
898 return self.conn
899 else:
900 assert self.error
901 raise self.error
902
903 def set(self, conn: CT) -> bool:
904 """Signal the client waiting that a connection is ready.
905
906 Return True if the client has "accepted" the connection, False
907 otherwise (typically because wait() has timed out).
908 """
909 with self._cond:
910 if self.conn or self.error:
911 return False
912
913 self.conn = conn
914 self._cond.notify_all()
915 return True
916
917 def fail(self, error: Exception) -> bool:
918 """Signal the client that, alas, they won't have a connection today.
919
920 Return True if the client has "accepted" the error, False otherwise
921 (typically because wait() has timed out).
922 """
923 with self._cond:
924 if self.conn or self.error:
925 return False
926
927 self.error = error
928 self._cond.notify_all()
929 return True

932class MaintenanceTask(ABC):
933 """A task to run asynchronously to maintain the pool state."""
934
935 def __init__(self, pool: ConnectionPool[Any]):
936 self.pool = ref(pool)
937
938 def __repr__(self) -> str:
939 pool = self.pool()
940 name = repr(pool.name) if pool else "<pool is gone>"
941 return f"<{self.__class__.__name__} {name} at 0x{id(self):x}>"
942
943 def run(self) -> None:
944 """Run the task.
945
946 This usually happens in a worker. Call the concrete _run()
947 implementation, if the pool is still alive.
948 """
949 pool = self.pool()
950 if not pool or pool.closed:
951 # Pool is no more working. Quietly discard the operation.
952 logger.debug("task run discarded: %s", self)
953 return
954
955 logger.debug("task running in %s: %s", current_thread_name(), self)
956 self._run(pool)

962 run the task for real in order to free the scheduler immediately.
963 """
964 pool = self.pool()
965 if not pool or pool.closed:
966 # Pool is no more working. Quietly discard the operation.
967 logger.debug("task tick discarded: %s", self)
968 return
