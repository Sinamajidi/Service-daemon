378 def _query_params_args(
379 self,
380 command: bytes,
381 param_values: Sequence[abc.Buffer | None] | None,
382 param_types: Sequence[int] | None = None,
383 param_formats: Sequence[int] | None = None,
384 result_format: int = Format.TEXT,
385 ) -> Any:
386 if not isinstance(command, bytes):
387 raise TypeError(f"bytes expected, got {type(command)} instead")
388
389 aparams: Array[c_char_p] | None
390 alenghts: Array[c_int] | None
391 if param_values:
392 nparams = len(param_values)
393 aparams = (c_char_p * nparams)(
394 *(
395 # convert bytearray/memoryview to bytes
396 b if b is None or isinstance(b, bytes) else bytes(b)
397 for b in param_values
398 )
399 )
400 alenghts = (c_int * nparams)(*(len(p) if p else 0 for p in param_values))
401 else:
402 nparams = 0
403 aparams = alenghts = None
404
405 atypes: Array[impl.Oid] | None
406 if not param_types:
407 atypes = None
408 else:
409 if len(param_types) != nparams:
410 raise ValueError(
411 "got %d param_values but %d param_types"
412 % (nparams, len(param_types))
413 )
414 atypes = (impl.Oid * nparams)(*param_types)
415
416 if not param_formats:
417 aformats = None
418 else:
419 if len(param_formats) != nparams:
420 raise ValueError(
421 "got %d param_values but %d param_formats"
422 % (nparams, len(param_formats))
423 )
424 aformats = (c_int * nparams)(*param_formats)
425
426 return (
427 self._pgconn_ptr,
428 command,
429 nparams,
430 atypes,
431 aparams,
432 alenghts,
433 aformats,
434 result_format,
435 )
