1"""
2Generators implementing communication protocols with the libpq
3
4Certain operations (connection, querying) are an interleave of libpq calls and
5waiting for the socket to be ready. This module contains the code to execute
6the operations, yielding a polling state whenever there is to wait. The
7functions in the `waiting` module are the ones who wait more or less
8cooperatively for the socket to be ready and make these generators continue.
9
10These generators yield `Wait` objects whenever an operation would block. These
11generators assume the connection fileno will not change. In case of the
12connection function, where the fileno may change, the generators yield pairs
13(fileno, `Wait`).
14
15The generator can be restarted sending the appropriate `Ready` state when the
16file descriptor is ready. If a None value is sent, it means that the wait
17function timed out without any file descriptor becoming ready; in this case the
18generator should probably yield the same value again in order to wait more.
19"""

62def _connect(conninfo: str, *, timeout: float = 0.0) -> PQGenConn[PGconn]:
63 """
64 Generator to create a database connection without blocking.
65 """
66 deadline = monotonic() + timeout if timeout else 0.0
67
68 # To debug slowdown during connection:
69 #
70 # $ PSYCOPG_IMPL=python python
71 # >>> import logging
72 # >>> logging.basicConfig(level=logging.DEBUG, format='%(asctime)s %(message)s')
73 # >>> logging.getLogger("psycopg").setLevel(logging.DEBUG)
74
75 conn = pq.PGconn.connect_start(conninfo.encode())
76 logger.debug("connection started: %s", conn)
77 while True:
78 if conn.status == BAD:
79 encoding = conninfo_encoding(conninfo)
80 raise e.OperationalError(
81 f"connection is bad: {conn.get_error_message(encoding)}", pgconn=conn
82 )
83
84 status = conn.connect_poll()
85 logger.debug("connection polled: %s", conn)
86
87 if status == POLL_READING or status == POLL_WRITING:
88 wait = WAIT_R if status == POLL_READING else WAIT_W
89 while True:
90 ready = yield conn.socket, wait
91 if deadline and monotonic() > deadline:
92 raise e.ConnectionTimeout("connection timeout expired")
93 if ready:
94 break
95
96 elif status == POLL_OK:
97 break
98 elif status == POLL_FAILED:
99 encoding = conninfo_encoding(conninfo)
100 raise e.OperationalError(
101 f"connection failed: {conn.get_error_message(encoding)}",
102 pgconn=e.finish_pgconn(conn),
103 )
104 else:
105 raise e.InternalError(
106 f"unexpected poll status: {status}", pgconn=e.finish_pgconn(conn)
107 )
108
109 conn.nonblocking = 1
110 return conn

113def _cancel(cancel_conn: PGcancelConn, *, timeout: float = 0.0) -> PQGenConn[None]:
114 deadline = monotonic() + timeout if timeout else 0.0
115 while True:
116 if deadline and monotonic() > deadline:
117 raise e.CancellationTimeout("cancellation timeout expired")
118
119 if (status := cancel_conn.poll()) == POLL_OK:
120 break
121 elif status == POLL_READING:
122 yield cancel_conn.socket, WAIT_R
123 elif status == POLL_WRITING:
124 yield cancel_conn.socket, WAIT_W
125 elif status == POLL_FAILED:
126 raise e.OperationalError(
127 f"cancellation failed: {cancel_conn.get_error_message()}"
128 )
129 else:
130 raise e.InternalError(f"unexpected poll status: {status}")

149def _send(pgconn: PGconn) -> PQGen[None]:
150 """
151 Generator to send a query to the server without blocking.
152
153 The query must have already been sent using `pgconn.send_query()` or
154 similar. Flush the query and then return the result using nonblocking
155 functions.
156
157 After this generator has finished you may want to cycle using `fetch()`
158 to retrieve the results available.
159 """
160 while pgconn.flush() != 0:
161
162 while not (ready := (yield WAIT_RW)):
163 continue
164
165 if ready & READY_R:
166 # This call may read notifies: they will be saved in the
167 # PGconn buffer and passed to Python later, in `fetch()`.
168 pgconn.consume_input()

171def _fetch_many(pgconn: PGconn) -> PQGen[list[PGresult]]:
172 """
173 Generator retrieving results from the database without blocking.
174
175 The query must have already been sent to the server, so pgconn.flush() has
176 already returned 0.
177
178 Return the list of results returned by the database (whether success
179 or error).
180 """
181 results: list[PGresult] = []
182 while True:
183 try:
184 res = yield from _fetch(pgconn)
185 except e.DatabaseError:
186 # What might have happened here is that a previuos error disconnected
187 # the connection, for example a idle in transaction timeout.
188 # Check if we had received an error before: if that's the case
189 # just exit the loop. Our callers will handle this error and raise
190 # it as an exception.
191 if any(res.status == FATAL_ERROR for res in results):
192 break
193 else:
194 raise
195
196 if not res:
197 break
198
199 results.append(res)
200 status = res.status
201 if status == COPY_IN or status == COPY_OUT or status == COPY_BOTH:
202 # After entering copy mode the libpq will create a phony result
203 # for every request so let's break the endless loop.
204 break
205
206 if status == PIPELINE_SYNC:
207 # PIPELINE_SYNC is not followed by a NULL, but we return it alone
208 # similarly to other result sets.
209 assert len(results) == 1, results
210 break
211
212 return results

215def _fetch(pgconn: PGconn) -> PQGen[PGresult | None]:
216 """
217 Generator retrieving a single result from the database without blocking.
218
219 The query must have already been sent to the server, so pgconn.flush() has
220 already returned 0.
221
222 Return a result from the database (whether success or error).
223 """
224 if pgconn.is_busy():
225 while not (yield WAIT_R):
226 continue
227
228 while True:
229 pgconn.consume_input()
230 if not pgconn.is_busy():
231 break
232 while not (yield WAIT_R):
233 continue
234
235 _consume_notifies(pgconn)
236
237 return pgconn.get_result()

240def _pipeline_communicate(
241 pgconn: PGconn, commands: deque[PipelineCommand]
242) -> PQGen[list[list[PGresult]]]:
243 """Generator to send queries from a connection in pipeline mode while also
244 receiving results.
245
246 Return a list results, including single PIPELINE_SYNC elements.
247 """
248 results = []
249
250 while True:
251 while not (ready := (yield WAIT_RW)):
252 continue
253
254 if ready & READY_R:
255 pgconn.consume_input()
256 _consume_notifies(pgconn)
257
258 res: list[PGresult] = []
259 while not pgconn.is_busy():
260 if (r := pgconn.get_result()) is None:
261 if not res:
262 break
263 results.append(res)
264 res = []
265 elif (status := r.status) == PIPELINE_SYNC:
266 assert not res
267 results.append([r])
268 elif status == COPY_IN or status == COPY_OUT or status == COPY_BOTH:
269 # This shouldn't happen, but insisting hard enough, it will.
270 # For instance, in test_executemany_badquery(), with the COPY
271 # statement and the AsyncClientCursor, which disables
272 # prepared statements).
273 # Bail out from the resulting infinite loop.
274 raise e.NotSupportedError("COPY cannot be used in pipeline mode")
275 else:
276 res.append(r)
277
278 if ready & READY_W:
279 pgconn.flush()
280 if not commands:
281 break
282 commands.popleft()()
283
284 return results

307def copy_from(pgconn: PGconn) -> PQGen[memoryview | PGresult]:
308 while True:
309 nbytes, data = pgconn.get_copy_data(1)
310 if nbytes != 0:
311 break
312
313 # would block
314 while not (yield WAIT_R):
315 continue
316 pgconn.consume_input()
317
318 if nbytes > 0:
319 # some data
320 return data
321
322 # Retrieve the final result of copy
323
324 if len(results := (yield from _fetch_many(pgconn))) > 1:
325 # TODO: too brutal? Copy worked.
326 raise e.ProgrammingError("you cannot mix COPY with other operations")
327
328 if (result := results[0]).status != COMMAND_OK:
329 raise e.error_from_result(result, encoding=pgconn._encoding)
330
331 return result

334def copy_to(pgconn: PGconn, buffer: Buffer, flush: bool = True) -> PQGen[None]:
335 # Retry enqueuing data until successful.
336 #
337 # WARNING! This can cause an infinite loop if the buffer is too large. (see
338 # ticket #255). We avoid it in the Copy object by splitting a large buffer
339 # into smaller ones. We prefer to do it there instead of here in order to
340 # do it upstream the queue decoupling the writer task from the producer one.
341 while pgconn.put_copy_data(buffer) == 0:
342 while not (yield WAIT_W):
343 continue
344
345 # Flushing often has a good effect on macOS because memcpy operations
346 # seem expensive on this platform so accumulating a large buffer has a
347 # bad effect (see #745).
348 if flush:
349 # Repeat until it the message is flushed to the server
350 while True:
351 while not (yield WAIT_W):
352 continue
353
354 if pgconn.flush() == 0:
355 break

358def copy_end(pgconn: PGconn, error: bytes | None) -> PQGen[PGresult]:
359 # Retry enqueuing end copy message until successful
360 while pgconn.put_copy_end(error) == 0:
361 while not (yield WAIT_W):
362 continue
363
364 # Repeat until it the message is flushed to the server
365 while True:
366 while not (yield WAIT_W):
367 continue
368
369 if pgconn.flush() == 0:
370 break
371
372 # Retrieve the final result of copy
373 (result,) = yield from _fetch_many(pgconn)
374 if result.status != COMMAND_OK:
375 raise e.error_from_result(result, encoding=pgconn._encoding)
376
377 return result
