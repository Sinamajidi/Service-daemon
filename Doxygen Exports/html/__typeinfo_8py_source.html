94 @classmethod
95 def _fetch(cls: type[T], conn: Connection[Any], name: str) -> T | None:
96 # This might result in a nested transaction. What we want is to leave
97 # the function with the connection in the state we found (either idle
98 # or intrans)
99 try:
100 from psycopg import Cursor
101
102 with conn.transaction(), Cursor(conn, row_factory=dict_row) as cur:
103 if conn_encoding(conn) == "ascii":
104 cur.execute("set local client_encoding to utf8")
105 cur.execute(cls._get_info_query(conn), {"name": name})
106 recs = cur.fetchall()
107 except e.UndefinedObject:
108 return None
109
110 return cls._from_records(name, recs)

112 @classmethod
113 async def _fetch_async(
114 cls: type[T], conn: AsyncConnection[Any], name: str
115 ) -> T | None:
116 try:
117 from psycopg import AsyncCursor
118
119 async with conn.transaction():
120 async with AsyncCursor(conn, row_factory=dict_row) as cur:
121 if conn_encoding(conn) == "ascii":
122 await cur.execute("set local client_encoding to utf8")
123 await cur.execute(cls._get_info_query(conn), {"name": name})
124 recs = await cur.fetchall()
125 except e.UndefinedObject:
126 return None
127

159 @classmethod
160 def _get_info_query(cls, conn: BaseConnection[Any]) -> QueryNoTemplate:
161 return sql.SQL(
162 """\
163SELECT
164 typname AS name, oid, typarray AS array_oid,
165 oid::regtype::text AS regtype, typdelim AS delimiter
166FROM pg_type t
167WHERE t.oid = {regtype}
168ORDER BY t.oid
169"""
170 ).format(regtype=cls._to_regtype(conn))

172 @classmethod
173 def _has_to_regtype_function(cls, conn: BaseConnection[Any]) -> bool:
174 # to_regtype() introduced in PostgreSQL 9.4 and CockroachDB 22.2
175 if (info := conn.info).vendor == "PostgreSQL":
176 return info.server_version >= 90400
177 elif info.vendor == "CockroachDB":
178 return info.server_version >= 220200
179 else:
180 return False

182 @classmethod
183 def _to_regtype(cls, conn: BaseConnection[Any]) -> sql.SQL:
184 # `to_regtype()` returns the type oid or NULL, unlike the :: operator,
185 # which returns the type or raises an exception, which requires
186 # a transaction rollback and leaves traces in the server logs.
187
188 if cls._has_to_regtype_function(conn):
189 return sql.SQL("to_regtype(%(name)s)")
190 else:
191 return sql.SQL("%(name)s::regtype")

240 def add(self, info: TypeInfo) -> None:
241 self._ensure_own_state()
242 if info.oid:
243 self._registry[info.oid] = info
244 if info.array_oid:
245 self._registry[info.array_oid] = info
246 self._registry[info.name] = info
247
248 if info.regtype and info.regtype not in self._registry:
249 self._registry[info.regtype] = info

251 # Allow info to customise further their relation with the registry
252 info._added(self)

267 def __getitem__(self, key: RegistryKey) -> TypeInfo:
268 """
269 Return info about a type, specified by name or oid
270
271 :param key: the name or oid of the type to look for.
272
273 Raise KeyError if not found.
274 """
275 if isinstance(key, str):
276 if key.endswith("[]"):
277 key = key[:-2]

279 raise TypeError(f"the key must be an oid or a name, got {type(key)}")
