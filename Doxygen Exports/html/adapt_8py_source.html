52 def quote(self, obj: Any) -> Buffer:
53 """
54 By default return the `dump()` value quoted and sanitised, so
55 that the result can be used to build a SQL string. This works well
56 for most types and you won't likely have to implement this method in a
57 subclass.
58 """
59 if (value := self.dump(obj)) is None:
60 return b"NULL"
61
62 if self.connection:
63 esc = pq.Escaping(self.connection.pgconn)
64 # escaping and quoting
65 return esc.escape_literal(value)
66
67 # This path is taken when quote is asked without a connection,
68 # usually it means by psycopg.sql.quote() or by
69 # 'Composible.as_string(None)'. Most often than not this is done by
70 # someone generating a SQL file to consume elsewhere.
71
72 # No quoting, only quote escaping, random bs escaping. See further.
73 esc = pq.Escaping()
74 out = esc.escape_string(value)
75
76 # b"\\" in memoryview doesn't work so search for the ascii value
77 if ORD_BS not in out:
78 # If the string has no backslash, the result is correct and we
79 # don't need to bother with standard_conforming_strings.
80 return b"'" + out + b"'"
81
82 # The libpq has a crazy behaviour: PQescapeString uses the last
83 # standard_conforming_strings setting seen on a connection. This
84 # means that backslashes might be escaped or might not.
85 #
86 # A syntax E'\\' works everywhere, whereas E'\' is an error. OTOH,
87 # if scs is off, '\\' raises a warning and '\' is an error.
88 #
89 # Check what the libpq does, and if it doesn't escape the backslash
90 # let's do it on our own. Never mind the race condition.
91 rv: bytes = b" E'" + out + b"'"
92 if esc.escape_string(b"\\") == b"\\":
93 rv = rv.replace(b"\\", b"\\\\")
94 return rv
