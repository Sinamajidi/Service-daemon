26def make_conninfo(conninfo: str = "", **kwargs: ConnParam) -> str:
27 """
28 Merge a string and keyword params into a single conninfo string.
29
30 :param conninfo: A `connection string`__ as accepted by PostgreSQL.
31 :param kwargs: Parameters overriding the ones specified in `!conninfo`.
32 :return: A connection string valid for PostgreSQL, with the `!kwargs`
33 parameters merged.
34
35 Raise `~psycopg.ProgrammingError` if the input doesn't make a valid
36 conninfo string.
37
38 .. __: https://www.postgresql.org/docs/current/libpq-connect.html
39 #LIBPQ-CONNSTRING
40 """
41 if not conninfo and not kwargs:
42 return ""
43
44 # If no kwarg specified don't mung the conninfo but check if it's correct.
45 # Make sure to return a string, not a subtype, to avoid making Liskov sad.
46 if not kwargs:
47 _parse_conninfo(conninfo)
48 return str(conninfo)
49
50 # Override the conninfo with the parameters
51 # Drop the None arguments
52 kwargs = {k: v for (k, v) in kwargs.items() if v is not None}
53
54 if conninfo:
55 tmp = conninfo_to_dict(conninfo)
56 tmp.update(kwargs)
57 kwargs = tmp
58
59 conninfo = " ".join(f"{k}={_param_escape(str(v))}" for (k, v) in kwargs.items())
60
61 # Verify the result is valid
62 _parse_conninfo(conninfo)
63
64 return conninfo

67def conninfo_to_dict(conninfo: str = "", **kwargs: ConnParam) -> ConnDict:
68 """
69 Convert the `!conninfo` string into a dictionary of parameters.
70
71 :param conninfo: A `connection string`__ as accepted by PostgreSQL.
72 :param kwargs: Parameters overriding the ones specified in `!conninfo`.
73 :return: Dictionary with the parameters parsed from `!conninfo` and
74 `!kwargs`.
75
76 Raise `~psycopg.ProgrammingError` if `!conninfo` is not a a valid connection
77 string.
78
79 .. __: https://www.postgresql.org/docs/current/libpq-connect.html
80 #LIBPQ-CONNSTRING
81 """
82 opts = _parse_conninfo(conninfo)
83 rv: ConnDict = {
84 opt.keyword.decode(): opt.val.decode() for opt in opts if opt.val is not None
85 }
86 for k, v in kwargs.items():
87 if v is not None:
88 rv[k] = v
89 return rv

92def _parse_conninfo(conninfo: str) -> list[pq.ConninfoOption]:
93 """
94 Verify that `!conninfo` is a valid connection string.
95
96 Raise ProgrammingError if the string is not valid.
97
98 Return the result of pq.Conninfo.parse() on success.
99 """
100 try:
101 return pq.Conninfo.parse(conninfo.encode())
102 except e.OperationalError as ex:
103 raise e.ProgrammingError(str(ex)) from None

110def _param_escape(s: str) -> str:
111 """
112 Apply the escaping rule required by PQconnectdb
113 """
114 if not s:
115 return "''"
116
117 if re_space.search(s := re_escape.sub(r"\\\1", s)):
118 s = "'" + s + "'"
119
120 return s

123def timeout_from_conninfo(params: ConnDict) -> int:
124 """
125 Return the timeout in seconds from the connection parameters.
126 """
127 # Follow the libpq convention:
128 #
129 # - 0 or less means no timeout (but we will use a default to simulate
130 # the socket timeout)
131 # - at least 2 seconds.
132 #
133 # See connectDBComplete in fe-connect.c
134 value: str | int | None = _conninfo_utils.get_param(params, "connect_timeout")
135 if value is None:
136 value = _DEFAULT_CONNECT_TIMEOUT
137 try:
138 timeout = int(float(value))
139 except ValueError:
140 raise e.ProgrammingError(f"bad value for connect_timeout: {value!r}") from None
141
142 if timeout <= 0:
143 # The sync connect function will stop on the default socket timeout
144 # Because in async connection mode we need to enforce the timeout
145 # ourselves, we need a finite value.
146 timeout = _DEFAULT_CONNECT_TIMEOUT
147 elif timeout < 2:
148 # Enforce a 2s min
149 timeout = 2
150
151 return timeout
