74 def wait(self, timeout: float = 30.0) -> None:
75 """
76 Create a connection for test.
77
78 Calling this function will verify that the connectivity with the
79 database works as expected. However the connection will not be stored
80 in the pool.
81
82 Close the pool, and raise `PoolTimeout`, if not ready within *timeout*
83 sec.
84 """
85 self._check_open_getconn()
86
87 with self._lock:
88 assert not self._pool_full_event
89 self._pool_full_event = Event()
90
91 logger.info("waiting for pool %r initialization", self.name)
92 self.run_task(AddConnection(self))
93 if not self._pool_full_event.wait(timeout):
94 self.close() # stop all the tasks
95 raise PoolTimeout(f"pool initialization incomplete after {timeout} sec")
96
97 with self._lock:
98 assert self._pool_full_event
99 self._pool_full_event = None

101 logger.info("pool %r is ready to use", self.name)

103 def _get_ready_connection(self, timeout: float | None) -> CT | None:
104 if timeout is not None and timeout <= 0.0:
105 raise PoolTimeout()
106
107 conn: CT | None = None
108 if self.max_size == 0 or self._nconns < self.max_size:
109 # Create a new connection for the client
110 try:
111 conn = self._connect(timeout=timeout)
112 except ConnectionTimeout as ex:
113 raise PoolTimeout(str(ex)) from None
114 self._nconns += 1
115 elif self.max_waiting and len(self._waiting) >= self.max_waiting:
116 self._stats[self._REQUESTS_ERRORS] += 1
117 raise TooManyRequests(

123 def _maybe_close_connection(self, conn: CT) -> bool:
124 # Close the connection if no client is waiting for it, or if the pool
125 # is closed. For extra refcare remove the pool reference from it.
126 # Maintain the stats.
127 with self._lock:
128 if not self._closed and self._waiting:
129 return False

131 conn._pool = None
132 if conn.pgconn.transaction_status == TransactionStatus.UNKNOWN:
133 self._stats[self._RETURNS_BAD] += 1

156 def _add_to_pool(self, conn: CT) -> None:
157 # Remove the pool reference from the connection before returning it
158 # to the state, to avoid to create a reference loop.
159 # Also disable the warning for open connection in conn.__del__
160 conn._pool = None
161
162 # Critical section: if there is a client waiting give it the connection
163 # otherwise put it back into the pool.
164 with self._lock:
165 while self._waiting:
166 # If there is a client waiting (which is still waiting and
167 # hasn't timed out), give it the connection and notify it.
168 if self._waiting.popleft().set(conn):
