74class Range(Generic[T]):
75 """Python representation for a PostgreSQL range type.
76
77 :param lower: lower bound for the range. `!None` means unbound
78 :param upper: upper bound for the range. `!None` means unbound
79 :param bounds: one of the literal strings ``()``, ``[)``, ``(]``, ``[]``,
80 representing whether the lower or upper bounds are included
81 :param empty: if `!True`, the range is empty
82
83 """
84
85 __slots__ = ("_lower", "_upper", "_bounds")
86
87 def __init__(
88 self,
89 lower: T | None = None,
90 upper: T | None = None,
91 bounds: str = "[)",
92 empty: bool = False,
93 ):
94 if not empty:
95 if bounds not in ("[)", "(]", "()", "[]"):
96 raise ValueError("bound flags not valid: %r" % bounds)
97
98 self._lower = lower
99 self._upper = upper
100
101 # Make bounds consistent with infs
102 if lower is None and bounds[0] == "[":
103 bounds = "(" + bounds[1]
104 if upper is None and bounds[1] == "]":
105 bounds = bounds[0] + ")"
106
107 self._bounds = bounds
108 else:
109 self._lower = self._upper = None
110 self._bounds = ""
111
112 def __repr__(self) -> str:
113 if self._bounds:
114 args = f"{self._lower!r}, {self._upper!r}, {self._bounds!r}"
115 else:
116 args = "empty=True"
117
118 return f"{self.__class__.__name__}({args})"
119
120 def __str__(self) -> str:
121 if not self._bounds:
122 return "empty"
123
124 items = [
125 self._bounds[0],
126 str(self._lower),
127 ", ",
128 str(self._upper),
129 self._bounds[1],
130 ]
131 return "".join(items)
132
133 @property
134 def lower(self) -> T | None:
135 """The lower bound of the range. `!None` if empty or unbound."""
136 return self._lower
137
138 @property
139 def upper(self) -> T | None:
140 """The upper bound of the range. `!None` if empty or unbound."""
141 return self._upper
142
143 @property
144 def bounds(self) -> str:
145 """The bounds string (two characters from '[', '(', ']', ')')."""
146 return self._bounds
147
148 @property
149 def isempty(self) -> bool:
150 """`!True` if the range is empty."""
151 return not self._bounds
152
153 @property
154 def lower_inf(self) -> bool:
155 """`!True` if the range doesn't have a lower bound."""
156 if not self._bounds:
157 return False
158 return self._lower is None
159
160 @property
161 def upper_inf(self) -> bool:
162 """`!True` if the range doesn't have an upper bound."""
163 if not self._bounds:
164 return False
165 return self._upper is None
166
167 @property
168 def lower_inc(self) -> bool:
169 """`!True` if the lower bound is included in the range."""
170 if not self._bounds or self._lower is None:
171 return False
172 return self._bounds[0] == "["
173
174 @property
175 def upper_inc(self) -> bool:
176 """`!True` if the upper bound is included in the range."""
177 if not self._bounds or self._upper is None:
178 return False
179 return self._bounds[1] == "]"
180
181 def __contains__(self, x: T) -> bool:
182 if not self._bounds:
183 return False
184
185 if self._lower is not None:
186 if self._bounds[0] == "[":
187 # It doesn't seem that Python has an ABC for ordered types.
188 if x < self._lower: # type: ignore[operator]
189 return False
190 elif x <= self._lower: # type: ignore[operator]
191 return False
192
193 if self._upper is not None:
194 if self._bounds[1] == "]":
195 if x > self._upper: # type: ignore[operator]
196 return False
197 elif x >= self._upper: # type: ignore[operator]
198 return False
199
200 return True
201
202 def __bool__(self) -> bool:
203 return bool(self._bounds)
204
205 def __eq__(self, other: Any) -> bool:
206 if not isinstance(other, Range):
207 return False
208 return (
209 self._lower == other._lower
210 and self._upper == other._upper
211 and self._bounds == other._bounds
212 )
213
214 def __hash__(self) -> int:
215 return hash((self._lower, self._upper, self._bounds))
216
217 # as the postgres docs describe for the server-side stuff,
218 # ordering is rather arbitrary, but will remain stable
219 # and consistent.
220
221 def __lt__(self, other: Any) -> bool:
222 if not isinstance(other, Range):
223 return NotImplemented
224 for attr in ("_lower", "_upper", "_bounds"):
225 self_value = getattr(self, attr)
226 if self_value == (other_value := getattr(other, attr)):
227 pass
228 elif self_value is None:
229 return True
230 elif other_value is None:
231 return False
232 else:
233 return cast(bool, self_value < other_value)
234 return False
235
236 def __le__(self, other: Any) -> bool:
237 return self == other or self < other # type: ignore
238
239 def __gt__(self, other: Any) -> bool:
240 if isinstance(other, Range):
241 return other < self
242 else:
243 return NotImplemented
244
245 def __ge__(self, other: Any) -> bool:
246 return self == other or self > other # type: ignore
247
248 def __getstate__(self) -> dict[str, Any]:
249 return {
250 slot: getattr(self, slot) for slot in self.__slots__ if hasattr(self, slot)
251 }
252
253 def __setstate__(self, state: dict[str, Any]) -> None:
254 for slot, value in state.items():
255 setattr(self, slot, value)

286class BaseRangeDumper(RecursiveDumper):
287 def __init__(self, cls: type, context: AdaptContext | None = None):
288 super().__init__(cls, context)
289 self.sub_dumper: Dumper | None = None
290 self._adapt_format = PyFormat.from_pq(self.format)
291
292 def get_key(self, obj: Range[Any], format: PyFormat) -> DumperKey:
293 # If we are a subclass whose oid is specified we don't need upgrade
294 if self.cls is not Range:
295 return self.cls
296 if (item := self._get_item(obj)) is not None:
297 sd = self._tx.get_dumper(item, self._adapt_format)
298 return (self.cls, sd.get_key(item, format))
299 else:
300 return (self.cls,)
301
302 def upgrade(self, obj: Range[Any], format: PyFormat) -> BaseRangeDumper:
303 # If we are a subclass whose oid is specified we don't need upgrade
304 if self.cls is not Range:
305 return self
306 if (item := self._get_item(obj)) is None:
307 return self
308
309 dumper: BaseRangeDumper
310 if type(item) is int:
311 # postgres won't cast int4range -> int8range so we must use
312 # text format and unknown oid here
313 sd = self._tx.get_dumper(item, PyFormat.TEXT)
314 dumper = RangeDumper(self.cls, self._tx)
315 dumper.sub_dumper = sd
316 dumper.oid = INVALID_OID
317 return dumper
318
319 sd = self._tx.get_dumper(item, format)
320 dumper = type(self)(self.cls, self._tx)
321 dumper.sub_dumper = sd
322 if sd.oid == INVALID_OID and isinstance(item, str):
323 # Work around the normal mapping where text is dumped as unknown
324 dumper.oid = self._get_range_oid(TEXT_OID)
325 else:
326 dumper.oid = self._get_range_oid(sd.oid)
327
328 return dumper

330 def _get_item(self, obj: Range[Any]) -> Any:
331 """
332 Return a member representative of the range
333 """
334 rv = obj.lower
335 return rv if rv is not None else obj.upper
336
337 def _get_range_oid(self, sub_oid: int) -> int:
338 """
339 Return the oid of the range from the oid of its elements.
340 """
341 info = self._tx.adapters.types.get_by_subtype(RangeInfo, sub_oid)
342 return info.oid if info else INVALID_OID

345class RangeDumper(BaseRangeDumper):
346 """
347 Dumper for range types.
348
349 The dumper can upgrade to one specific for a different range type.
350 """
351
352 def dump(self, obj: Range[Any]) -> Buffer | None:
353 if (item := self._get_item(obj)) is not None:
354 dump = self._tx.get_dumper(item, self._adapt_format).dump
355 else:
356 dump = fail_dump
357
358 return dump_range_text(obj, dump)
359
360
361def dump_range_text(obj: Range[Any], dump: DumpFunc) -> Buffer:
362 if obj.isempty:
363 return b"empty"
364
365 parts: list[Buffer] = [b"[" if obj.lower_inc else b"("]
366
367 def dump_item(item: Any) -> Buffer:
368 if (ad := dump(item)) is None:
369 return b""
370 elif not ad:
371 return b'""'
372 elif _re_needs_quotes.search(ad):
373 return b'"' + _re_esc.sub(rb"\1\1", ad) + b'"'
374 else:
375 return ad
376
377 if obj.lower is not None:
378 parts.append(dump_item(obj.lower))
379
380 parts.append(b",")
381
382 if obj.upper is not None:
383 parts.append(dump_item(obj.upper))
384
385 parts.append(b"]" if obj.upper_inc else b")")
386
387 return b"".join(parts)
388
389
390_re_needs_quotes = re.compile(rb'[",\\\s()\‍[\‍]]')
391_re_esc = re.compile(rb"([\\\"])")

406def dump_range_binary(obj: Range[Any], dump: DumpFunc) -> Buffer:
407 if not obj:
408 return _EMPTY_HEAD
409
410 out = bytearray([0]) # will replace the head later
411
412 head = 0
413 if obj.lower_inc:
414 head |= RANGE_LB_INC
415 if obj.upper_inc:
416 head |= RANGE_UB_INC
417
418 if obj.lower is not None:
419 if (data := dump(obj.lower)) is not None:
420 out += pack_len(len(data))
421 out += data
422 else:
423 head |= RANGE_LB_INF
424 else:
425 head |= RANGE_LB_INF
426
427 if obj.upper is not None:
428 if (data := dump(obj.upper)) is not None:
429 out += pack_len(len(data))
430 out += data
431 else:
432 head |= RANGE_UB_INF
433 else:
434 head |= RANGE_UB_INF
435
436 out[0] = head
437 return out

457class RangeLoader(BaseRangeLoader[T]):
458 def load(self, data: Buffer) -> Range[T]:
459 return load_range_text(data, self._load)[0]

462def load_range_text(data: Buffer, load: LoadFunc) -> tuple[Range[Any], int]:
463 if data == b"empty":
464 return Range(empty=True), 5
465 if (m := _re_range.match(data)) is None:
466 raise e.DataError(
467 f"failed to parse range: '{bytes(data).decode('utf8', 'replace')}'"
468 )
469
470 lower = None
471 if (item := m.group(3)) is None:
472 if (item := m.group(2)) is not None:
473 lower = load(_re_undouble.sub(rb"\1", item))
474 else:
475 lower = load(item)
476
477 upper = None
478 if (item := m.group(5)) is None:
479 if (item := m.group(4)) is not None:
480 upper = load(_re_undouble.sub(rb"\1", item))
481 else:
482 upper = load(item)
483
484 bounds = (m.group(1) + m.group(6)).decode()
485
486 return Range(lower, upper, bounds), m.end()
487

489_re_range = re.compile(
490 rb"""
491 ( \‍(|\‍[ ) # lower bound flag
492 (?: # lower bound:
493 " ( (?: [^"] | "")* ) " # - a quoted string
494 | ( [^",]+ ) # - or an unquoted string
495 )? # - or empty (not caught)
496 ,
497 (?: # upper bound:
498 " ( (?: [^"] | "")* ) " # - a quoted string
499 | ( [^"\‍)\‍]]+ ) # - or an unquoted string
500 )? # - or empty (not caught)
501 ( \‍)|\‍] ) # upper bound flag
502 """,
503 re.VERBOSE,
504)
505
506_re_undouble = re.compile(rb'(["\\])\1')

509class RangeBinaryLoader(BaseRangeLoader[T]):
510 format = Format.BINARY
511
512 def load(self, data: Buffer) -> Range[T]:
513 return load_range_binary(data, self._load)
514
515
516def load_range_binary(data: Buffer, load: LoadFunc) -> Range[Any]:
517 if (head := data[0]) & RANGE_EMPTY:
518 return Range(empty=True)
519
520 lb = "[" if head & RANGE_LB_INC else "("
521 ub = "]" if head & RANGE_UB_INC else ")"
522
523 pos = 1 # after the head
524 if head & RANGE_LB_INF:
525 min = None
526 else:
527 length = unpack_len(data, pos)[0]
528 pos += 4
529 min = load(data[pos : pos + length])
530 pos += length
531
532 if head & RANGE_UB_INF:
533 max = None
534 else:
535 length = unpack_len(data, pos)[0]
536 pos += 4
537 max = load(data[pos : pos + length])
538 pos += length
539
540 return Range(min, max, lb + ub)

543def register_range(info: RangeInfo, context: AdaptContext | None = None) -> None:
544 """Register the adapters to load and dump a range type.
545
546 :param info: The object with the information about the range to register.
547 :param context: The context where to register the adapters. If `!None`,
548 register it globally.
549
550 Register loaders so that loading data of this type will result in a `Range`
551 with bounds parsed as the right subtype.
552
553 .. note::
554
555 Registering the adapters doesn't affect objects already created, even
556 if they are children of the registered context. For instance,
557 registering the adapter globally doesn't affect already existing
558 connections.
559 """
560 # A friendly error warning instead of an AttributeError in case fetch()
561 # failed and it wasn't noticed.
562 if not info:
563 raise TypeError("no info passed. Is the requested range available?")
564
565 # Register arrays and type info
566 info.register(context)
567
568 adapters = context.adapters if context else postgres.adapters
569
570 # generate and register a customized text loader
571 loader: type[BaseRangeLoader[Any]]
572 loader = _make_loader(info.name, info.subtype_oid)
573 adapters.register_loader(info.oid, loader)
574
575 # generate and register a customized binary loader
576 loader = _make_binary_loader(info.name, info.subtype_oid)
577 adapters.register_loader(info.oid, loader)
