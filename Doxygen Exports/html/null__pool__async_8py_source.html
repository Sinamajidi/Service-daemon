70 async def wait(self, timeout: float = 30.0) -> None:
71 """
72 Create a connection for test.
73
74 Calling this function will verify that the connectivity with the
75 database works as expected. However the connection will not be stored
76 in the pool.
77
78 Close the pool, and raise `PoolTimeout`, if not ready within *timeout*
79 sec.
80 """
81 self._check_open_getconn()
82
83 async with self._lock:
84 assert not self._pool_full_event
85 self._pool_full_event = AEvent()
86
87 logger.info("waiting for pool %r initialization", self.name)
88 self.run_task(AddConnection(self))
89 if not await self._pool_full_event.wait_timeout(timeout):
90 await self.close() # stop all the tasks
91 raise PoolTimeout(f"pool initialization incomplete after {timeout} sec")
92
93 async with self._lock:
94 assert self._pool_full_event
95 self._pool_full_event = None
96
97 logger.info("pool %r is ready to use", self.name)

99 async def _get_ready_connection(self, timeout: float | None) -> ACT | None:
100 if timeout is not None and timeout <= 0.0:
101 raise PoolTimeout()
102
103 conn: ACT | None = None
104 if self.max_size == 0 or self._nconns < self.max_size:
105 # Create a new connection for the client
106 try:
107 conn = await self._connect(timeout=timeout)
108 except ConnectionTimeout as ex:
109 raise PoolTimeout(str(ex)) from None
110 self._nconns += 1
111
112 elif self.max_waiting and len(self._waiting) >= self.max_waiting:
113 self._stats[self._REQUESTS_ERRORS] += 1
114 raise TooManyRequests(
115 f"the pool {self.name!r} has already"
116 + f" {len(self._waiting)} requests waiting"
117 )
118 return conn

120 async def _maybe_close_connection(self, conn: ACT) -> bool:
121 # Close the connection if no client is waiting for it, or if the pool
122 # is closed. For extra refcare remove the pool reference from it.
123 # Maintain the stats.
124 async with self._lock:
125 if not self._closed and self._waiting:
126 return False
127
128 conn._pool = None
129 if conn.pgconn.transaction_status == TransactionStatus.UNKNOWN:
130 self._stats[self._RETURNS_BAD] += 1
131 await self._close_connection(conn)
132 self._nconns -= 1
133 return True

153 async def _add_to_pool(self, conn: ACT) -> None:
154 # Remove the pool reference from the connection before returning it
155 # to the state, to avoid to create a reference loop.
156 # Also disable the warning for open connection in conn.__del__
157 conn._pool = None
158
159 # Critical section: if there is a client waiting give it the connection
160 # otherwise put it back into the pool.
161 async with self._lock:
162 while self._waiting:
163 # If there is a client waiting (which is still waiting and
164 # hasn't timed out), give it the connection and notify it.
165 if await self._waiting.popleft().set(conn):
166 break
167 else:
168 # No client waiting for a connection: close the connection
169 await self._close_connection(conn)
170 # If we have been asked to wait for pool init, notify the
171 # waiter if the pool is ready.
172 if self._pool_full_event:
173 self._pool_full_event.set()
174 else:
175 # The connection created by wait shouldn't decrease the
176 # count of the number of connection used.
177 self._nconns -= 1
