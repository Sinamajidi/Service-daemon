23async def conninfo_attempts_async(params: ConnMapping) -> list[ConnDict]:
24 """Split a set of connection params on the single attempts to perform.
25
26 A connection param can perform more than one attempt more than one ``host``
27 is provided.
28
29 Also perform async resolution of the hostname into hostaddr. Because a host
30 can resolve to more than one address, this can lead to yield more attempts
31 too. Raise `OperationalError` if no host could be resolved.
32
33 Because the libpq async function doesn't honour the timeout, we need to
34 reimplement the repeated attempts.
35 """
36 last_exc = None
37 attempts = []
38 if prefer_standby := get_param(params, "target_session_attrs") == "prefer-standby":
39 params = {k: v for k, v in params.items() if k != "target_session_attrs"}
40
41 for attempt in split_attempts(params):
42 try:
43 attempts.extend(await _resolve_hostnames(attempt))
44 except OSError as ex:
45 last_exc = e.OperationalError(
46 f"failed to resolve host {attempt.get('host')!r}: {ex}"
47 )
48 logger.debug("%s", last_exc)
49
50 if not attempts:
51 assert last_exc
52 # We couldn't resolve anything
53 raise last_exc
54
55 if get_param(params, "load_balance_hosts") == "random":
56 shuffle(attempts)
57
58 # Order matters: first try all the load-balanced host in standby mode,
59 # then allow primary
60 if prefer_standby:
61 attempts = [
62 {**a, "target_session_attrs": "standby"} for a in attempts
63 ] + attempts
64
65 return attempts

68async def _resolve_hostnames(params: ConnDict) -> list[ConnDict]:
69 """
70 Perform async DNS lookup of the hosts and return a list of connection attempts.
71
72 If a ``host`` param is present but not ``hostname``, resolve the host
73 addresses asynchronously.
74
75 :param params: The input parameters, for instance as returned by
76 `~psycopg.conninfo.conninfo_to_dict()`. The function expects at most
77 a single entry for host, hostaddr because it is designed to further
78 process the input of split_attempts().
79
80 :return: A list of attempts to make (to include the case of a hostname
81 resolving to more than one IP).
82 """
83 host = get_param(params, "host")
84 if not host or host.startswith("/") or host[1:2] == ":":
85 # Local path, or no host to resolve
86 return [params]
87
88 if get_param(params, "hostaddr"):
89 # Already resolved
90 return [params]
91
92 if is_ip_address(host):
93 # If the host is already an ip address don't try to resolve it
94 return [{**params, "hostaddr": host}]
95
96 if not (port := get_param(params, "port")):
97 port_def = get_param_def("port")
98 port = port_def and port_def.compiled or "5432"
99
100 if True: # ASYNC:
101 loop = asyncio.get_running_loop()
102 ans = await loop.getaddrinfo(
103 host, port, proto=socket.IPPROTO_TCP, type=socket.SOCK_STREAM
104 )
105 else:
106 ans = socket.getaddrinfo(
107 host, port, proto=socket.IPPROTO_TCP, type=socket.SOCK_STREAM
108 )
109
110 return [{**params, "hostaddr": item[4][0]} for item in ans]
