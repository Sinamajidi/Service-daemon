99 def _enter_gen(self) -> PQGen[None]:
100 if self.status != self.Status.NOT_STARTED:
101 raise TypeError("transaction blocks can be used only once")
102 self.status = self.Status.ACTIVE
103
104 self._push_savepoint()
105 for command in self._get_enter_commands():
106 yield from self._conn._exec_command(command)
107
108 def _exit_gen(
109 self,
110 exc_type: type[BaseException] | None,
111 exc_val: BaseException | None,
112 exc_tb: TracebackType | None,
113 ) -> PQGen[bool]:
114 if not exc_val and not self.force_rollback:
115 yield from self._commit_gen()
116 return False
117 else:
118 # try to rollback, but if there are problems (connection in a bad
119 # state) just warn without clobbering the exception bubbling up.
120 try:
121 return (yield from self._rollback_gen(exc_val))
122 except OutOfOrderTransactionNesting:
123 # Clobber an exception happened in the block with the exception
124 # caused by out-of-order transaction detected, so make the
125 # behaviour consistent with _commit_gen and to make sure the
126 # user fixes this condition, which is unrelated from
127 # operational error that might arise in the block.
128 raise
129 except Exception as exc2:
130 logger.warning("error ignored in rollback of %s: %s", self, exc2)
131 return False
132
133 def _commit_gen(self) -> PQGen[None]:
134 ex = self._pop_savepoint("commit")
135 self.status = self.Status.COMMITTED
136 if ex:
137 raise ex
138
139 for command in self._get_commit_commands():
140 yield from self._conn._exec_command(command)
141
142 def _rollback_gen(self, exc_val: BaseException | None) -> PQGen[bool]:
143 if isinstance(exc_val, Rollback):
144 logger.debug(f"{self._conn}: Explicit rollback from: ", exc_info=True)
145
146 ex = self._pop_savepoint("rollback")
147
148 if isinstance(exc_val, Rollback) or self.force_rollback:
149 self.status = self.Status.ROLLED_BACK_EXPLICITLY
150 else:
151 self.status = self.Status.ROLLED_BACK_WITH_ERROR
152
153 if ex:
154 raise ex
155
156 for command in self._get_rollback_commands():
157 yield from self._conn._exec_command(command)
158
159 # Also clear the prepared statements cache.
160 self._conn._prepared.clear()
161 yield from self._conn._prepared.maintain_gen(self._conn)
162
163 if isinstance(exc_val, Rollback):
164 if not exc_val.transaction or exc_val.transaction is self:
165 return True # Swallow the exception
166
167 return False

169 def _get_enter_commands(self) -> Iterator[bytes]:
170 if self._outer_transaction:
171 yield self._conn._get_tx_start_command()

180 def _get_commit_commands(self) -> Iterator[bytes]:
181 if self._savepoint_name and not self._outer_transaction:
182 yield (
183 sql.SQL("RELEASE {}")
184 .format(sql.Identifier(self._savepoint_name))
185 .as_bytes(self._conn)
186 )
187
188 if self._outer_transaction:
189 assert not self._conn._num_transactions
190 yield b"COMMIT"
191
192 def _get_rollback_commands(self) -> Iterator[bytes]:
193 if self._savepoint_name and not self._outer_transaction:
194 yield (
195 sql.SQL("ROLLBACK TO {n}")
196 .format(n=sql.Identifier(self._savepoint_name))
197 .as_bytes(self._conn)
198 )
199 yield (
200 sql.SQL("RELEASE {n}")
201 .format(n=sql.Identifier(self._savepoint_name))
202 .as_bytes(self._conn)
203 )
204
205 if self._outer_transaction:
206 assert not self._conn._num_transactions
207 yield b"ROLLBACK"

209 def _push_savepoint(self) -> None:
210 """
211 Push the transaction on the connection transactions stack.
212
213 Also set the internal state of the object and verify consistency.
214 """
215 self._outer_transaction = self.pgconn.transaction_status == IDLE
216 if self._outer_transaction:
217 # outer transaction: if no name it's only a begin, else
218 # there will be an additional savepoint
219 assert not self._conn._num_transactions
220 else:
221 # inner transaction: it always has a name
222 if not self._savepoint_name:
223 self._savepoint_name = f"_pg3_{self._conn._num_transactions + 1}"
224
225 self._stack_index = self._conn._num_transactions
226 self._conn._num_transactions += 1
227
228 def _pop_savepoint(self, action: str) -> Exception | None:
229 """
230 Pop the transaction from the connection transactions stack.
231
232 Also verify the state consistency.
233 """
234 self._conn._num_transactions -= 1
235 if self._conn._num_transactions == self._stack_index:
236 return None
237
238 return OutOfOrderTransactionNesting(
239 f"transaction {action} at the wrong nesting level: {self}"
240 )
