78 def get_parameters(self) -> dict[str, str]:
79 """Return the connection parameters values.
80
81 Return all the parameters set to a non-default value, which might come
82 either from the connection string and parameters passed to
83 `~Connection.connect()` or from environment variables. The password
84 is never returned (you can read it using the `password` attribute).
85 """
86 pyenc = self.encoding
87
88 # Get the known defaults to avoid reporting them
89 defaults = {
90 i.keyword: i.compiled
91 for i in pq.Conninfo.get_defaults()
92 if i.compiled is not None
93 }
94 # Not returned by the libq. Bug? Bet we're using SSH.
95 defaults.setdefault(b"channel_binding", b"prefer")
96 defaults[b"passfile"] = str(Path.home() / ".pgpass").encode()
97
98 return {
99 i.keyword.decode(pyenc): i.val.decode(pyenc)
100 for i in self.pgconn.info
101 if i.val is not None
102 and i.keyword != b"password"
103 and i.val != defaults.get(i.keyword)
104 }
