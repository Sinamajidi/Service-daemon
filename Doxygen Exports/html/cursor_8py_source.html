151 self,

153 params: Params | None = None,
154 *,
155 binary: bool | None = None,
156 size: int = 1,
157 ) -> Iterator[Row]:
158 """
159 Iterate row-by-row on a result from the database.
160
161 :param size: if greater than 1, results will be retrieved by chunks of
162 this size from the server (but still yielded row-by-row); this is only
163 available from version 17 of the libpq.
164 """
165 if self._pgconn.pipeline_status:
166 raise e.ProgrammingError("stream() cannot be used in pipeline mode")
167
168 with self._conn.lock:
169 try:
170 self._conn.wait(
171 self._stream_send_gen(query, params, binary=binary, size=size)
172 )
173 first = True
174 while res := self._conn.wait(self._stream_fetchone_gen(first)):
175 for pos in range(res.ntuples):
176 yield self._tx.load_row(pos, self._make_row)
177 first = False
178 except e._NO_TRACEBACK as ex:
179 raise ex.with_traceback(None)
180 finally:
181 if self._pgconn.transaction_status == ACTIVE:
182 # Try to cancel the query, then consume the results
183 # already received.
184 self._conn._try_cancel()
185 try:
186 while self._conn.wait(self._stream_fetchone_gen(first=False)):
187 pass
188 except Exception:
189 pass
190
191 # Try to get out of ACTIVE state. Just do a single attempt, which
192 # should work to recover from an error or query cancelled.
193 try:
194 self._conn.wait(self._stream_fetchone_gen(first=False))
195 except Exception:
196 pass
