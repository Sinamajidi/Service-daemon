120 async def executemany(
121 self, query: Query, params_seq: Iterable[Params], *, returning: bool = False
122 ) -> None:
123 """
124 Execute the same command with a sequence of input data.
125 """
126 try:
127 async with self._conn.lock:
128 if AsyncPipeline.is_supported():

131 if self._conn._pipeline:

150 async def stream(
151 self,
152 query: Query,
153 params: Params | None = None,
154 *,
155 binary: bool | None = None,
156 size: int = 1,
157 ) -> AsyncIterator[Row]:
158 """
159 Iterate row-by-row on a result from the database.
160
161 :param size: if greater than 1, results will be retrieved by chunks of
162 this size from the server (but still yielded row-by-row); this is only
163 available from version 17 of the libpq.
164 """
165 if self._pgconn.pipeline_status:
166 raise e.ProgrammingError("stream() cannot be used in pipeline mode")
167
168 async with self._conn.lock:
169 try:
170 await self._conn.wait(
171 self._stream_send_gen(query, params, binary=binary, size=size)
172 )
173 first = True
174 while res := await self._conn.wait(self._stream_fetchone_gen(first)):
175 for pos in range(res.ntuples):
176 yield self._tx.load_row(pos, self._make_row)
177 first = False
178 except e._NO_TRACEBACK as ex:
179 raise ex.with_traceback(None)
180 finally:
181 if self._pgconn.transaction_status == ACTIVE:
182 # Try to cancel the query, then consume the results
183 # already received.
184 await self._conn._try_cancel()
185 try:
186 while await self._conn.wait(
187 self._stream_fetchone_gen(first=False)
188 ):
189 pass
190 except Exception:
191 pass
192
193 # Try to get out of ACTIVE state. Just do a single attempt, which
194 # should work to recover from an error or query cancelled.
195 try:
196 await self._conn.wait(self._stream_fetchone_gen(first=False))
197 except Exception:
198 pass

215 async def set_result(self, index: int) -> Self:
216 """
217 Move to a specific result set.
218
219 :arg index: index of the result to go to
220 :type index: `!int`
221
222 More than one result will be available after executing calling
223 `executemany()` or `execute()` with more than one query.
224
225 `!index` is 0-based and supports negative values, counting from the end,
226 the same way you can index items in a list.
227
228 The function returns self, so that the result may be followed by a
229 fetch operation. See `results()` for details.
230 """
231 if not -len(self._results) <= index < len(self._results):
232 raise IndexError(
233 f"index {index} out of range: {len(self._results)} result(s) available"
234 )
235 if index < 0:
236 index = len(self._results) + index
237
238 self._select_current_result(index)
239 return self

241 async def fetchone(self) -> Row | None:
242 """
243 Return the next record from the current result set.
244
245 Return `!None` the result set is finished.

247 :rtype: Row | None, with Row defined by `row_factory`

249 await self._fetch_pipeline()

252 record = self._tx.load_row(self._pos, self._make_row)

257 async def fetchmany(self, size: int = 0) -> list[Row]:
258 """
259 Return the next `!size` records from the current result set.
260
261 `!size` default to `!self.arraysize` if not specified.
262
263 :rtype: Sequence[Row], with Row defined by `row_factory`
264 """
265 await self._fetch_pipeline()
266 res = self._check_result_for_fetch()
267
268 if not size:
269 size = self.arraysize
270 records = self._tx.load_rows(
271 self._pos, min(self._pos + size, res.ntuples), self._make_row
272 )
273 self._pos += len(records)
274 return records

300 async def scroll(self, value: int, mode: str = "relative") -> None:
301 """
302 Move the cursor in the result set to a new position according to mode.
303
304 If `!mode` is ``'relative'`` (default), `!value` is taken as offset to
305 the current position in the result set; if set to ``'absolute'``,
306 `!value` states an absolute target position.
307
308 Raise `!IndexError` in case a scroll operation would leave the result
309 set. In this case the position will not change.
310 """
311 await self._fetch_pipeline()
312 self._scroll(value, mode)
