92 def _has_feature(self, feature: str, want_version: int, check: bool) -> bool:
93 """
94 Check is a version is supported.
95
96 If `check` is true, raise an exception with an explicative message
97 explaining why the feature is not supported.
98
99 The expletive messages, are left to the user.
100 """
101 if feature in self._cache:
102 msg = self._cache[feature]
103 else:
104 msg = self._get_unsupported_message(feature, want_version)
105 self._cache[feature] = msg
106
107 if not msg:
108 return True
109 elif check:
110 raise NotSupportedError(msg)
111 else:
112 return False
113
114 def _get_unsupported_message(self, feature: str, want_version: int) -> str:
115 """
116 Return a descriptinve message to describe why a feature is unsupported.
117
118 Return an empty string if the feature is supported.
119 """
120 if pq.version() < want_version:
121 return (
122 f"the feature '{feature}' is not available:"
123 f" the client libpq version (imported from {self._libpq_source()})"
124 f" is {pq.version_pretty(pq.version())}; the feature"
125 f" requires libpq version {pq.version_pretty(want_version)}"
126 " or newer"
127 )
128
129 elif pq.__build_version__ < want_version:
130 return (
131 f"the feature '{feature}' is not available:"
132 f" you are using a psycopg[{pq.__impl__}] libpq wrapper built"
133 f" with libpq version {pq.version_pretty(pq.__build_version__)};"
134 " the feature requires libpq version"
135 f" {pq.version_pretty(want_version)} or newer"
136 )
137 else:
138 return ""
