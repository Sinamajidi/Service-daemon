559class TimestamptzBinaryLoader(Loader):
560 format = Format.BINARY
561
562 def __init__(self, oid: int, context: AdaptContext | None = None):
563 super().__init__(oid, context)
564 self._timezone = get_tzinfo(self.connection.pgconn if self.connection else None)
565
566 def load(self, data: Buffer) -> datetime:
567 micros = unpack_int8(data)[0]
568 try:
569 ts = _pg_datetimetz_epoch + timedelta(microseconds=micros)
570 return ts.astimezone(self._timezone)
571 except OverflowError:
572 # If we were asked about a timestamp which would overflow in UTC,
573 # but not in the desired timezone (e.g. datetime.max at Chicago
574 # timezone) we can still save the day by shifting the value by the
575 # timezone offset and then replacing the timezone.
576 if self._timezone:
577 if utcoff := self._timezone.utcoffset(
578 datetime.min if micros < 0 else datetime.max
579 ):
580 usoff = 1_000_000 * int(utcoff.total_seconds())
581 try:
582 ts = _pg_datetime_epoch + timedelta(microseconds=micros + usoff)
583 except OverflowError:
584 pass # will raise downstream
585 else:
586 return ts.replace(tzinfo=self._timezone)
587
588 if micros <= 0:
589 raise DataError("timestamp too small (before year 1)") from None
590 else:
591 raise DataError("timestamp too large (after year 10K)") from None
