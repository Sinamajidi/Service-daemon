95 def format_for_error(self) -> str:
96 # Convert comma-separated specifiers into "A, B, ..., F and G"
97 # This makes the specifier a bit more "human readable", without
98 # risking a change in meaning. (Hopefully! Not all edge cases have
99 # been checked)
100 parts = [s.strip() for s in str(self).split(",")]
101 if len(parts) == 0:
102 return ""
103 elif len(parts) == 1:
104 return parts[0]
105
106 return ", ".join(parts[:-1]) + " and " + parts[-1]

158class RequiresPythonRequirement(Requirement):

162 self.specifier = specifier
163 self._specifier_string = str(specifier) # for faster __eq__
164 self._hash: int | None = None
165 self._candidate = match

167 # Pre-compute candidate lookup to avoid repeated specifier checks

169 self._candidate_lookup: CandidateLookup = (match, None)

171 self._candidate_lookup = (None, None)

173 def __str__(self) -> str:

181 return self._hash

183 self._hash = hash((self._specifier_string, self._candidate))
184 return self._hash

186 def __eq__(self, other: Any) -> bool:
187 if not isinstance(other, RequiresPythonRequirement):
