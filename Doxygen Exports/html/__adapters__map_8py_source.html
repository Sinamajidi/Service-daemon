26class AdaptersMap:
27 r"""
28 Establish how types should be converted between Python and PostgreSQL in
29 an `~psycopg.abc.AdaptContext`.
30
31 `!AdaptersMap` maps Python types to `~psycopg.adapt.Dumper` classes to
32 define how Python types are converted to PostgreSQL, and maps OIDs to
33 `~psycopg.adapt.Loader` classes to establish how query results are
34 converted to Python.
35
36 Every `!AdaptContext` object has an underlying `!AdaptersMap` defining how
37 types are converted in that context, exposed as the
38 `~psycopg.abc.AdaptContext.adapters` attribute: changing such map allows
39 to customise adaptation in a context without changing separated contexts.
40
41 When a context is created from another context (for instance when a
42 `~psycopg.Cursor` is created from a `~psycopg.Connection`), the parent's
43 `!adapters` are used as template for the child's `!adapters`, so that every
44 cursor created from the same connection use the connection's types
45 configuration, but separate connections have independent mappings.
46
47 Once created, `!AdaptersMap` are independent. This means that objects
48 already created are not affected if a wider scope (e.g. the global one) is
49 changed.
50
51 The connections adapters are initialised using a global `!AdptersMap`
52 template, exposed as `psycopg.adapters`: changing such mapping allows to
53 customise the type mapping for every connections created afterwards.
54
55 The object can start empty or copy from another object of the same class.
56 Copies are copy-on-write: if the maps are updated make a copy. This way
57 extending e.g. global map by a connection or a connection map from a cursor
58 is cheap: a copy is only made on customisation.
59 """

114 def register_dumper(self, cls: type | str | None, dumper: type[Dumper]) -> None:
115 """
116 Configure the context to use `!dumper` to convert objects of type `!cls`.
117
118 If two dumpers with different `~Dumper.format` are registered for the
119 same type, the last one registered will be chosen when the query
120 doesn't specify a format (i.e. when the value is used with a ``%s``
121 "`~PyFormat.AUTO`" placeholder).
122
123 :param cls: The type to manage.
124 :param dumper: The dumper to register for `!cls`.
125
126 If `!cls` is specified as string it will be lazy-loaded, so that it
127 will be possible to register it without importing it before. In this
128 case it should be the fully qualified name of the object (e.g.
129 ``"uuid.UUID"``).
130
131 If `!cls` is None, only use the dumper when looking up using
132 `get_dumper_by_oid()`, which happens when we know the Postgres type to
133 adapt to, but not the Python type that will be adapted (e.g. in COPY
134 after using `~psycopg.Copy.set_types()`).
135
136 """
137 if not (cls is None or isinstance(cls, (str, type))):
138 raise TypeError(
139 f"dumpers should be registered on classes, got {cls} instead"
140 )
141
142 if _psycopg:
143 dumper = self._get_optimised(dumper)
144
145 # Register the dumper both as its format and as auto
146 # so that the last dumper registered is used in auto (%s) format
147 if cls:
148 for fmt in (PyFormat.from_pq(dumper.format), PyFormat.AUTO):
149 if not self._own_dumpers[fmt]:
150 self._dumpers[fmt] = self._dumpers[fmt].copy()
151 self._own_dumpers[fmt] = True
152
153 self._dumpers[fmt][cls] = dumper
154
155 # Register the dumper by oid, if the oid of the dumper is fixed
156 if dumper.oid:
157 if not self._own_dumpers_by_oid[dumper.format]:
158 self._dumpers_by_oid[dumper.format] = self._dumpers_by_oid[
159 dumper.format
160 ].copy()
161 self._own_dumpers_by_oid[dumper.format] = True
162
163 self._dumpers_by_oid[dumper.format][dumper.oid] = dumper

165 def register_loader(self, oid: int | str, loader: type[Loader]) -> None:
166 """
167 Configure the context to use `!loader` to convert data of oid `!oid`.
168
169 :param oid: The PostgreSQL OID or type name to manage.
170 :param loader: The loar to register for `!oid`.
171
172 If `oid` is specified as string, it refers to a type name, which is
173 looked up in the `types` registry.
174
175 """
176 if isinstance(oid, str):
177 oid = self.types[oid].oid
178 if not isinstance(oid, int):
179 raise TypeError(f"loaders should be registered on oid, got {oid} instead")
180
181 if _psycopg:
182 loader = self._get_optimised(loader)
183
184 if not self._own_loaders[(fmt := loader.format)]:
185 self._loaders[fmt] = self._loaders[fmt].copy()
186 self._own_loaders[fmt] = True
187
188 self._loaders[fmt][oid] = loader
189 if self._register_loader_callback:
190 self._register_loader_callback(oid, loader)

192 def get_dumper(self, cls: type, format: PyFormat) -> type[Dumper]:
193 """
194 Return the dumper class for the given type and format.
195
196 Raise `~psycopg.ProgrammingError` if a class is not available.
197
198 :param cls: The class to adapt.
199 :param format: The format to dump to. If `~psycopg.adapt.PyFormat.AUTO`,
200 use the last one of the dumpers registered on `!cls`.
201 """
202 try:
203 # Fast path: the class has a known dumper.
204 return self._dumpers[format][cls]
205 except KeyError:
206 if format not in self._dumpers:
207 raise ValueError(f"bad dumper format: {format}")
208
209 # If the KeyError was caused by cls missing from dmap, let's
210 # look for different cases.
211 dmap = self._dumpers[format]
212
213 # Look for the right class, including looking at superclasses
214 for scls in cls.__mro__:
215 if scls in dmap:
216 return dmap[scls]
217
218 # If the adapter is not found, look for its name as a string
219 fqn = scls.__module__ + "." + scls.__qualname__
220 if (d := dmap.get(fqn)) is not None:
221 # Replace the class name with the class itself
222 dmap[scls] = d
223 dmap.pop(fqn, None)
224 return d
225
226 format = PyFormat(format)
227 raise e.ProgrammingError(
228 f"cannot adapt type {cls.__name__!r} using placeholder '%{format.value}'"
229 f" (format: {format.name})"
230 )

232 def get_dumper_by_oid(self, oid: int, format: pq.Format) -> type[Dumper]:
233 """
234 Return the dumper class for the given oid and format.
235
236 Raise `~psycopg.ProgrammingError` if a class is not available.
237
238 :param oid: The oid of the type to dump to.
239 :param format: The format to dump to.
240 """
241 try:
242 dmap = self._dumpers_by_oid[format]
243 except KeyError:
244 raise ValueError(f"bad dumper format: {format}")
245
246 try:
247 return dmap[oid]
248 except KeyError:
249 info = self.types.get(oid)
250 if info:
251 msg = (
252 f"cannot find a dumper for type {info.name} (oid {oid})"
253 f" format {pq.Format(format).name}"
254 )
255 else:
256 msg = (
257 f"cannot find a dumper for unknown type with oid {oid}"
258 f" format {pq.Format(format).name}"
259 )
260 raise e.ProgrammingError(msg)

263 """
264 Return the loader class for the given oid and format.
265
266 Return `!None` if not found.
267

273 @classmethod
274 def _get_optimised(self, cls: type[RV]) -> type[RV]:
275 """Return the optimised version of a Dumper or Loader class.
276
277 Return the input class itself if there is no optimised version.
278 """
279 try:
280 return self._optimised[cls]
281 except KeyError:
282 pass
283
284 # Check if the class comes from psycopg.types and there is a class
285 # with the same name in psycopg_c._psycopg.
286 from psycopg import types
287
288 if cls.__module__.startswith(types.__name__):
289 if new := cast(type[RV], getattr(_psycopg, cls.__name__, None)):
290 self._optimised[cls] = new
291 return new
292
293 self._optimised[cls] = cls
294 return cls
