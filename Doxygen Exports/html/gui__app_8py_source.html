248class PostgresServiceManager:
249 """! @brief Cross-platform PostgreSQL service helper."""
250 def run(self, action: str) -> tuple[bool, str]:
251 """! @brief Run the service command and return status."""
252 commands: list[list[str]] = []
253 system = platform.system().lower()
254 if system == "windows":
255 commands.append(["sc", action, "postgresql"])
256 elif system == "darwin":
257 commands.append(["brew", "services", action, "postgresql"])
258 else:
259 commands.append(["systemctl", action, "postgresql"])
260 commands.append(["service", "postgresql", action])
261
262 errors: list[str] = []
263 for cmd in commands:
264 try:
265 result = subprocess.run(cmd, capture_output=True, text=True, check=True)
266 message = result.stdout.strip() or f"PostgreSQL {action} command executed."
267 return True, message
268 except (OSError, subprocess.CalledProcessError) as exc:
269 detail = getattr(exc, "stderr", None) or str(exc)
270 errors.append(detail.strip())
271
272 error_summary = "; ".join(error for error in errors if error)
273 message = (
274 error_summary
275 or f"Unable to {action} PostgreSQL service. Check system permissions."
276 )
277 return False, message

280 """! @brief Start PostgreSQL service."""
281 return self.run("start")
282
283 def stop(self) -> tuple[bool, str]:
284 """! @brief Stop PostgreSQL service."""
285 return self.run("stop")
