35class NumericTypeModifier(TypeModifier):
36 """Handle numeric type modifier."""

38 def get_modifier(self, typemod: int) -> tuple[int, ...] | None:
39 precision = self.get_precision(typemod)
40 scale = self.get_scale(typemod)
41 return None if precision is None or scale is None else (precision, scale)

43 def get_precision(self, typemod: int) -> int | None:

46 def get_scale(self, typemod: int) -> int | None:
47 if typemod < 0:
48 return None

50 scale = (typemod - 4) & 0xFFFF
51 if scale >= 0x400:
52 scale = scale - 0x800
