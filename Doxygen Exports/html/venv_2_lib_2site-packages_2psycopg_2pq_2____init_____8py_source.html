50def import_from_libpq() -> None:
51 """
52 Import pq objects implementation from the best libpq wrapper available.
53
54 If an implementation is requested try to import only it, otherwise
55 try to import the best implementation available.
56 """
57 # import these names into the module on success as side effect
58 global __impl__, version, __build_version__
59 global PGconn, PGresult, Conninfo, Escaping, PGcancel, PGcancelConn
60
61 impl = os.environ.get("PSYCOPG_IMPL", "").lower()
62 module = None
63 attempts: list[str] = []
64
65 def handle_error(name: str, e: Exception) -> None:
66 if not impl:
67 msg = f"couldn't import psycopg '{name}' implementation: {e}"
68 attempts.append(msg)
69 else:
70 msg = f"couldn't import requested psycopg '{name}' implementation: {e}"
71 raise ImportError(msg) from e
72
73 # The best implementation: fast but requires the system libpq installed
74 if not impl or impl == "c":
75 try:
76 from psycopg_c import pq as module # type: ignore
77 except Exception as e:
78 handle_error("c", e)
79
80 # Second best implementation: fast and stand-alone
81 if not module and (not impl or impl == "binary"):
82 try:
83 from psycopg_binary import pq as module # type: ignore
84 except Exception as e:
85 handle_error("binary", e)
86
87 # Pure Python implementation, slow and requires the system libpq installed.
88 if not module and (not impl or impl == "python"):
89 try:
90 from . import pq_ctypes as module # type: ignore[assignment]
91 except Exception as e:
92 handle_error("python", e)
93
94 if module:
95 __impl__ = module.__impl__
96 version = module.version
97 PGconn = module.PGconn
98 PGresult = module.PGresult
99 Conninfo = module.Conninfo
100 Escaping = module.Escaping
101 PGcancel = module.PGcancel
102 PGcancelConn = module.PGcancelConn
103 __build_version__ = module.__build_version__
104 elif impl:
105 raise ImportError(f"requested psycopg implementation '{impl}' unknown")
106 else:
107 sattempts = "\n".join(f"- {attempt}" for attempt in attempts)
108 raise ImportError(
109 f"""\
110no pq wrapper available.
111Attempts made:
112{sattempts}"""
113 )
