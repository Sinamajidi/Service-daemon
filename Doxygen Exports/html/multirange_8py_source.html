31class MultirangeInfo(TypeInfo):
32 """Manage information about a multirange type."""
33
34 def __init__(
35 self,
36 name: str,
37 oid: int,
38 array_oid: int,
39 *,
40 regtype: str = "",
41 range_oid: int,
42 subtype_oid: int,
43 ):
44 super().__init__(name, oid, array_oid, regtype=regtype)
45 self.range_oid = range_oid
46 self.subtype_oid = subtype_oid

48 @classmethod
49 def _get_info_query(cls, conn: BaseConnection[Any]) -> QueryNoTemplate:
50 if conn.info.server_version < 140000:
51 raise e.NotSupportedError(
52 "multirange types are only available from PostgreSQL 14"
53 )
54 return sql.SQL(
55 """\
56SELECT t.typname AS name, t.oid AS oid, t.typarray AS array_oid,
57 t.oid::regtype::text AS regtype,
58 r.rngtypid AS range_oid, r.rngsubtype AS subtype_oid
59FROM pg_type t
60JOIN pg_range r ON t.oid = r.rngmultitypid
61WHERE t.oid = {regtype}
62"""
63 ).format(regtype=cls._to_regtype(conn))

71class Multirange(MutableSequence[Range[T]]):
72 """Python representation for a PostgreSQL multirange type.
73
74 :param items: Sequence of ranges to initialise the object.
75 """
76
77 def __init__(self, items: Iterable[Range[T]] = ()):
78 self._ranges: list[Range[T]] = list(map(self._check_type, items))
79
80 def _check_type(self, item: Any) -> Range[Any]:
81 if not isinstance(item, Range):
82 raise TypeError(
83 f"Multirange is a sequence of Range, got {type(item).__name__}"
84 )
85 return item
86
87 def __repr__(self) -> str:
88 return f"{self.__class__.__name__}({self._ranges!r})"
89
90 def __str__(self) -> str:
91 return f"{{{', '.join(map(str, self._ranges))}}}"
92
93 @overload
94 def __getitem__(self, index: int) -> Range[T]: ...
95
96 @overload
97 def __getitem__(self, index: slice) -> Multirange[T]: ...
98
99 def __getitem__(self, index: int | slice) -> Range[T] | Multirange[T]:
100 if isinstance(index, int):
101 return self._ranges[index]
102 else:
103 return Multirange(self._ranges[index])
104
105 def __len__(self) -> int:
106 return len(self._ranges)
107
108 @overload
109 def __setitem__(self, index: int, value: Range[T]) -> None: ...
110
111 @overload
112 def __setitem__(self, index: slice, value: Iterable[Range[T]]) -> None: ...
113
114 def __setitem__(
115 self, index: int | slice, value: Range[T] | Iterable[Range[T]]
116 ) -> None:
117 if isinstance(index, int):
118 self._check_type(value)
119 self._ranges[index] = self._check_type(value)
120 elif not isinstance(value, Iterable):
121 raise TypeError("can only assign an iterable")
122 else:
123 value = map(self._check_type, value)
124 self._ranges[index] = value
125
126 def __delitem__(self, index: int | slice) -> None:
127 del self._ranges[index]
128
129 def insert(self, index: int, value: Range[T]) -> None:
130 self._ranges.insert(index, self._check_type(value))
131
132 def __eq__(self, other: Any) -> bool:
133 if not isinstance(other, Multirange):
134 return False
135 return self._ranges == other._ranges
136
137 # Order is arbitrary but consistent
138
139 def __lt__(self, other: Any) -> bool:
140 if not isinstance(other, Multirange):
141 return NotImplemented
142 return self._ranges < other._ranges
143
144 def __le__(self, other: Any) -> bool:
145 return self == other or self < other # type: ignore
146
147 def __gt__(self, other: Any) -> bool:
148 if not isinstance(other, Multirange):
149 return NotImplemented
150 return self._ranges > other._ranges
151
152 def __ge__(self, other: Any) -> bool:
153 return self == other or self > other # type: ignore

183class BaseMultirangeDumper(RecursiveDumper):
184 def __init__(self, cls: type, context: AdaptContext | None = None):
185 super().__init__(cls, context)
186 self.sub_dumper: Dumper | None = None
187 self._adapt_format = PyFormat.from_pq(self.format)
188
189 def get_key(self, obj: Multirange[Any], format: PyFormat) -> DumperKey:
190 # If we are a subclass whose oid is specified we don't need upgrade
191 if self.cls is not Multirange:
192 return self.cls
193
194 if (item := self._get_item(obj)) is not None:
195 sd = self._tx.get_dumper(item, self._adapt_format)
196 return (self.cls, sd.get_key(item, format))
197 else:
198 return (self.cls,)
199
200 def upgrade(self, obj: Multirange[Any], format: PyFormat) -> BaseMultirangeDumper:
201 # If we are a subclass whose oid is specified we don't need upgrade
202 if self.cls is not Multirange:
203 return self
204
205 if (item := self._get_item(obj)) is None:
206 return self
207
208 dumper: BaseMultirangeDumper
209 if type(item) is int:
210 # postgres won't cast int4range -> int8range so we must use
211 # text format and unknown oid here
212 sd = self._tx.get_dumper(item, PyFormat.TEXT)
213 dumper = MultirangeDumper(self.cls, self._tx)
214 dumper.sub_dumper = sd
215 dumper.oid = INVALID_OID
216 return dumper
217
218 sd = self._tx.get_dumper(item, format)
219 dumper = type(self)(self.cls, self._tx)
220 dumper.sub_dumper = sd
221 if sd.oid == INVALID_OID and isinstance(item, str):
222 # Work around the normal mapping where text is dumped as unknown
223 dumper.oid = self._get_multirange_oid(TEXT_OID)
224 else:
225 dumper.oid = self._get_multirange_oid(sd.oid)
226
227 return dumper
228
229 def _get_item(self, obj: Multirange[Any]) -> Any:
230 """
231 Return a member representative of the multirange
232 """
233 for r in obj:
234 if r.lower is not None:
235 return r.lower
236 if r.upper is not None:
237 return r.upper
238 return None
239
240 def _get_multirange_oid(self, sub_oid: int) -> int:
241 """
242 Return the oid of the range from the oid of its elements.
243 """
244 info = self._tx.adapters.types.get_by_subtype(MultirangeInfo, sub_oid)
245 return info.oid if info else INVALID_OID

248class MultirangeDumper(BaseMultirangeDumper):
249 """
250 Dumper for multirange types.
251
252 The dumper can upgrade to one specific for a different range type.
253 """
254
255 def dump(self, obj: Multirange[Any]) -> Buffer | None:
256 if not obj:
257 return b"{}"
258
259 if (item := self._get_item(obj)) is not None:
260 dump = self._tx.get_dumper(item, self._adapt_format).dump
261 else:
262 dump = fail_dump
263
264 out: list[Buffer] = [b"{"]
265 for r in obj:
266 out.append(dump_range_text(r, dump))
267 out.append(b",")
268 out[-1] = b"}"
269 return b"".join(out)

272class MultirangeBinaryDumper(BaseMultirangeDumper):
273 format = Format.BINARY
274
275 def dump(self, obj: Multirange[Any]) -> Buffer | None:
276 if (item := self._get_item(obj)) is not None:
277 dump = self._tx.get_dumper(item, self._adapt_format).dump

297class MultirangeLoader(BaseMultirangeLoader[T]):
298 def load(self, data: Buffer) -> Multirange[T]:
299 if not data or data[0] != _START_INT:
300 raise e.DataError(
301 "malformed multirange starting with"
302 f" {bytes(data[:1]).decode('utf8', 'replace')}"
303 )
304
305 out = Multirange[T]()
306 if data == b"{}":
307 return out
308
309 pos = 1
310 data = data[pos:]
311 try:
312 while True:
313 r, pos = load_range_text(data, self._load)
314 out.append(r)
315
316 sep = data[pos] # can raise IndexError
317 if sep == _SEP_INT:
318 data = data[pos + 1 :]
319 continue
320 elif sep == _END_INT:
321 if len(data) == pos + 1:
322 return out
323 else:
324 raise e.DataError(
325 "malformed multirange: data after closing brace"
326 )
327 else:
328 raise e.DataError(
329 f"malformed multirange: found unexpected {chr(sep)}"
330 )
331
332 except IndexError:
333 raise e.DataError("malformed multirange: separator missing")
334
335 return out
336
337
338_SEP_INT = ord(",")
339_START_INT = ord("{")
340_END_INT = ord("}")

343class MultirangeBinaryLoader(BaseMultirangeLoader[T]):
344 format = Format.BINARY
345
346 def load(self, data: Buffer) -> Multirange[T]:
347 nelems = unpack_len(data, 0)[0]
348 pos = 4
349 out = Multirange[T]()
350 for i in range(nelems):
351 length = unpack_len(data, pos)[0]
352 pos += 4
353 out.append(load_range_binary(data[pos : pos + length], self._load))
354 pos += length
355
356 if pos != len(data):
357 raise e.DataError("unexpected trailing data in multirange")
358
359 return out

362def register_multirange(
363 info: MultirangeInfo, context: AdaptContext | None = None
364) -> None:
365 """Register the adapters to load and dump a multirange type.
366
367 :param info: The object with the information about the range to register.
368 :param context: The context where to register the adapters. If `!None`,
369 register it globally.
370
371 Register loaders so that loading data of this type will result in a `Range`
372 with bounds parsed as the right subtype.
373
374 .. note::
375
376 Registering the adapters doesn't affect objects already created, even
377 if they are children of the registered context. For instance,
378 registering the adapter globally doesn't affect already existing
379 connections.
380 """
381 # A friendly error warning instead of an AttributeError in case fetch()
382 # failed and it wasn't noticed.
383 if not info:
384 raise TypeError("no info passed. Is the requested multirange available?")
385
386 # Register arrays and type info
387 info.register(context)
388
389 adapters = context.adapters if context else postgres.adapters
390
391 # generate and register a customized text loader
392 loader: type[BaseMultirangeLoader[Any]]
393 loader = _make_loader(info.name, info.subtype_oid)
394 adapters.register_loader(info.oid, loader)
395
396 # generate and register a customized binary loader
397 loader = _make_binary_loader(info.name, info.subtype_oid)
398 adapters.register_loader(info.oid, loader)
399
400
401# Cache all dynamically-generated types to avoid leaks in case the types
402# cannot be GC'd.
403

405@cache
406def _make_loader(name: str, oid: int) -> type[MultirangeLoader[Any]]:
407 return type(f"{name.title()}Loader", (MultirangeLoader,), {"subtype_oid": oid})
