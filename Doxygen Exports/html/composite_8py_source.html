68SELECT

124class _SequenceBinaryDumper(Dumper, Generic[T], ABC):
125 """
126 Base class for binary dumpers taking an object and dumping it as a composite.
127
128 Abstract class: subclasses must specify the names and types of the fields
129 making the target composite and the `make_sequence()` static method to
130 convert the object to dump to a sequence of values.
131 """
132
133 format = pq.Format.BINARY
134 # Subclasses must set this attribute
135 info: CompositeInfo
136
137 def __init__(self, cls: type[T], context: abc.AdaptContext | None = None):
138 super().__init__(cls, context)
139
140 # Note: this class is not a RecursiveDumper because it would use the
141 # same Transformer of the context, which would confuse dump_sequence()
142 # in case the composite contains another composite. Make sure to use
143 # a separate Transformer instance instead.
144 self._tx = Transformer(context)
145 self._tx.set_dumper_types(self.info.field_types, self.format)
146
147 nfields = len(self.info.field_types)
148 self._formats = (PyFormat.from_pq(self.format),) * nfields
149
150 def dump(self, obj: T) -> Buffer | None:
151 seq = type(self).make_sequence(obj, self.info)
152 return _dump_binary_sequence(
153 seq, self.info.field_types, self._formats, self._tx

161class RecordLoader(RecursiveLoader):
162 """
163 Load a `record` field from PostgreSQL.
164
165 In text mode we don't have type information of the composite's fields, so
166 convert every item as text. Note that in binary loading we have per-field
167 oids instead.
168 """
169
170 def load(self, data: abc.Buffer) -> tuple[Any, ...]:
171 if data == b"()":
172 return ()
173
174 cast = self._tx.get_loader(TEXT_OID, self.format).load
175 record = _parse_text_record(data[1:-1])
176 for i in range(len(record)):
177 if (f := record[i]) is not None:
178 record[i] = cast(f)
179
180 return tuple(record)

183class RecordBinaryLoader(Loader):
184 """
185 Load a `record` field from PostgreSQL.
186
187 Unlike in text mode, the composite data contains oids of the fields,
188 so we can actually parse the records in its original types.
189 """
190
191 format = pq.Format.BINARY
192
193 def __init__(self, oid: int, context: abc.AdaptContext | None = None):
194 super().__init__(oid, context)
195 self._ctx = context
196 # Cache a transformer for each sequence of oid found.
197 # Usually there will be only one, but if there is more than one
198 # row in the same query (in different columns, or even in different
199 # records), oids might differ and we'd need separate transformers.
200 self._txs: dict[tuple[int, ...], abc.Transformer] = {}
201
202 def load(self, data: abc.Buffer) -> tuple[Any, ...]:
203 record, oids = _parse_binary_record(data)
204 if not record:
205 return ()
206
207 tx = self._get_transformer(tuple(oids))
208 return tx.load_sequence(record)
209
210 def _get_transformer(self, key: tuple[int, ...]) -> abc.Transformer:
211 if key in self._txs:
212 return self._txs[key]
213
214 tx = Transformer(self._ctx)
215 tx.set_loader_types([*key], self.format)
216 self._txs[key] = tx
217 return tx

220class _CompositeLoader(Loader, Generic[T], ABC):
221 """
222 Base class to create text loaders of specific composite types.
223
224 The class is complete but lack information about the fields types and
225 object factory. These will be added by register_composite(), which will
226 create a subclass of this class.
227 """
228
229 # Subclasses must set this attribute
230 info: CompositeInfo

232 def __init__(self, oid: int, context: abc.AdaptContext | None = None):
233 super().__init__(oid, context)
234 # Note: we cannot use the RecursiveLoader base class here because we
235 # always want a different Transformer instance, otherwise the types
236 # loaded will conflict with the types loaded by the record.

280def register_composite(
281 info: CompositeInfo,
282 context: abc.AdaptContext | None = None,
283 factory: Callable[..., T] | None = None,
284 *,
285 make_object: ObjectMaker[T] | None = None,
286 make_sequence: SequenceMaker[T] | None = None,
287) -> None:
288 """Register the adapters to load and dump a composite type.
289
290 :param info: The object with the information about the composite to register.
291 :type info: `CompositeInfo`
292 :param context: The context where to register the adapters. If `!None`,
293 register it globally.
294 :type context: `~psycopg.abc.AdaptContext` | `!None`
295 :param factory: Callable to create a Python object from the sequence of
296 attributes read from the composite.
297 :type factory: `!Callable[..., T]` | `!None`
298 :param make_object: optional function that will be used when loading a
299 composite type from the database if the Python type is not a sequence
300 compatible with the composite fields
301 :type make_object: `!Callable[[Sequence[Any], CompositeInfo], T]` | `!None`
302 :param make_sequence: optional function that will be used when dumping an
303 object to the database if the object is not a sequence compatible
304 with the composite fields
305 :type make_sequence: `!Callable[[T, CompositeInfo], Sequence[Any]]` | `!None`
306
307 .. note::
308
309 Registering the adapters doesn't affect objects already created, even
310 if they are children of the registered context. For instance,
311 registering the adapter globally doesn't affect already existing
312 connections.
313 """
314
315 # A friendly error warning instead of an AttributeError in case fetch()
316 # failed and it wasn't noticed.
317 if not info:
318 raise TypeError("no info passed. Is the requested composite available?")
319
320 # Register arrays and type info
321 info.register(context)
322
323 if not factory:
324 factory = cast("Callable[..., T]", _nt_from_info(info))
325
326 if not make_object:
327
328 def make_object(values: Sequence[Any], info: CompositeInfo) -> T:
329 return factory(*values)
330
331 adapters = context.adapters if context else postgres.adapters
332
333 # generate and register a customized text loader
334 loader: type[Loader] = _make_loader(info, make_object)
335 adapters.register_loader(info.oid, loader)
336
337 # generate and register a customized binary loader
338 loader = _make_binary_loader(info, make_object)
339 adapters.register_loader(info.oid, loader)
340
341 # If the factory is a type, create and register dumpers for it
342 if isinstance(factory, type):
343
344 # Optimistically assume that the factory type is a sequence.
345 # If it is not, it will create a non-functioning dumper, but we don't
346 # risk backward incompatibility.
347 if not make_sequence:
348
349 if not issubclass(factory, Sequence):
350 logger.warning(
351 "the type %r is not a sequence: dumping these objects to the"
352 " database will fail. Please specify a `make_sequence`"
353 " argument in the `register_composite()` call",
354 factory.__name__,
355 )
356
357 def make_sequence(obj: T, into: CompositeInfo) -> Sequence[Any]:
358 raise TypeError(
359 f"{type(obj).__name__!r} objects cannot be dumped without"
360 " specifying 'make_sequence' in 'register_composite()'"
361 )
362
363 else:
364
365 def make_sequence(obj: T, info: CompositeInfo) -> Sequence[Any]:
366 return obj # type: ignore[return-value] # it's a sequence
367
368 type_name = factory.__name__
369 dumper: type[Dumper] = _make_binary_dumper(type_name, info, make_sequence)
370 adapters.register_dumper(factory, dumper)
371
372 # Default to the text dumper because it is more flexible
373 dumper = _make_dumper(type_name, info, make_sequence)
374 adapters.register_dumper(factory, dumper)
375
376 info.python_type = factory
377
378 else:
379 if make_sequence:
380 raise TypeError(
381 "the factory {factory.__name__!r} is not a type: you cannot"
382 " create a dumper by specifying `make_sequence`."
383 )

400def _dump_text_sequence(seq: Sequence[Any], tx: abc.Transformer) -> bytes:
401 if not seq:
402 return b"()"
403
404 parts: list[abc.Buffer] = [b"("]
405
406 for item in seq:
407 if item is None:
408 parts.append(b",")
409 continue
410
411 dumper = tx.get_dumper(item, PyFormat.TEXT)
412 if (ad := dumper.dump(item)) is None:
413 ad = b""
414 elif not ad:
415 ad = b'""'
416 elif _re_needs_quotes.search(ad):
417 ad = b'"' + _re_esc.sub(rb"\1\1", ad) + b'"'
418
419 parts.append(ad)
420 parts.append(b",")
421
422 parts[-1] = b")"
423
424 return b"".join(parts)
425
426
427_re_needs_quotes = re.compile(rb'[",\\\s()]')
428_re_esc = re.compile(rb"([\\\"])")

431def _dump_binary_sequence(
432 seq: Sequence[Any],
433 types: Sequence[int],
434 formats: Sequence[PyFormat],
435 tx: abc.Transformer,
436) -> bytearray:
437 out = bytearray(pack_len(len(seq)))
438 adapted = tx.dump_sequence(seq, formats)
439 for i in range(len(seq)):
440 b = adapted[i]
441 oid = types[i]
442 if b is not None:
443 out += _pack_oidlen(oid, len(b))
444 out += b
445 else:
446 out += _pack_oidlen(oid, -1)
447
448 return out

451def _parse_text_record(data: abc.Buffer) -> list[bytes | None]:
452 """
453 Split a non-empty representation of a composite type into components.
454
455 Terminators shouldn't be used in `!data` (so that both record and range
456 representations can be parsed).
457 """
458 record: list[bytes | None] = []
459 for m in _re_tokenize.finditer(data):
460 if m.group(1):
461 record.append(None)
462 elif m.group(2) is not None:
463 record.append(_re_undouble.sub(rb"\1", m.group(2)))
464 else:
465 record.append(m.group(3))
466
467 # If the final group ended in `,` there is a final NULL in the record
468 # that the regexp couldn't parse.
469 if m and m.group().endswith(b","):
470 record.append(None)
471
472 return record
473
474
475_re_tokenize = re.compile(
476 rb"""(?x)
477 (,) # an empty token, representing NULL
478 | " ((?: [^"] | "")*) " ,? # or a quoted string
479 | ([^",)]+) ,? # or an unquoted string
480 """
481)
482_re_undouble = re.compile(rb'(["\\])\1')

485def _parse_binary_record(data: abc.Buffer) -> tuple[list[Buffer | None], list[int]]:
486 """
487 Parse the binary representation of a composite type.
488
489 Return the sequence of fields and oids found in the type. The fields
490 are returned as buffer: they will need a Transformer to be converted
491 to Python types.
492 """
493 nfields = unpack_len(data, 0)[0]
494 offset = 4
495 oids = []
496 record: list[Buffer | None] = []
497 for _ in range(nfields):
498 oid, length = _unpack_oidlen(data, offset)
499 offset += 8
500 oids.append(oid)
501 if length >= 0:
502 record.append(data[offset : offset + length])
503 offset += length
504 else:
505 record.append(None)
506
507 return record, oids
