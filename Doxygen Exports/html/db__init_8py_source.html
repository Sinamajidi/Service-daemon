47def drop_all_tables(db_conf) -> None:
48 """Drop all tables in the database (CASCADE to handle dependencies)."""
49 with psycopg.connect(**db_conf) as conn:
50 with conn.cursor() as cur:
51 # Drop all tables in the public schema
52 cur.execute("""
53 DO $$ DECLARE
54 r RECORD;
55 BEGIN
56 -- Drop all tables
57 FOR r IN (SELECT tablename FROM pg_tables WHERE schemaname = 'public') LOOP
58 EXECUTE 'DROP TABLE IF EXISTS ' || quote_ident(r.tablename) || ' CASCADE';
59 END LOOP;
60
61 -- Drop all sequences
62 FOR r IN (SELECT sequence_name FROM information_schema.sequences WHERE sequence_schema = 'public') LOOP
63 EXECUTE 'DROP SEQUENCE IF EXISTS ' || quote_ident(r.sequence_name) || ' CASCADE';
64 END LOOP;
65
66 -- Drop all types
67 FOR r IN (SELECT typname FROM pg_type WHERE typnamespace = (SELECT oid FROM pg_namespace WHERE nspname = 'public') AND typtype = 'e') LOOP
68 EXECUTE 'DROP TYPE IF EXISTS ' || quote_ident(r.typname) || ' CASCADE';
69 END LOOP;
70 END $$;
71 """)
72 conn.commit()
73 print("✓ All existing tables dropped")
