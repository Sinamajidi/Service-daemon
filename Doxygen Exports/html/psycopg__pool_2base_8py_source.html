190 """

204class AttemptWithBackoff:
205 """
206 Keep the state of a repeated operation attempt with exponential backoff.
207 """
208
209 INITIAL_DELAY = 1.0
210 DELAY_JITTER = 0.1
211 DELAY_BACKOFF = 2.0
212
213 def __init__(self, *, timeout: float):
214 self.timeout = timeout
215 self.delay = 0.0
216 self.give_up_at = 0.0
217
218 def update_delay(self, now: float) -> None:
219 """Calculate how long to wait for a new connection attempt"""
220 if self.delay == 0.0:
221 self.give_up_at = now + self.timeout
222 self.delay = BasePool._jitter(
223 self.INITIAL_DELAY, -self.DELAY_JITTER, self.DELAY_JITTER
224 )
225 else:
226 self.delay *= self.DELAY_BACKOFF
227
228 if self.delay + now > self.give_up_at:
229 self.delay = max(0.0, self.give_up_at - now)
230
231 def time_to_give_up(self, now: float) -> bool:
232 """Return True if we are tired of trying this attempt. Meh."""
233 return self.give_up_at > 0.0 and now >= self.give_up_at
