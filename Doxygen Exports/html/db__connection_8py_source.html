84 @contextmanager
85 def get_connection(self):
86 """
87 Context manager for getting a connection from the pool.
88 Automatically returns connection to pool when done.
89
90 Usage:
91 with db.get_connection() as conn:
92 # use connection
93 pass
94
95 Yields:
96 Database connection
97 """
98 self._ensure_pool() # Lazy initialization
99
100 if self._pool is None:
101 raise RuntimeError("Connection pool not initialized")
102
103 conn = None
104 try:
105 conn = self._pool.getconn()
106 yield conn
107 except psycopg2.Error as e:
108 if conn:
109 conn.rollback()
110 logger.error(f"Database error: {e}")
111 raise
112 finally:
113 if conn:
114 self._pool.putconn(conn)

118 """
