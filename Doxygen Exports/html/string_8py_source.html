44 """
45 Base class to dump a Python strings to a Postgres text type, in text format.

113 def load(self, data: Buffer) -> bytes | str:
114 if self._encoding:

129class BytesDumper(Dumper):
130 oid = _oids.BYTEA_OID
131 _qprefix = b""
132
133 def __init__(self, cls: type, context: AdaptContext | None = None):
134 super().__init__(cls, context)
135 self._esc = Escaping(self.connection.pgconn if self.connection else None)
136
137 def dump(self, obj: Buffer) -> Buffer | None:
138 return self._esc.escape_bytea(obj)
139
140 def quote(self, obj: Buffer) -> Buffer:
141 if (escaped := self.dump(obj)) is None:
142 return b"NULL"
143
144 # We cannot use the base quoting because escape_bytea already returns
145 # the quotes content. if scs is off it will escape the backslashes in
146 # the format, otherwise it won't, but it doesn't tell us what quotes to
147 # use.
148 if self.connection:
149 if not self._qprefix:
150 scs = self.connection.pgconn.parameter_status(
151 b"standard_conforming_strings"
152 )
153 self._qprefix = b"'" if scs == b"on" else b" E'"
154
155 return self._qprefix + escaped + b"'"
156
157 # We don't have a connection, so someone is using us to generate a file
158 # to use off-line or something like that. PQescapeBytea, like its
159 # string counterpart, is not predictable whether it will escape
160 # backslashes.
161 rv: bytes = b" E'" + escaped + b"'"
162 if self._esc.escape_bytea(b"\x00") == b"\\000":
163 rv = rv.replace(b"\\", b"\\\\")
164 return rv

197 # NOTE: the order the dumpers are registered is relevant. The last one
198 # registered becomes the default for each type. Usually, binary is the
199 # default dumper. For text we use the text dumper as default because it
200 # plays the role of unknown, and it can be cast automatically to other
201 # types. However, before that, we register dumper with 'text', 'varchar',
202 # 'name' oids, which will be used when a text dumper is looked up by oid.
