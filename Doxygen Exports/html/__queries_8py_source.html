112 def is_params_sequence(vars: Params) -> TypeGuard[Sequence[Any]]:
113 # Try concrete types, then abstract types
114 t = type(vars)
115 if t is list or t is tuple:
116 sequence = True
117 elif t is dict:
118 sequence = False
119 elif isinstance(vars, Sequence) and not isinstance(vars, (bytes, str)):
120 sequence = True
121 elif isinstance(vars, Mapping):
122 sequence = False
123 else:

126 f" got {type(vars).__qualname__}"

198def _query2pg_nocache(
199 query: bytes, encoding: str
200) -> tuple[bytes, list[PyFormat], list[str] | None, list[QueryPart]]:
201 """
202 Convert Python query and params into something Postgres understands.
203
204 - Convert Python placeholders (``%s``, ``%(name)s``) into Postgres
205 format (``$1``, ``$2``)
206 - placeholders can be %s, %t, or %b (auto, text or binary)
207 - return ``query`` (bytes), ``formats`` (list of formats) ``order``
208 (sequence of names used in the query, in the position they appear)
209 ``parts`` (splits of queries and placeholders).
210 """
211 parts = _split_query(query, encoding)
212 order: list[str] | None = None
213 chunks: list[bytes] = []
214 formats = []
215
216 if isinstance(parts[0].item, int):
217 for part in parts[:-1]:
218 assert isinstance(part.item, int)
219 chunks.append(part.pre)
220 chunks.append(b"$%d" % (part.item + 1))
221 formats.append(part.format)
222
223 elif isinstance(parts[0].item, str):
224 seen: dict[str, tuple[bytes, PyFormat]] = {}
225 order = []
226 for part in parts[:-1]:
227 assert isinstance(part.item, str)
228 chunks.append(part.pre)
229 if part.item not in seen:
230 ph = b"$%d" % (len(seen) + 1)
231 seen[part.item] = (ph, part.format)
232 order.append(part.item)
233 chunks.append(ph)
234 formats.append(part.format)
235 else:
236 if seen[part.item][1] != part.format:
237 raise e.ProgrammingError(
238 f"placeholder '{part.item}' cannot have different formats"
239 )
240 chunks.append(seen[part.item][0])
241
242 # last part
243 chunks.append(parts[-1].pre)
244
245 return b"".join(chunks), formats, order, parts
246
247
248# Note: the cache size is 128 items, but someone has reported throwing ~12k
249# queries (of type `INSERT ... VALUES (...), (...)` with a varying amount of
250# records), and the resulting cache size is >100Mb. So, we will avoid to cache
251# large queries or queries with a large number of params. See
252# https://github.com/sqlalchemy/sqlalchemy/discussions/10270
253_query2pg = lru_cache(_query2pg_nocache)

378def _split_query(
379 query: bytes, encoding: str = "ascii", collapse_double_percent: bool = True
380) -> list[QueryPart]:
381 parts: list[tuple[bytes, re.Match[bytes] | None]] = []
382 cur = 0
383
384 # pairs [(fragment, match], with the last match None
385 m = None
386 for m in _re_placeholder.finditer(query):
387 pre = query[cur : m.span(0)[0]]
388 parts.append((pre, m))
389 cur = m.span(0)[1]
390 if m:
391 parts.append((query[cur:], None))
392 else:
393 parts.append((query, None))
394
395 rv = []
396
397 # drop the "%%", validate
398 i = 0
399 phtype = None
400 while i < len(parts):
401 pre, m = parts[i]
402 if m is None:
403 # last part
404 rv.append(QueryPart(pre, 0, PyFormat.AUTO))
405 break
406
407 if (ph := m.group(0)) == b"%%":
408 # unescape '%%' to '%' if necessary, then merge the parts
409 if collapse_double_percent:
410 ph = b"%"
411 pre1, m1 = parts[i + 1]
412 parts[i + 1] = (pre + ph + pre1, m1)
413 del parts[i]
414 continue
415
416 if ph == b"%(":
417 raise e.ProgrammingError(
418 "incomplete placeholder:"
419 f" '{query[m.span(0)[0]:].split()[0].decode(encoding)}'"
420 )
421 elif ph == b"% ":
422 # explicit message for a typical error
423 raise e.ProgrammingError(
424 "incomplete placeholder: '%'; if you want to use '%' as an"
425 " operator you can double it up, i.e. use '%%'"
426 )
427 elif ph[-1:] not in b"sbt":
428 raise e.ProgrammingError(
429 "only '%s', '%b', '%t' are allowed as placeholders, got"
430 f" '{m.group(0).decode(encoding)}'"
431 )
432
433 # Index or name
434 item: int | str
435 item = m.group(1).decode(encoding) if m.group(1) else i
436
437 if not phtype:
438 phtype = type(item)
439 elif phtype is not type(item):
440 raise e.ProgrammingError(
441 "positional and named placeholders cannot be mixed"
442 )
443
444 format = _ph_to_fmt[ph[-1:]]
445 rv.append(QueryPart(pre, item, format))
446 i += 1
447
448 return rv
449
450
451_ph_to_fmt = {
452 b"s": PyFormat.AUTO,
453 b"t": PyFormat.TEXT,
454 b"b": PyFormat.BINARY,
455}
